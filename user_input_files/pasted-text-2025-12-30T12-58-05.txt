                   COMPUTE WS-CALC-TAX = 
                       WS-CALC-AMOUNT * WS-BRACKET-1-RATE
               WHEN WS-CALC-AMOUNT <= WS-BRACKET-2-MAX
                   COMPUTE WS-CALC-TAX = 
                       (WS-BRACKET-1-MAX * WS-BRACKET-1-RATE) +
                       ((WS-CALC-AMOUNT - WS-BRACKET-1-MAX) * 
                        WS-BRACKET-2-RATE)
               WHEN WS-CALC-AMOUNT <= WS-BRACKET-3-MAX
                   COMPUTE WS-CALC-TAX = 
                       (WS-BRACKET-1-MAX * WS-BRACKET-1-RATE) +
                       ((WS-BRACKET-2-MAX - WS-BRACKET-1-MAX) * 
                        WS-BRACKET-2-RATE) +
                       ((WS-CALC-AMOUNT - WS-BRACKET-2-MAX) * 
                        WS-BRACKET-3-RATE)
               WHEN OTHER
                   COMPUTE WS-CALC-TAX = 
                       WS-CALC-AMOUNT * WS-BRACKET-5-RATE
           END-EVALUATE."""
        match True:
            case self.WS_CALC_AMOUNT <= self.WS_BRACKET_1_MAX:
                self.WS_CALC_TAX = self.WS_CALC_AMOUNT * self.WS_BRACKET_1_RATE
            case self.WS_CALC_AMOUNT <= self.WS_BRACKET_2_MAX:
                self.WS_CALC_TAX = (self.WS_BRACKET_1_MAX * self.WS_BRACKET_1_RATE) + \
                                    ((self.WS_CALC_AMOUNT - self.WS_BRACKET_1_MAX) * self.WS_BRACKET_2_RATE)
            case self.WS_CALC_AMOUNT <= self.WS_BRACKET_3_MAX:
                self.WS_CALC_TAX = (self.WS_BRACKET_1_MAX * self.WS_BRACKET_1_RATE) + \
                                    ((self.WS_BRACKET_2_MAX - self.WS_BRACKET_1_MAX) * self.WS_BRACKET_2_RATE) + \
                                    ((self.WS_CALC_AMOUNT - self.WS_BRACKET_2_MAX) * self.WS_BRACKET_3_RATE)
            case _:
                self.WS_CALC_TAX = self.WS_CALC_AMOUNT * self.WS_BRACKET_5_RATE

    def calculate_premium(self) -> None:
        """
        COBOL Equivalent:
        EVALUATE TRUE
              WHEN 6 THRU 10
                 ADD 100 TO WS-BASE-PREMIUM
              WHEN OTHER
                 ADD 50 TO WS-BASE-PREMIUM
           END-EVALUATE
           IF WS-DRIVER-AGE < 25
              MULTIPLY 1.5 BY WS-BASE-PREMIUM
           END-IF
           IF WS-ACCIDENTS-3YR > 0
              COMPUTE WS-ACCIDENT-SURCHARGE = 
                 WS-ACCIDENTS-3YR * 200
              ADD WS-ACCIDENT-SURCHARGE TO WS-BASE-PREMIUM
           END-IF
           IF WS-VIOLATIONS-3YR > 0
              COMPUTE WS-VIOLATION-SURCHARGE = 
                 WS-VIOLATIONS-3YR * 100
              ADD WS-VIOLATION-SURCHARGE TO WS-BASE-PREMIUM
           END-IF
           MOVE WS-BASE-PREMIUM TO WS-ANNUAL-PREMIUM
           COMPUTE WS-MONTHLY-PREMIUM = 
              WS-ANNUAL-PREMIUM / 12.
        """
        match True:
            case 6 <= self.WS_DRIVER_AGE <= 10:
                self.WS_BASE_PREMIUM += Decimal("100.00")
            case _:
                self.WS_BASE_PREMIUM += Decimal("50.00")

        if self.WS_DRIVER_AGE < 25:
            self.WS_BASE_PREMIUM *= Decimal("1.5")

        if self.WS_ACCIDENTS_3YR > 0:
            self.WS_ACCIDENT_SURCHARGE = Decimal(self.WS_ACCIDENTS_3YR * 200)
            self.WS_BASE_PREMIUM += self.WS_ACCIDENT_SURCHARGE

        if self.WS_VIOLATIONS_3YR > 0:
            self.WS_VIOLATION_SURCHARGE = Decimal(self.WS_VIOLATIONS_3YR * 100)
            self.WS_BASE_PREMIUM += self.WS_VIOLATION_SURCHARGE

        self.WS_ANNUAL_PREMIUM = self.WS_BASE_PREMIUM
        self.WS_MONTHLY_PREMIUM = self.WS_ANNUAL_PREMIUM / Decimal("12.0")

    def calc_home_premium(self) -> None:
        """13230-CALC-HOME-PREMIUM.
           COMPUTE WS-BASE-PREMIUM = 
              WS-COVERAGE-AMOUNT * 0.003
           EVALUATE WS-HOME-AGE
              WHEN 0 THRU 10
                 MULTIPLY 0.9 BY WS-BASE-PREMIUM
              WHEN 11 THRU 25
                 MULTIPLY 1.0 BY WS-BASE-PREMIUM
              WHEN 26 THRU 50
                 MULTIPLY 1.2 BY WS-BASE-PREMIUM
              WHEN OTHER
                 MULTIPLY 1.5 BY WS-BASE-PREMIUM
           END-EVALUATE
           IF WS-FLOOD-ZONE = 'Y'
              MULTIPLY 1.5 BY WS-BASE-PREMIUM
           END-IF
           IF WS-SECURITY-SYSTEM = 'Y'
              MULTIPLY 0.9 BY WS-BASE-PREMIUM
           END-IF
           COMPUTE WS-DEDUCTIBLE-CREDIT = 
              WS-DEDUCTIBLE / 1000 * 50
           SUBTRACT WS-DEDUCTIBLE-CREDIT FROM WS-BASE-PREMIUM
           IF WS-BASE-PREMIUM < 200
              MOVE 200 TO WS-BASE-PREMIUM
           END-IF
           MOVE WS-BASE-PREMIUM TO WS-ANNUAL-PREMIUM
           COMPUTE WS-MONTHLY-PREMIUM = 
              WS-ANNUAL-PREMIUM / 12."""
        self.WS_BASE_PREMIUM = self.WS_COVERAGE_AMOUNT * Decimal("0.003")
        match True:
            case 0 <= self.WS_HOME_AGE <= 10:
                self.WS_BASE_PREMIUM *= Decimal("0.9")
            case 11 <= self.WS_HOME_AGE <= 25:
                self.WS_BASE_PREMIUM *= Decimal("1.0")
            case 26 <= self.WS_HOME_AGE <= 50:
                self.WS_BASE_PREMIUM *= Decimal("1.2")
            case _:
                self.WS_BASE_PREMIUM *= Decimal("1.5")

        if self.WS_FLOOD_ZONE == 'Y':
            self.WS_BASE_PREMIUM *= Decimal("1.5")
        if self.WS_SECURITY_SYSTEM == 'Y':
            self.WS_BASE_PREMIUM *= Decimal("0.9")

        self.WS_DEDUCTIBLE_CREDIT = self.WS_DEDUCTIBLE / Decimal("1000.0") * Decimal("50.0")
        self.WS_BASE_PREMIUM -= self.WS_DEDUCTIBLE_CREDIT

        if self.WS_BASE_PREMIUM < Decimal("200.0"):
            self.WS_BASE_PREMIUM = Decimal("200.0")

        self.WS_ANNUAL_PREMIUM = self.WS_BASE_PREMIUM
        self.WS_MONTHLY_PREMIUM = self.WS_ANNUAL_PREMIUM / Decimal("12.0")

    def calc_health_premium(self) -> None:
        """13240-CALC-HEALTH-PREMIUM.
           MOVE 300 TO WS-BASE-PREMIUM
           EVALUATE WS-INSURED-AGE

                 MULTIPLY 0.5 BY WS-BASE-PREMIUM
              WHEN 19 THRU 30
                 MULTIPLY 1.0 BY WS-BASE-PREMIUM

        if self.WS_FAMILY_PLAN == 'Y':
            self.WS_BASE_PREMIUM *= Decimal("2.5")

        self.WS_MONTHLY_PREMIUM = self.WS_BASE_PREMIUM

        """13300-UNDERWRITING.

    def evaluate_risk_factors(self) -> None:
        """13310-EVALUATE-RISK-FACTORS.
           MOVE ZEROES TO WS-RISK-POINTS
           IF POLICY-LIFE
              IF WS-BMI > 30
                 ADD 10 TO WS-RISK-POINTS
              END-IF
              IF WS-SMOKER-FLAG = 'Y'
                 ADD 25 TO WS-RISK-POINTS
              END-IF
              IF WS-HAZARDOUS-OCCUPATION = 'Y'
                 ADD 15 TO WS-RISK-POINTS
              END-IF
           END-IF
           IF POLICY-AUTO
              IF WS-DRIVER-AGE < 21
                 ADD 20 TO WS-RISK-POINTS
              END-IF
              IF WS-ACCIDENTS-3YR > 1
                 ADD 15 TO WS-RISK-POINTS
              END-IF
           END-IF."""
        self.WS_RISK_POINTS = 0
        if self.POLICY_LIFE:
            if self.WS_BMI > Decimal("30.0"):
                self.WS_RISK_POINTS += 10
            if self.WS_SMOKER_FLAG == 'Y':
                self.WS_RISK_POINTS += 25
            if self.WS_HAZARDOUS_OCCUPATION == 'Y':
                self.WS_RISK_POINTS += 15
        if self.POLICY_AUTO:
            if self.WS_DRIVER_AGE < 21:
                self.WS_RISK_POINTS += 20
            if self.WS_ACCIDENTS_3YR > 1:
                self.WS_RISK_POINTS += 15

    def check_medical_history(self) -> None:
        """13320-CHECK-MEDICAL-HISTORY.
           IF WS-CHRONIC-CONDITIONS > 0
              COMPUTE WS-CONDITION-POINTS = 
                 WS-CHRONIC-CONDITIONS * 5
              ADD WS-CONDITION-POINTS TO WS-RISK-POINTS
           END-IF
           IF WS-RECENT-HOSPITALIZATION = 'Y'
              ADD 10 TO WS-RISK-POINTS
           END-IF
           IF WS-PRESCRIPTION-COUNT > 5
              ADD 5 TO WS-RISK-POINTS
           END-IF."""
        if self.WS_CHRONIC_CONDITIONS > 0:
            self.WS_CONDITION_POINTS = Decimal(self.WS_CHRONIC_CONDITIONS * 5)
            self.WS_RISK_POINTS += int(self.WS_CONDITION_POINTS)
        if self.WS_RECENT_HOSPITALIZATION == 'Y':
            self.WS_RISK_POINTS += 10
        if self.WS_PRESCRIPTION_COUNT > 5:
            self.WS_RISK_POINTS += 5

        """13335-CHECK-FRAUD-INDICATORS.
           IF WS-RECENT-CLAIMS > 3

              MOVE 'Y' TO WS-FRAUD-FLAG
           END-IF
           IF WS-ADDRESS-MISMATCH = 'Y'

           END-IF."""
        if self.WS_RECENT_CLAIMS > 3:

           IF WS-DOC-MISSING = 'Y'

    def determine_decision(self) -> None:
        """13340-DETERMINE-DECISION.
           EVALUATE TRUE

    def calculate_match_score(self):
        """16120-CALCULATE-MATCH-SCORE."""
        if self.WS_OFAC_SCORE > 0:
            self.WS_MATCH_SCORE += self.WS_OFAC_SCORE
        if self.WS_PEP_SCORE > 0:
            self.WS_MATCH_SCORE += self.WS_PEP_SCORE
        self.WS_MATCH_SCORE = self.WS_MATCH_SCORE / self.WS_WATCHLIST_HITS

        """16130-DETERMINE-DISPOSITION."""
        match True:
            case self.WS_MATCH_SCORE >= 90:
                self.WS_MATCH_TYPE = 'CONFIRMED'
                self.WS_SAR_REQUIRED = 'Y'
            case self.WS_MATCH_SCORE >= 75:
                self.WS_MATCH_TYPE = 'POTENTIAL'
                self.WS_CASE_STATUS = 'REVIEW'
            case self.WS_MATCH_SCORE >= 50:
                self.WS_MATCH_TYPE = 'WEAK'
                self.WS_CASE_STATUS = 'CLEARED'

                self.WS_MATCH_TYPE = 'FALSE POSITIVE'
                self.WS_CASE_STATUS = 'CLEARED'

            self.WS_PATTERN_FLAG = 'Y'

            self.WS_LOCATION_FLAG = 'Y'
            self.WS_FRAUD_SCORE += 25
        if self.WS_NEW_DEVICE == 'Y':
            self.WS_DEVICE_FLAG = 'Y'
            self.WS_FRAUD_SCORE += 10

    def calculate_risk_score(self):
        """16440-CALCULATE-RISK-SCORE."""

                self.WS_FRAUD_DECISION = 'BLOCK'

            case self.WS_FRAUD_SCORE >= 60:
                self.WS_FRAUD_DECISION = 'REVIEW'

            case self.WS_FRAUD_SCORE >= 40:
                self.WS_FRAUD_DECISION = 'MONITOR'
            case _:
                self.WS_FRAUD_DECISION = 'APPROVE'

        match self.WS_CASE_TYPE:
            case 'BILLING INQUIRY':

                self.WS_CASE_PRIORITY = 1
            case 'ACCOUNT ACCESS':
                self.WS_CASE_PRIORITY = 1
            case 'GENERAL INQUIRY':
                self.WS_CASE_PRIORITY = 3
            case _:
                self.WS_CASE_PRIORITY = 3
        self.WS_TARGET_DATE = self.integer_of_date(self.WS_OPEN_DATE) + self.WS_CASE_PRIORITY * 2

        """17324-CHECK-PREVIOUS-CASES."""
        self.CASE_SEARCH_KEY = self.WS_CUSTOMER_ID
        self.WS_EOF_FLAG = 'N'
        while self.WS_EOF_FLAG == 'N': #PERFORM UNTIL WS-EOF-FLAG = 'Y'
            case_file = self.read_case_file(self.CASE_SEARCH_KEY) #READ CASE-FILE INTO WS-PREVIOUS-CASE KEY IS CASE-CUSTOMER
            if case_file is None: #AT END
                self.WS_EOF_FLAG = 'Y'
            else: #NOT AT END
                self.WS_PREVIOUS_CASE_COUNT += 1
        self.WS_EOF_FLAG = 'N'

def calculate_simple_interest(ws_account_balance: Decimal, ws_interest_rate: Decimal, ws_days_in_period: int) -> Decimal:
    """
    Calculates simple interest.
    7200-CALCULATE-SIMPLE-INTEREST.
    """
    #7200 COMPUTE WS-SIMPLE-INTEREST =
    #7200    WS-ACCOUNT-BALANCE * WS-INTEREST-RATE
    #7200    * WS-DAYS-IN-PERIOD / 36500.
    ws_simple_interest = ws_account_balance * ws_interest_rate * Decimal(ws_days_in_period) / Decimal('36500')
    return ws_simple_interest

def calculate_compound_interest(ws_account_balance: Decimal, ws_interest_rate: Decimal, ws_days_in_period: int) -> Decimal:
    """
    Calculates compound interest.
    7300-CALCULATE-COMPOUND-INTEREST.
    """
    #7300 COMPUTE WS-COMPOUND-FACTOR =
    #7300    (1 + WS-INTEREST-RATE / 36500)
    #7300    ** WS-DAYS-IN-PERIOD
    ws_compound_factor = (1 + ws_interest_rate / Decimal('36500')) ** ws_days_in_period
    #7300 COMPUTE WS-COMPOUND-INTEREST =
    #7300    WS-ACCOUNT-BALANCE * (WS-COMPOUND-FACTOR - 1).
    ws_compound_interest = ws_account_balance * (Decimal(ws_compound_factor) - 1)
    return ws_compound_interest

def calculate_monthly_fee(ws_account_type: str) -> Decimal:
    """
    Calculates the monthly fee based on account type.
    8100-CALCULATE-MONTHLY-FEE.
    """
    #8100 EVALUATE WS-ACCOUNT-TYPE
    match ws_account_type:
        #8100    WHEN 'CHK'
        case 'CHK':
            #8100       MOVE 12.00 TO WS-MONTHLY-FEE
            ws_monthly_fee = Decimal('12.00')
        #8100    WHEN 'SAV'
        case 'SAV':
            #8100       MOVE 5.00 TO WS-MONTHLY-FEE
            ws_monthly_fee = Decimal('5.00')
        #8100    WHEN 'PRM'
        case 'PRM':
            #8100       MOVE 25.00 TO WS-MONTHLY-FEE
            ws_monthly_fee = Decimal('25.00')
        #8100    WHEN OTHER
        case _:
            #8100       MOVE 0.00 TO WS-MONTHLY-FEE
            ws_monthly_fee = Decimal('0.00')
    #8100 END-EVALUATE.
    return ws_monthly_fee

def calculate_transaction_fees(ws_trans_count: int, ws_free_trans_limit: int, ws_per_trans_fee: Decimal) -> Decimal:
    """
    Calculates transaction fees.
    8200-CALCULATE-TRANSACTION-FEES.
    """
    #8200 IF WS-TRANS-COUNT > WS-FREE-TRANS-LIMIT
    if ws_trans_count > ws_free_trans_limit:
        #8200    COMPUTE WS-EXCESS-TRANS =
        #8200       WS-TRANS-COUNT - WS-FREE-TRANS-LIMIT
        ws_excess_trans = ws_trans_count - ws_free_trans_limit
        #8200    COMPUTE WS-TRANS-FEE =
        #8200       WS-EXCESS-TRANS * WS-PER-TRANS-FEE
        ws_trans_fee = ws_excess_trans * ws_per_trans_fee
    else:
        #8200    MOVE ZEROES TO WS-TRANS-FEE
        ws_trans_fee = Decimal('0.00')
    #8200 END-IF.
    return ws_trans_fee

def apply_fee_waivers(ws_account_balance: Decimal, ws_min_balance_waiver: Decimal, ws_customer_tier: str, ws_trans_fee: Decimal, ws_monthly_fee: Decimal) -> tuple[Decimal, Decimal]:
    """
    Applies fee waivers based on account balance and customer tier.
    8300-APPLY-FEE-WAIVERS.
    """
    #8300 IF WS-ACCOUNT-BALANCE >= WS-MIN-BALANCE-WAIVER
    if ws_account_balance >= ws_min_balance_waiver:
        #8300    MOVE ZEROES TO WS-MONTHLY-FEE
        ws_monthly_fee = Decimal('0.00')
    #8300 END-IF
    #8300 IF WS-CUSTOMER-TIER = 'GOLD' OR 'PLATINUM'
    if ws_customer_tier == 'GOLD' or ws_customer_tier == 'PLATINUM':
        #8300    COMPUTE WS-TRANS-FEE = WS-TRANS-FEE * 0.5
        ws_trans_fee = ws_trans_fee * Decimal('0.5')
    #8300 END-IF.
    return ws_trans_fee, ws_monthly_fee

def deduct_fees(ws_monthly_fee: Decimal, ws_trans_fee: Decimal, ws_account_balance: Decimal, update_account, record_fee_transaction) -> Decimal:
    """
    Deducts fees from the account balance.
    8400-DEDUCT-FEES.
    """
    #8400 COMPUTE WS-TOTAL-FEES =
    #8400    WS-MONTHLY-FEE + WS-TRANS-FEE
    ws_total_fees = ws_monthly_fee + ws_trans_fee
    #8400 SUBTRACT WS-TOTAL-FEES FROM WS-ACCOUNT-BALANCE
    ws_account_balance -= ws_total_fees
    #8400 PERFORM 2350-UPDATE-ACCOUNT
    update_account()
    #8400 PERFORM 8450-RECORD-FEE-TRANSACTION.
    record_fee_transaction(ws_total_fees)
    return ws_account_balance

def calculate_next_run_date(ws_last_run_date: int, schedule_type: str) -> int:
    """Original COBOL lines:
                    FUNCTION INTEGER-OF-DATE(WS-LAST-RUN-DATE) + 1
              WHEN 'WEEKLY'
                 COMPUTE WS-NEXT-RUN-DATE =

                 COMPUTE WS-NEXT-RUN-DATE =

def collect_transaction_metrics(self):
    """Original COBOL lines:
       21110-COLLECT-TRANSACTION-METRICS.
           MOVE ZEROES TO WS-TOTAL-TRANS-AMOUNT
           MOVE ZEROES TO WS-TOTAL-TRANS-COUNT
           MOVE ZEROES TO WS-AVG-TRANS-AMOUNT
           PERFORM UNTIL WS-EOF-FLAG = 'Y'
              READ TRANSACTION-FILE INTO WS-TRANS-REC
                 AT END
                    MOVE 'Y' TO WS-EOF-FLAG
                 NOT AT END
                    ADD 1 TO WS-TOTAL-TRANS-COUNT
                    ADD TRANS-AMOUNT TO WS-TOTAL-TRANS-AMOUNT
              END-READ
           END-PERFORM
           IF WS-TOTAL-TRANS-COUNT > 0
              COMPUTE WS-AVG-TRANS-AMOUNT = 
                 WS-TOTAL-TRANS-AMOUNT / WS-TOTAL-TRANS-COUNT
           END-IF
           MOVE 'N' TO WS-EOF-FLAG."""
    self.ws_total_trans_amount = Decimal('0')
    self.ws_total_trans_count = 0
    self.ws_avg_trans_amount = Decimal('0')
    self.ws_eof_flag = 'N'
    while self.ws_eof_flag == 'N':
        try:
            ws_trans_rec: TransactionRecord = self.read_transaction_file()
            self.ws_total_trans_count += 1
            self.ws_total_trans_amount += ws_trans_rec.trans_amount
        except EOFError:
            self.ws_eof_flag = 'Y'
    if self.ws_total_trans_count > 0:
        self.ws_avg_trans_amount = self.ws_total_trans_amount / self.ws_total_trans_count
    self.ws_eof_flag = 'N'

def collect_customer_metrics(self):
    """Original COBOL lines:
       21120-COLLECT-CUSTOMER-METRICS.
           MOVE ZEROES TO WS-ACTIVE-CUSTOMERS
           MOVE ZEROES TO WS-NEW-CUSTOMERS
           MOVE ZEROES TO WS-CHURNED-CUSTOMERS
           PERFORM UNTIL WS-EOF-FLAG = 'Y'
              READ CUSTOMER-FILE INTO WS-CUST-REC
                 AT END
                    MOVE 'Y' TO WS-EOF-FLAG
                 NOT AT END
                    IF CUST-STATUS = 'A'
                       ADD 1 TO WS-ACTIVE-CUSTOMERS
                    END-IF
                    IF CUST-OPEN-DATE >= WS-PERIOD-START
                       ADD 1 TO WS-NEW-CUSTOMERS
                    END-IF
                    IF CUST-CLOSE-DATE >= WS-PERIOD-START
                       ADD 1 TO WS-CHURNED-CUSTOMERS
                    END-IF
              END-READ
           END-PERFORM
           MOVE 'N' TO WS-EOF-FLAG."""
    self.ws_active_customers = 0
    self.ws_new_customers = 0
    self.ws_churned_customers = 0
    self.ws_eof_flag = 'N'
    while self.ws_eof_flag == 'N':
        try:
            ws_cust_rec: CustomerRecord = self.read_customer_file()
            if ws_cust_rec.cust_status == 'A':
                self.ws_active_customers += 1
            if ws_cust_rec.cust_open_date >= self.ws_period_start:
                self.ws_new_customers += 1
            if ws_cust_rec.cust_close_date >= self.ws_period_start:
                self.ws_churned_customers += 1
        except EOFError:
            self.ws_eof_flag = 'Y'
    self.ws_eof_flag = 'N'

    """Original COBOL lines:

    self.calc_financial_kpi()
    self.calc_operational_kpi()
    self.calc_customer_kpi()

def calc_financial_kpi(self):
    """Original COBOL lines:
       21310-CALC-FINANCIAL-KPI.
           IF WS-TOTAL-ASSETS > 0
              COMPUTE WS-ROA = 
                 (WS-NET-INCOME / WS-TOTAL-ASSETS) * 100
           END-IF
           IF WS-TOTAL-EQUITY > 0
              COMPUTE WS-ROE = 
                 (WS-NET-INCOME / WS-TOTAL-EQUITY) * 100
           END-IF
           IF WS-INTEREST-EXPENSE > 0
              COMPUTE WS-NIM = 
                 ((WS-INTEREST-INCOME - WS-INTEREST-EXPENSE) /
                  WS-EARNING-ASSETS) * 100
           END-IF."""
    if self.ws_total_assets > 0:
        self.ws_roa = (self.ws_net_income / self.ws_total_assets) * 100
    if self.ws_total_equity > 0:
        self.ws_roe = (self.ws_net_income / self.ws_total_equity) * 100
    if self.ws_interest_expense > 0:
        self.ws_nim = ((self.ws_interest_income - self.ws_interest_expense) /
                       self.ws_earning_assets) * 100

def calc_operational_kpi(self):
    """Original COBOL lines:
       21320-CALC-OPERATIONAL-KPI.
           IF WS-TOTAL-TRANS-COUNT > 0
              COMPUTE WS-ERROR-RATE = 
                 (WS-ERROR-COUNT / WS-TOTAL-TRANS-COUNT) * 100
           END-IF
           COMPUTE WS-SLA-COMPLIANCE = 
              (WS-WITHIN-SLA-COUNT / WS-TOTAL-CASES) * 100
           COMPUTE WS-FIRST-CALL-RESOLUTION = 
              (WS-FCR-COUNT / WS-TOTAL-CALLS) * 100."""
    if self.ws_total_trans_count > 0:
        self.ws_error_rate = (self.ws_error_count / self.ws_total_trans_count) * 100
    self.ws_sla_compliance = (self.ws_within_sla_count / self.ws_total_cases) * 100
    self.ws_first_call_resolution = (self.ws_fcr_count / self.ws_total_calls) * 100

def calc_customer_kpi(self):
    """Original COBOL lines:
       21330-CALC-CUSTOMER-KPI.
           IF WS-ACTIVE-CUSTOMERS > 0
              COMPUTE WS-CHURN-RATE = 
                 (WS-CHURNED-CUSTOMERS / WS-ACTIVE-CUSTOMERS) * 100
           END-IF
           COMPUTE WS-ACQUISITION-COST = 
              WS-MARKETING-SPEND / WS-NEW-CUSTOMERS
           COMPUTE WS-LIFETIME-VALUE = 
              WS-AVG-REVENUE-PER-CUSTOMER * WS-AVG-CUSTOMER-TENURE."""
    if self.ws_active_customers > 0:
        self.ws_churn_rate = (self.ws_churned_customers / self.ws_active_customers) * 100
    self.ws_acquisition_cost = self.ws_marketing_spend / self.ws_new_customers
    self.ws_lifetime_value = self.ws_avg_revenue_per_customer * self.ws_avg_customer_tenure

def check_activity(self, ws_account_rec: 'AccountRecord'):
    """Original COBOL lines:
       22110-CHECK-ACTIVITY.
           COMPUTE WS-DAYS-INACTIVE = 
              FUNCTION INTEGER-OF-DATE(WS-PROCESS-DATE) -
              FUNCTION INTEGER-OF-DATE(ACCT-LAST-ACTIVITY)
           IF WS-DAYS-INACTIVE > 365
              MOVE 'D' TO ACCT-STATUS
              PERFORM 22120-MARK-DORMANT
           END-IF."""
    ws_days_inactive = self.integer_of_date(self.ws_process_date) - self.integer_of_date(ws_account_rec.acct_last_activity)
    if ws_days_inactive > 365:
        ws_account_rec.acct_status = 'D'
        self.mark_dormant(ws_account_rec)

    def perform_32000_treasury_management(self) -> None:
        """32000-TREASURY-MANAGEMENT."""
        self.perform_32100_calculate_cash_position()
        self.perform_32200_project_cash_flows()
        self.perform_32300_manage_reserves()
        self.perform_32400_manage_investments()
        self.perform_32500_manage_borrowings()

    def perform_32100_calculate_cash_position(self) -> None:
        """32100-CALCULATE-CASH-POSITION."""
        self.ws_treasury_management.ws_cash_position = Decimal('0.00')
        self.perform_32110_sum_vault_cash()
        self.perform_32120_sum_fed_account()
        self.perform_32130_sum_correspondent_balances()

    def perform_32110_sum_vault_cash(self) -> None:
        """32110-SUM-VAULT-CASH."""
        self.ws_eof_flag = 'N'
        while self.ws_eof_flag != 'Y':
            try:
                ws_vault_rec = self.read_vault_cash_file()
                self.ws_treasury_management.ws_cash_position += ws_vault_rec.vault_balance
            except StopIteration:
                self.ws_eof_flag = 'Y'
        self.ws_eof_flag = 'N'

    def perform_32120_sum_fed_account(self) -> None:
        """32120-SUM-FED-ACCOUNT."""
        ws_fed_balance = self.read_fed_account_file()
        self.ws_treasury_management.ws_cash_position += ws_fed_balance.ws_fed_balance

    def perform_32130_sum_correspondent_balances(self) -> None:
        """32130-SUM-CORRESPONDENT-BALANCES."""
        self.ws_eof_flag = 'N'
        while self.ws_eof_flag != 'Y':
            try:
                ws_corr_rec = self.read_correspondent_file()
                self.ws_treasury_management.ws_cash_position += ws_corr_rec.corr_balance
            except StopIteration:
                self.ws_eof_flag = 'Y'
        self.ws_eof_flag = 'N'

    def perform_32200_project_cash_flows(self) -> None:
        """32200-PROJECT-CASH-FLOWS."""
        self.ws_treasury_management.ws_projected_inflows = Decimal('0.00')
        self.ws_treasury_management.ws_projected_outflows = Decimal('0.00')
        self.perform_32210_project_loan_payments()
        self.perform_32220_project_deposit_flows()
        self.perform_32230_project_investment_maturities()
        self.ws_treasury_management.ws_net_position = (
            self.ws_treasury_management.ws_cash_position +
            self.ws_treasury_management.ws_projected_inflows -
            self.ws_treasury_management.ws_projected_outflows
        )

    def perform_32210_project_loan_payments(self) -> None:
        """32210-PROJECT-LOAN-PAYMENTS."""
        self.ws_eof_flag = 'N'
        while self.ws_eof_flag != 'Y':
            try:
                ws_loan_pmt_rec = self.read_loan_schedule_file()
                if ws_loan_pmt_rec.loan_pmt_date <= self.ws_projection_date:
                    self.ws_treasury_management.ws_projected_inflows += ws_loan_pmt_rec.loan_pmt_amount
            except StopIteration:
                self.ws_eof_flag = 'Y'
        self.ws_eof_flag = 'N'

    def perform_32220_project_deposit_flows(self) -> None:
        """32220-PROJECT-DEPOSIT-FLOWS."""
        ws_expected_deposits = self.ws_avg_daily_deposits * self.ws_projection_days
        ws_expected_withdrawals = self.ws_avg_daily_withdrawals * self.ws_projection_days
        self.ws_treasury_management.ws_projected_inflows += ws_expected_deposits
        self.ws_treasury_management.ws_projected_outflows += ws_expected_withdrawals

    def perform_32230_project_investment_maturities(self) -> None:
        """32230-PROJECT-INVESTMENT-MATURITIES."""
        self.ws_eof_flag = 'N'
        while self.ws_eof_flag != 'Y':
            try:
                ws_inv_rec = self.read_investment_file()
                if ws_inv_rec.inv_maturity_date <= self.ws_projection_date:
                    self.ws_treasury_management.ws_projected_inflows += ws_inv_rec.inv_par_value
            except StopIteration:
                self.ws_eof_flag = 'Y'
        self.ws_eof_flag = 'N'
