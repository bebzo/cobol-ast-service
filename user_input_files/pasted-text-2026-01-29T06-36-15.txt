PATCH_1_EVALUATE_HANDLER = """
# Fichier: transpiler/evaluate_handler.py
# Ligne: ~85-120 (section handle_evaluate_when)

class EvaluateHandler:
    '''Handles COBOL EVALUATE statement transpilation'''
    
    def handle_evaluate_true(self, evaluate_node):
        '''
        Corrige la génération de EVALUATE TRUE.
        
        COBOL:
            EVALUATE TRUE
            WHEN TRX-AMOUNT < 0.01
                MOVE 'ERR-2100' TO ERROR-CODE
            WHEN TRX-AMOUNT > TRANSACTION-LIMIT
                MOVE 'ERR-2101' TO ERROR-CODE
            WHEN TRX-TYPE = 'WIRE'
                PERFORM CHECK-LARGE-TRANSACTION
            END-EVALUATE
        
        AVANT (BUGUÉ):
            if True:
                self.error_code = 'ERR-2100'
            elif True:
                self.error_code = 'ERR-2101'
        
        APRÈS (CORRIGÉ):
            if self.trx_amount < Decimal('0.01'):
                self.error_code = 'ERR-2100'
            elif self.trx_amount > self.transaction_limit:
                self.error_code = 'ERR-2101'
        '''
        conditions = []
        
        for when_clause in evaluate_node.when_clauses:
            # ❌ ANCIEN CODE (BUGUÉ)
            # condition = "True"  # BUG: Ignore la condition réelle
            
            # ✅ NOUVEAU CODE (CORRIGÉ)
            condition = self.parse_when_condition(when_clause.condition)
            
            actions = self.generate_actions(when_clause.statements)
            conditions.append((condition, actions))
        
        # Générer le if/elif/else
        return self.build_conditional_chain(conditions)
    
    def parse_when_condition(self, condition_node):
        '''
        Parse la condition WHEN et la convertit en Python.
        
        AJOUT CRITIQUE: Cette méthode manquait complètement.
        '''
        from .condition_mapper import ConditionMapper
        
        mapper = ConditionMapper()
        
        # Gérer les différents types de conditions
        if condition_node.type == 'COMPARISON':
            left = mapper.map_variable(condition_node.left)
            operator = mapper.map_operator(condition_node.operator)
            right = mapper.map_value(condition_node.right)
            return f"{left} {operator} {right}"
        
        elif condition_node.type == 'LITERAL_TEST':
            # WHEN 'WIRE' devient self.trx_type == 'WIRE'
            # ❌ AVANT: return f"'{condition_node.value}'"  # BUG!
            # ✅ APRÈS:
            var = self.infer_variable_from_context(condition_node.value)
            return f"{var} == '{condition_node.value}'"
        
        elif condition_node.type == 'RANGE':
            # WHEN 300 THRU 579
            var = mapper.map_variable(condition_node.variable)
            low = mapper.map_value(condition_node.low)
            high = mapper.map_value(condition_node.high)
            return f"({low} <= {var} <= {high})"
        
        else:
            # Fallback sécurisé
            self.logger.warning(f"Unknown WHEN condition type: {condition_node.type}")
            return f"True  # TODO: Implement {condition_node.type}"
    
    def infer_variable_from_context(self, literal_value):
        '''
        Infère la variable à comparer depuis le contexte.
        
        AJOUT CRITIQUE: Détecte le type de littéral et trouve la variable.
        '''
        # Patterns de transaction types
        if literal_value in ['WIRE', 'ACH', 'CASH', 'DEPO', 'WTHD', 'XFER', 'FEE']:
            return 'self.trx_type'
        
        # Patterns de statuts
        elif literal_value in ['A', 'I', 'C', 'P', 'S']:
            return 'self.status'
        
        # Patterns de secteurs
        elif literal_value in ['TECH', 'FINS', 'ENRG', 'HLTH', 'CONS']:
            return 'self.sector'
        
        # Devises
        elif literal_value in ['USD', 'EUR', 'GBP', 'JPY', 'CHF']:
            return 'self.currency'
        
        # Si impossible à déterminer, utiliser le contexte de la fonction
        else:
            # Chercher dans le scope actuel
            current_function = self.get_current_function_name()
            if 'transaction' in current_function.lower():
                return 'self.trx_type'
            elif 'customer' in current_function.lower():
                return 'self.cust_type'
            else:
                self.logger.warning(f"Cannot infer variable for literal '{literal_value}'")
                return f"self.unknown_var  # TODO: Fix - was literal '{literal_value}'"

PATCH_2_TYPE_ANALYZER = """
# Fichier: transpiler/type_analyzer.py
# Ligne: ~45-80 (section analyze_assignment)

class TypeAnalyzer:
    '''Analyzes COBOL types and validates assignments'''
    
    def validate_assignment(self, target, source, pic_clauses):
        '''
        Valide qu'une affectation est compatible en types.
        
        ❌ AVANT: Aucune validation
        ✅ APRÈS: Vérifie les types et émet des warnings
        '''
        target_type = self.get_type_from_pic(pic_clauses.get(target))
        source_type = self.get_type_from_pic(pic_clauses.get(source))
        
        # Si types incompatibles, NE PAS générer l'affectation
        if not self.are_compatible(target_type, source_type):
            self.logger.warning(
                f"Type mismatch: {target} ({target_type}) = {source} ({source_type})"
            )
            # ❌ AVANT: return True  # Génère quand même
            # ✅ APRÈS: return False  # Skip cette affectation
            return False
        
        return True
    
    def get_type_from_pic(self, pic_clause):
        '''Détermine le type Python depuis PIC COBOL'''
        if not pic_clause:
            return 'unknown'
        
        pic = pic_clause.upper()
        
        # Numeric
        if '9' in pic or 'COMP' in pic or 'PACKED' in pic:
            return 'Decimal'
        
        # String
        elif 'X' in pic or 'A' in pic:
            return 'str'
        
        # Date
        elif 'DATE' in pic:
            return 'datetime'
        
        else:
            return 'unknown'
    
    def are_compatible(self, type1, type2):
        '''Vérifie si deux types sont compatibles'''
        if type1 == 'unknown' or type2 == 'unknown':
            return True  # Assume OK si type inconnu
        
        # Types identiques
        if type1 == type2:
            return True
        
        # Conversions autorisées
        compatible_pairs = [
            ('Decimal', 'int'),
            ('int', 'Decimal'),
            ('str', 'bytes'),
        ]
        
        return (type1, type2) in compatible_pairs or (type2, type1) in compatible_pairs
"""

PATCH_3_SEARCH_HANDLER = """
# Fichier: transpiler/cobol_parser.py
# Ligne: ~230-280 (section handle_search)

class SearchHandler:
    '''Handles COBOL SEARCH statement transpilation'''
    
    def generate_search(self, search_node):
        '''
        Génère un SEARCH COBOL en Python.
        
        COBOL:
            SEARCH WS-CCY-CODE
            WHEN WS-CCY(CCY-IDX) = WS-HOLD-CURRENCY(HOLD-IDX)
                MOVE CCY-IDX TO FX-TO-IDX
            END-SEARCH
        
        ❌ AVANT (BUGUÉ):
            for _idx, _item in enumerate(self.ccy_code):
                if _item == _item:  # BUG: Compare item à lui-même!
        
        ✅ APRÈS (CORRIGÉ):
            search_key = self.hold_currency[int(self.hold_idx) - 1]
            for _idx, _item in enumerate(self.ccy_code):
                if _item == search_key:
        '''
        table_name = search_node.table_variable
        when_condition = search_node.when_condition
        
        # ❌ ANCIEN CODE (BUGUÉ)
        # return f'''
        # for _idx, _item in enumerate(self.{table_name}):
        #     if _item == _item:  # BUG!
        # '''
        
        # ✅ NOUVEAU CODE (CORRIGÉ)
        # 1. Extraire la clé de recherche depuis la condition WHEN
        search_key_var = self.extract_search_key(when_condition)
        
        # 2. Générer le code
        code = []
        code.append(f"# SEARCH {table_name}")
        code.append(f"search_key = {search_key_var}")
        code.append(f"_search_found = False")
        code.append(f"for _idx, _item in enumerate(self.{self.pythonize(table_name)}):")
        code.append(f"    if _item == search_key:")
        
        # Actions dans le WHEN
        for action in search_node.when_actions:
            code.append(f"        {self.generate_statement(action)}")
        
        code.append(f"        _search_found = True")
        code.append(f"        break")
        
        # AT END clause si présente
        if search_node.at_end_actions:
            code.append(f"if not _search_found:")
            for action in search_node.at_end_actions:
                code.append(f"    {self.generate_statement(action)}")
        
        return '\\n'.join(code)
    
    def extract_search_key(self, when_condition):
        '''
        Extrait la clé de recherche depuis WHEN condition.
        
        AJOUT CRITIQUE: Cette méthode manquait.
        
        Example:
            WHEN WS-CCY(CCY-IDX) = WS-HOLD-CURRENCY(HOLD-IDX)
            → Extrait: self.hold_currency[int(self.hold_idx) - 1]
        '''
        # Parser la condition
        # Format: TABLE(INDEX) = KEY-VAR(KEY-INDEX)
        parts = when_condition.split('=')
        if len(parts) != 2:
            self.logger.error(f"Invalid SEARCH WHEN condition: {when_condition}")
            return "search_key_todo  # TODO: Fix search key"
        
        # La clé est à droite de =
        key_expression = parts[1].strip()
        
        # Convertir en Python
        return self.map_cobol_expression_to_python(key_expression)
    
    def map_cobol_expression_to_python(self, cobol_expr):
        '''Convertit une expression COBOL en Python avec indices 1-based'''
        import re
        
        # WS-HOLD-CURRENCY(HOLD-IDX) → self.hold_currency[int(self.hold_idx) - 1]
        match = re.match(r'([A-Z0-9-]+)\\(([A-Z0-9-]+)\\)', cobol_expr)
        if match:
            var_name = self.pythonize(match.group(1))
            index_name = self.pythonize(match.group(2))
            return f"self.{var_name}[int(self.{index_name}) - 1]"
        
        # Simple variable: WS-CURRENCY → self.currency
        else:
            return f"self.{self.pythonize(cobol_expr)}"
"""

PATCH_4_POST_VALIDATION = """
# Fichier: transpiler/validator.py (NOUVEAU FICHIER)

'''
Post-transpilation validator
Détecte les bugs communs dans le code généré
'''

import re
from typing import List, Tuple

class TranspilerValidator:
    '''Validates generated Python code for common bugs'''
    
    def __init__(self):
        self.warnings = []
        self.errors = []
    
    def validate(self, python_code: str) -> Tuple[bool, List[str]]:
        '''
        Valide le code généré et retourne (is_valid, issues).
        
        ✅ AJOUT CRITIQUE: Détection automatique des bugs après génération
        '''
        self.warnings = []
        self.errors = []
        
        lines = python_code.split('\\n')
        
        # Check 1: Détecter "if True:" suspect
        for i, line in enumerate(lines, 1):
            if re.match(r'^\\s*(if|elif)\\s+True\\s*:', line):
                # Vérifier si c'est un pattern valide ou un bug
                if not self._is_valid_true_condition(lines, i-1):
                    self.errors.append(
                        f"Line {i}: Suspicious 'if True:' - likely missing condition"
                    )
        
        # Check 2: Détecter "_item == _item"
        for i, line in enumerate(lines, 1):
            if '_item == _item' in line:
                self.errors.append(
                    f"Line {i}: SEARCH bug - comparing item to itself"
                )
        
        # Check 3: Détecter conditions avec strings littérales
        for i, line in enumerate(lines, 1):
            match = re.match(r'^\\s*(elif|if)\\s+\\'([^\\']+)\\'\\s*:', line)
            if match and 'print' not in line:
                self.errors.append(
                    f"Line {i}: Literal string condition '{match.group(2)}' - always True"
                )
        
        # Check 4: Détecter variables COBOL non converties
        for i, line in enumerate(lines, 1):
            for var in ['self.zero', 'self.zeros', 'self.spaces', 'self.high_values']:
                if var in line.lower():
                    self.errors.append(
                        f"Line {i}: COBOL keyword not converted: {var}"
                    )
        
        # Check 5: Détecter affectations de types incompatibles
        for i, line in enumerate(lines, 1):
            # portfolio_size (Decimal) = pos_symbol_key (str)
            if re.search(r'self\\.\\w+_size\\s*=\\s*self\\.\\w+_(key|id|symbol)', line):
                self.warnings.append(
                    f"Line {i}: Possible type mismatch - size variable assigned non-numeric"
                )
        
        is_valid = len(self.errors) == 0
        all_issues = self.errors + self.warnings
        
        return is_valid, all_issues
    
    def _is_valid_true_condition(self, lines, line_idx):
        '''
        Vérifie si "if True:" est intentionnel ou un bug.
        
        Valide si:
        - Suivi d'un commentaire explicatif
        - Dans un contexte de test
        '''
        line = lines[line_idx]
        
        # Si commentaire explicite, OK
        if '# ' in line and any(kw in line.lower() for kw in ['todo', 'intentional', 'debug']):
            return True
        
        # Si dans une fonction de test, OK
        for i in range(max(0, line_idx - 10), line_idx):
            if 'def test_' in lines[i] or 'def mock_' in lines[i]:
                return True
        
        return False
    
    def generate_report(self) -> str:
        '''Génère un rapport de validation'''
        report = []
        report.append("=" * 70)
        report.append("TRANSPILATION VALIDATION REPORT")
        report.append("=" * 70)
        report.append("")
        
        if not self.errors and not self.warnings:
            report.append("✅ No issues detected")
        else:
            if self.errors:
                report.append(f"❌ {len(self.errors)} ERROR(S):")
                for error in self.errors:
                    report.append(f"   {error}")
                report.append("")
            
            if self.warnings:
                report.append(f"⚠️  {len(self.warnings)} WARNING(S):")
                for warning in self.warnings:
                    report.append(f"   {warning}")
        
        report.append("")
        report.append("=" * 70)
        
        return '\\n'.join(report)
"""
