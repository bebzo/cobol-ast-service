function splitCobolModules(code: string): { name: string; code: string }[] {
  const modules: { name: string; code: string }[] = [];
  const lines = code.split('\n');
  const divisions = ['IDENTIFICATION DIVISION', 'ENVIRONMENT DIVISION', 'DATA DIVISION', 'PROCEDURE DIVISION'];
  let currentModule = { name: 'HEADER', code: '' };
  
  for (const line of lines) {
    const upperLine = line.toUpperCase();
    let foundDivision = false;
    for (const div of divisions) {
      if (upperLine.includes(div)) {
        if (currentModule.code.trim()) modules.push({ ...currentModule });
        currentModule = { name: div, code: line + '\n' };
        foundDivision = true;
        break;
      }
    }
    if (!foundDivision) {
      const paragraphMatch = line.match(/^\s{7}(\d{4}-[\w-]+)\./);
      if (paragraphMatch && currentModule.name === 'PROCEDURE DIVISION' && currentModule.code.length > 500) {
        modules.push({ ...currentModule });
        currentModule = { name: `PROC:${paragraphMatch[1]}`, code: line + '\n' };
      } else {
        currentModule.code += line + '\n';
      }
    }
  }
  if (currentModule.code.trim()) modules.push(currentModule);
  return modules;
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const body = await req.json();
    const { action, cobolCode, query, pythonCode } = body;

    const apiKey = Deno.env.get('GEMINI_API_KEY');
    if (!apiKey) {
      return new Response(JSON.stringify({ error: "No API key" }), 
        { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
    }

    const genAI = new GoogleGenerativeAI(apiKey);

    if (action === 'voice') {
      const model = genAI.getGenerativeModel({ model: "gemini-2.0-flash-exp" });
      const prompt = `Voice assistant for COBOL migration. Answer in ENGLISH in 2-3 sentences.
COBOL: ${cobolCode?.substring(0, 500) || 'None'}
Python: ${pythonCode?.substring(0, 500) || 'None'}
Question: ${query}`;
      const result = await model.generateContent(prompt);
      return new Response(JSON.stringify({ response: result.response.text() }), 
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
    }

    if (!cobolCode) {
      return new Response(JSON.stringify({ error: "No COBOL code" }), 
        { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
    }

    const model = genAI.getGenerativeModel({ 
      model: "gemini-2.0-flash-exp",
      generationConfig: { responseMimeType: "application/json", maxOutputTokens: 8192 }
    });

    const lineCount = cobolCode.split('\n').length;
    
    if (lineCount > 200) {
      const modules = splitCobolModules(cobolCode);
      const moduleResults: any[] = [];
      
      for (const mod of modules) {
        if (mod.code.trim().length < 50) continue;
        for (let attempt = 1; attempt <= 2; attempt++) {
          try {
            const result = await model.generateContent(CHUNK_PROMPT + mod.code);
            let text = result.response.text().trim();
            if (text.startsWith("```")) text = text.replace(/```json?\n?/g, '').replace(/```/g, '').trim();
            const parsed = JSON.parse(text);
            parsed.module_name = mod.name;
            moduleResults.push(parsed);
            break;
          } catch { await new Promise(r => setTimeout(r, 500)); }
        }
      }
      
      for (let attempt = 1; attempt <= 3; attempt++) {
        try {
          const mergeResult = await model.generateContent(MERGE_PROMPT + JSON.stringify(moduleResults, null, 2));
          let text = mergeResult.response.text().trim();
          if (text.startsWith("```")) text = text.replace(/```json?\n?/g, '').replace(/```/g, '').trim();
          const parsed = JSON.parse(text);
          parsed.modules_analyzed = modules.length;
          return new Response(JSON.stringify(parsed), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
        } catch { await new Promise(r => setTimeout(r, 1000)); }
      }
    }
    
    for (let attempt = 1; attempt <= 3; attempt++) {
      try {
        const result = await model.generateContent(STANDARD_PROMPT + cobolCode);
        let text = result.response.text().trim();
        if (text.startsWith("```")) text = text.replace(/```json?\n?/g, '').replace(/```/g, '').trim();
        return new Response(JSON.stringify(JSON.parse(text)), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
      } catch { await new Promise(r => setTimeout(r, 1000)); }
    }
    
    throw new Error("Failed after retries");
  } catch (error) {
    return new Response(JSON.stringify({ error: error instanceof Error ? error.message : "Unknown error" }), 
      { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } });
  }
});
