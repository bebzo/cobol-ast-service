        )

import csv
import warnings
from dataclasses import dataclass
from decimal import Decimal, InvalidOperation
from typing import List, Optional, Dict

@dataclass(frozen=True)
class TrancheImposition:
    """Représente une tranche d'imposition avec ses limites et son taux."""
    limite_inferieure: Decimal
    limite_superieure: Decimal
    taux: Decimal

    def __post_init__(self):
        if self.limite_inferieure > self.limite_superieure:
            raise ValueError("La limite inférieure doit être inférieure ou égale à la limite supérieure.")
        if not (0 <= self.taux <= 1):
            raise ValueError("Le taux doit être compris entre 0 et 1.")


class ConfigFiscale:
    """Gère la configuration fiscale, y compris le chargement depuis un fichier JSON."""
    def __init__(self, annee: int, tranches: List[TrancheImposition]):
        self.annee = annee
        self.tranches = tranches

    @classmethod
    def charger(cls, fichier_json: str) -> 'ConfigFiscale':
        """Charge la configuration fiscale depuis un fichier JSON.

        Args:
            fichier_json: Le chemin vers le fichier JSON.

        Returns:
            Une instance de ConfigFiscale.
        """
        try:
            with open(fichier_json, 'r') as f:
                data = json.load(f)

            annee = int(data['annee'])
            tranches_data = data['tranches']
            tranches = []
            for tranche_data in tranches_data:
                tranches.append(TrancheImposition(
                    limite_inferieure=Decimal(str(tranche_data['limite_inferieure'])),
                    limite_superieure=Decimal(str(tranche_data['limite_superieure'])),
                    taux=Decimal(str(tranche_data['taux']))
                ))

            return cls(annee=annee, tranches=tranches)
        except (FileNotFoundError, json.JSONDecodeError, KeyError, ValueError, InvalidOperation) as e:
            raise ValueError(f"Erreur lors du chargement de la configuration fiscale: {e}")

class GestionnaireFiscal:
    """Gère les configurations fiscales pour différentes années avec un cache."""
    _cache: Dict[int, ConfigFiscale] = {}

    @classmethod
    def obtenir_config(cls, annee: int, fichier_json: str) -> ConfigFiscale:
        """Obtient la configuration fiscale pour une année donnée, en utilisant un cache.

        Args:
            annee: L'année pour laquelle obtenir la configuration.
            fichier_json: Le chemin vers le fichier JSON de configuration.

        Returns:
            Une instance de ConfigFiscale.
        """
        if annee not in cls._cache:
            config = ConfigFiscale.charger(fichier_json)
            if config.annee != annee:
                 warnings.warn(f"La configuration du fichier {fichier_json} est pour l'année {config.annee} et non pour l'année demandée {annee}")
            cls._cache[annee] = config

        return cls._cache[annee]


class AuditFiscal:
    """Enregistre les calculs d'impôt dans un fichier CSV pour l'audit."""
    def __init__(self, fichier_csv: str):
        self.fichier_csv = fichier_csv

    def enregistrer_audit(self, revenu: Decimal, impot: Decimal, annee: int) -> None:
        """Enregistre un calcul d'impôt dans le fichier CSV.

        Args:
            revenu: Le revenu annuel.
            impot: L'impôt calculé.
            annee: L'année fiscale.

        except IOError as e:

def calculer_impot(revenu: Decimal, config: ConfigFiscale) -> Decimal:
    """Calcule l'impôt sur le revenu en utilisant les tranches d'imposition.

    Args:
        revenu: Le revenu annuel.
        config: La configuration fiscale (instance de ConfigFiscale).

    Returns:
        Le montant de l'impôt calculé.
    """
    impot = Decimal('0')
    revenu_restant = revenu

    for tranche in config.tranches:
        if revenu_restant > tranche.limite_inferieure:
            montant_imposable = min(revenu_restant, tranche.limite_superieure) - tranche.limite_inferieure
            impot += montant_imposable * tranche.taux

    return impot.quantize(Decimal('0.00'))

def calculer_impot(revenu: Decimal, config: ConfigFiscale) -> Decimal:
    """Calcule l'impôt sur le revenu en utilisant les tranches d'imposition.
    
    Note: Le revenu est imposé par tranches successives.
    """
    if revenu <= Decimal('0'):
        return Decimal('0.00')
    
    impot_total = Decimal('0')
    revenu_imposable = revenu
    
    for i, tranche in enumerate(config.tranches):
        # Pour la dernière tranche (pas de limite supérieure ou très grande)
        if i == len(config.tranches) - 1 or revenu_imposable <= tranche.limite_superieure:
            montant_tranche = revenu_imposable - tranche.limite_inferieure
            if montant_tranche > 0:
                impot_total += montant_tranche * tranche.taux
            break
        
        # Tranches intermédiaires
        montant_tranche = tranche.limite_superieure - tranche.limite_inferieure
        impot_total += montant_tranche * tranche.taux
        revenu_imposable = tranche.limite_superieure
    
    return impot_total.quantize(Decimal('0.00'))

class RevenuAnnuel:
    montant: Decimal
    annee_reference: int
    
    def __post_init__(self):
        if self.montant < Decimal('0'):
            raise ValueError("Le revenu ne peut pas être négatif")
        if self.montant > Decimal('1000000000'):  # 1 milliard
            raise ValueError("Le revenu est trop élevé")
        if self.annee_reference < 1900 or self.annee_reference > 2100:
            raise ValueError("Année de référence invalide")
    
    def ajuster_inflation(self, nouvelle_annee: int, taux_inflation: Decimal) -> 'RevenuAnnuel':
        """Ajuste le revenu pour l'inflation."""
        annees = nouvelle_annee - self.annee_reference
        facteur = (Decimal('1') + taux_inflation) ** annees
        return RevenuAnnuel(self.montant * facteur, nouvelle_annee)

        """Valide qu'un chemin de fichier est sûr."""
        if extensions_autorisees is None:
            extensions_autorisees = ['.json', '.csv']
        
        path = Path(chemin)
        
        # Vérifier l'extension
        if path.suffix not in extensions_autorisees:
            raise ValueError(f"Extension de fichier non autorisée: {path.suffix}")
        
        # Vérifier que c'est un fichier (pas un répertoire)
        if path.exists() and path.is_dir():
            raise ValueError(f"Le chemin spécifié est un répertoire: {chemin}")
        
        # Vérifier les chemins relatifs (éviter les chemins absolus ou traversées)
        if os.path.isabs(chemin) or '..' in chemin:
            raise ValueError(f"Chemin de fichier potentiellement dangereux: {chemin}")
        
        return True

# analyzer/cobol_analyzer.py
import re
from dataclasses import dataclass
from datetime import datetime
from typing import List, Dict, Any

@dataclass
class CodeIssue:
    type: str
    severity: str  # LOW, MEDIUM, HIGH, CRITICAL
    line: int
    description: str
    suggestion: str
    automatic_fix: bool = False

class COBOLAnalyzer:
    def __init__(self, cobol_source: str):
        self.source = cobol_source
        self.lines = cobol_source.split('\n')
        self.issues: List[CodeIssue] = []
        self.metadata: Dict[str, Any] = {}
        
    def analyze(self):
        """Analyse complète du code COBOL"""
        self._extract_metadata()
        self._detect_obsolete_tax_rates()
        self._detect_missing_validations()
        self._detect_error_handling()
        self._detect_performance_issues()
        self._detect_security_issues()
        return self._generate_report()
    
    def _extract_metadata(self):
        """Extrait les métadonnées du programme COBOL"""
        # Détection de l'année dans les commentaires
        year_patterns = [
            r'19\d{2}', r'20\d{2}', 
            r'TAUX.*?(19\d{2}|20\d{2})',
            r'ANNEE.*?(19\d{2}|20\d{2})'
        ]
        
        for pattern in year_patterns:
            matches = re.findall(pattern, self.source, re.IGNORECASE)
            if matches:
                self.metadata['detected_years'] = list(set(matches))
        
        # Détection du type de programme
        if any(keyword in self.source.upper() for keyword in ['TAUX', 'IMPOSITION', 'IMPOT']):
            self.metadata['program_type'] = 'TAX_CALCULATION'
        
        # Détection des variables critiques
        var_pattern = r'01\s+(\w+-\w+)\s+PIC'
        self.metadata['variables'] = re.findall(var_pattern, self.source)
    
    def _detect_obsolete_tax_rates(self):
        """Détecte les taux d'imposition obsolètes"""
        current_year = datetime.now().year
        
        for i, line in enumerate(self.lines):
            line_upper = line.upper()
            
            # Recherche de références à des années
            year_match = re.search(r'(19\d{2}|20\d{1,2})', line)
            if year_match:
                year = int(year_match.group(1))
                if year < current_year - 5:  # > 5 ans = obsolète
                    self.issues.append(CodeIssue(
                        type="OBSOLETE_DATA",
                        severity="HIGH",
                        line=i+1,
                        description=f"Données fiscales de {year} détectées",
                        suggestion=f"Mettre à jour avec les taux de {current_year}",
                        automatic_fix=True
                    ))
            
            # Recherche de valeurs de taux fixes
            if any(keyword in line_upper for keyword in ['TAUX', 'RATE', 'VALUE .', 'VALUE 0.']):
                rate_match = re.search(r'VALUE\s+[\.\d]+', line)
                if rate_match:
                    self.issues.append(CodeIssue(
                        type="HARDCODED_VALUE",
                        severity="MEDIUM",
                        line=i+1,
                        description="Valeur de taux codée en dur",
                        suggestion="Externaliser dans un fichier de configuration",
                        automatic_fix=True
                    ))
    
    def _detect_missing_validations(self):
        """Détecte l'absence de validation des données"""
        # Recherche de zones de saisie sans validation
        input_patterns = [
            r'ACCEPT\s+(\w+)',
            r'MOVE\s+.*\s+TO\s+(\w+-?\w+)'
        ]
        
        for i, line in enumerate(self.lines):
            if 'IF' not in line.upper() and any(pattern in line.upper() for pattern in ['ACCEPT', 'MOVE']):
                self.issues.append(CodeIssue(
                    type="MISSING_VALIDATION",
                    severity="MEDIUM",
                    line=i+1,
                    description="Saisie de données sans validation",
                    suggestion="Ajouter des vérifications de plage et de format",
                    automatic_fix=False
                ))
    
    def _detect_error_handling(self):
        """Détecte l'absence de gestion d'erreurs"""
        error_keywords = ['ON SIZE ERROR', 'ON EXCEPTION', 'AT END']
        
        has_error_handling = any(keyword in self.source.upper() for keyword in error_keywords)
        
        if not has_error_handling:
            self.issues.append(CodeIssue(
                type="NO_ERROR_HANDLING",
                severity="MEDIUM",
                line=1,
                description="Absence de gestion d'erreurs",
                suggestion="Implémenter try/except avec logging",
                automatic_fix=True
            ))
    
    def _detect_performance_issues(self):
        """Détecte les problèmes de performance"""
        # Recherche de boucles potentiellement inefficaces
        for i, line in enumerate(self.lines):
            if 'PERFORM VARYING' in line.upper() and 'UNTIL' in line.upper():
                nested_perform = False
                # Vérifier si c'est imbriqué
                for j in range(i+1, min(i+10, len(self.lines))):
                    if 'PERFORM' in self.lines[j].upper():
                        nested_perform = True
                        break
                
                if nested_perform:
                    self.issues.append(CodeIssue(
                        type="NESTED_LOOP",
                        severity="LOW",
                        line=i+1,
                        description="Boucles potentiellement imbriquées",
                        suggestion="Optimiser les algorithmes",
                        automatic_fix=False
                    ))
    
    def _detect_security_issues(self):
        """Détecte les problèmes de sécurité"""
        security_patterns = [
            (r'CALL\s+[\'\"]', "EXTERNAL_CALL", "Appel système non sécurisé"),
            (r'DISPLAY\s+[\w-]+\s+UPON\s+CONSOLE', "SENSITIVE_OUTPUT", "Affichage de données sensibles"),
        ]
        
        for i, line in enumerate(self.lines):
            for pattern, issue_type, description in security_patterns:
                if re.search(pattern, line, re.IGNORECASE):
                    self.issues.append(CodeIssue(
                        type=issue_type,
                        severity="HIGH",
                        line=i+1,
                        description=description,
                        suggestion="Restreindre l'accès et logger les accès",
                        automatic_fix=False
                    ))
    
    def _generate_report(self) -> Dict[str, Any]:
        """Génère un rapport d'analyse détaillé"""
        report = {
            "metadata": self.metadata,
            "summary": {
                "total_issues": len(self.issues),
                "by_severity": {
                    "CRITICAL": sum(1 for i in self.issues if i.severity == "CRITICAL"),
                    "HIGH": sum(1 for i in self.issues if i.severity == "HIGH"),
                    "MEDIUM": sum(1 for i in self.issues if i.severity == "MEDIUM"),
                    "LOW": sum(1 for i in self.issues if i.severity == "LOW"),
                },
                "automatic_fix_count": sum(1 for i in self.issues if i.automatic_fix)
            },
            "issues": [issue.__dict__ for issue in self.issues],
            "recommendations": self._generate_recommendations()
        }
        
        return report
    
    def _generate_recommendations(self) -> List[str]:
        """Génère des recommandations basées sur l'analyse"""
        recommendations = []
        
        if any(i.type == "OBSOLETE_DATA" for i in self.issues):
            recommendations.append("Mettre à jour les données fiscales avec les valeurs actuelles")
        
        if any(i.type == "MISSING_VALIDATION" for i in self.issues):
            recommendations.append("Implémenter un système de validation des entrées")
        
        if any(i.type == "NO_ERROR_HANDLING" for i in self.issues):
            recommendations.append("Ajouter une gestion d'erreurs complète")
        
        return recommendations

from dataclasses import dataclass
from typing import List, Dict, Set
import re

@dataclass
class TestCase:
    name: str
    description: str
    input_values: Dict[str, str]
    expected_output: str
    test_type: str  # "unit", "integration", "edge_case"

class COBOLTestGenerator:
    def __init__(self, cobol_source: str):
        self.source = cobol_source
        self.test_cases: List[TestCase] = []
        self.variables = self._extract_variables()
        self.literals = self._extract_literals()
    
    def _extract_variables(self) -> Dict[str, str]:
        """Extrait les variables et leurs types"""
        variables = {}
        var_pattern = r'01\s+(\w+-\w+)\s+PIC\s+([\w\(\)]+)'
        
        for match in re.finditer(var_pattern, self.source, re.IGNORECASE):
            var_name = match.group(1)
            var_type = match.group(2)
            variables[var_name] = var_type
        
        return variables
    
    def _extract_literals(self) -> List[str]:
        """Extrait les valeurs littérales du code"""
        literals = []
        
        # Recherche des MOVE avec valeurs fixes
        move_pattern = r'MOVE\s+(\d+)\s+TO\s+\w+'
        for match in re.finditer(move_pattern, self.source, re.IGNORECASE):
            literals.append(match.group(1))
        
        # Recherche des VALUE initiales
        value_pattern = r'VALUE\s+(\d+)'
        for match in re.finditer(value_pattern, self.source, re.IGNORECASE):
            literals.append(match.group(1))
        
        return list(set(literals))
    
    def _extract_test_values(self) -> List[Dict[str, str]]:
        """Extrait les valeurs de test du code COBOL"""
        test_values = []
        
        # Recherche des MOVE de test
        for line in self.source.split('\n'):
            if 'MOVE' in line.upper() and 'TO' in line.upper():
                parts = line.split()
                for i, part in enumerate(parts):
                    if part.upper() == 'MOVE' and i + 2 < len(parts):
                        value = parts[i + 1]
                        variable = parts[i + 3]
                        
                        # Vérifie si c'est une valeur numérique
                        if value.isdigit():
                            test_values.append({
                                "variable": variable,
                                "value": value,
                                "line": line.strip()
                            })
        
        return test_values
    
    def generate_unit_tests(self) -> List[TestCase]:
        """Génère les tests unitaires basés sur le code"""
        test_values = self._extract_test_values()
        
        for test_val in test_values:
            var_name = test_val["variable"]
            value = test_val["value"]
            
            # Génère un nom de test significatif
            test_name = f"test_{var_name.lower().replace('-', '_')}_{value}"
            
            test_case = TestCase(
                name=test_name,
                description=f"Test de {var_name} avec valeur {value} (extrait du code COBOL)",
                input_values={var_name: value},
                expected_output=f"Vérifier le calcul avec {var_name}={value}",
                test_type="unit"
            )
            
            self.test_cases.append(test_case)

@app.post("/api/refactor", response_model=RefactorResponse)

def parse_gemini_response(text: str) -> dict:
    """Parse la réponse de Gemini en composants structurés"""
    # Logique de parsing (simplifiée)
    sections = {
        "python_code": "",
        "test_code": "",
        "config_code": "",
        "analysis_report": {},
        "confidence_score": 0.95
    }
    
    # Exemple de parsing simple
    lines = text.split('\n')
    current_section = None
    
    for line in lines:
        if "```python" in line:
            current_section = "python_code"
        elif "```json" in line:
            current_section = "config_code"
        elif "```" in line:
            current_section = None
        elif current_section:
            sections[current_section] += line + "\n"
    
    return sections

        if self.config.api_key:
            self.session.headers.update({"X-API-Key": self.config.api_key})
    
    def refactor_cobol(self, cobol_code: str, **kwargs) -> Dict[str, Any]:
        """
        Envoie du code COBOL à refactoriser
        """
        payload = {
            "cobol_code": cobol_code,
            **kwargs
        }
        
        try:
            response = self.session.post(
                f"{self.config.api_url}/api/refactor",
                json=payload,
                timeout=30
            )
            response.raise_for_status()
            return response.json()
        except requests.RequestException as e:
            print(f"Erreur API: {e}")
            # Fallback local si l'API échoue
            return self._local_fallback(cobol_code, **kwargs)
    
    def analyze_cobol(self, cobol_code: str) -> Dict[str, Any]:
        """
        Analyse du code COBOL
        """
        payload = {"cobol_code": cobol_code}
        
        try:
            response = self.session.post(
                f"{self.config.api_url}/api/analyze",
                json=payload,
                timeout=15
            )
            response.raise_for_status()
            return response.json()
        except requests.RequestException as e:
            print(f"Erreur API: {e}")
            return self._local_analysis(cobol_code)
    
    def _local_fallback(self, cobol_code: str, **kwargs) -> Dict[str, Any]:
        """Fallback local si l'API n'est pas disponible"""
        # Utilise vos algorithmes locaux existants
        from .local_refactor import refactor_locally
        
        return refactor_locally(cobol_code, **kwargs)
    
    def _local_analysis(self, cobol_code: str) -> Dict[str, Any]:
        """Analyse locale de fallback"""
        # Utilise votre analyseur local
        from .local_analyzer import analyze_locally
        
        return analyze_locally(cobol_code)

const vscode = require('vscode');
const axios = require('axios');

// VOTRE URL API - configurable
const API_URL = 'https://api.codeswitch.dev';

async function refactorCobol() {
    const editor = vscode.window.activeTextEditor;
    
    if (!editor) {
        vscode.window.showErrorMessage('Aucun éditeur actif');
        return;
    }
    
    const cobolCode = editor.document.getText();
    
    if (!cobolCode.trim()) {
        vscode.window.showWarningMessage('Aucun code COBOL à refactoriser');
        return;
    }
    
    // Affichage de la progression
    await vscode.window.withProgress({
        location: vscode.ProgressLocation.Notification,
        title: "Refactorisation COBOL en cours...",
        cancellable: false
    }, async (progress) => {
        try {
            progress.report({ increment: 20, message: "Analyse du code..." });
            
            // Appel à VOTRE API
            const response = await axios.post(`${API_URL}/api/refactor`, {
                cobol_code: cobolCode,
                language: 'python'
            });
            
            progress.report({ increment: 60, message: "Génération du code Python..." });
            
            // Création d'un nouveau fichier
            const pythonCode = response.data.python_code;
            const document = await vscode.workspace.openTextDocument({
                content: pythonCode,
                language: 'python'
            });
            
            await vscode.window.showTextDocument(document, vscode.ViewColumn.Beside);
            
            progress.report({ increment: 20, message: "Terminé !" });
            
            vscode.window.showInformationMessage(
                `Refactorisation réussie (${(response.data.confidence_score * 100).toFixed(1)}% confiance)`
            );
            
        } catch (error) {
            vscode.window.showErrorMessage(
                `Erreur de refactorisation: ${error.message}`
            );
        }
    });
}

function activate(context) {
    let disposable = vscode.commands.registerCommand('codeswitch.refactor', refactorCobol);
    context.subscriptions.push(disposable);
}

function deactivate() {}

module.exports = {
    activate,
    deactivate
};

from fastapi import Request, HTTPException
from datetime import datetime, timedelta
import redis
import hashlib

class RateLimiter:
    def __init__(self):
        self.redis = redis.Redis.from_url(os.getenv("REDIS_URL"))
        
    async def __call__(self, request: Request):
        client_ip = request.client.host
        endpoint = request.url.path
        
        key = f"ratelimit:{client_ip}:{endpoint}"
        current = self.redis.get(key)
        
        if current and int(current) > 100:  # 100 requêtes max
            raise HTTPException(status_code=429, detail="Trop de requêtes")
        
        self.redis.incr(key)
        self.redis.expire(key, 3600)  # 1 heure
        
        return True

import { serve } from "https://deno.land/std@0.168.0/http/server.ts"

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

serve(async (req) => {
  // Gestion CORS
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    const { cobol_code, language = 'python', year = 2025 } = await req.json()
    
    // VOTRE CLÉ API (dans les secrets Supabase)
    const GEMINI_API_KEY = Deno.env.get('GEMINI_API_KEY')
    
    if (!GEMINI_API_KEY) {
      throw new Error('API key not configured')
    }

    // Appel à Gemini API
    const response = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${GEMINI_API_KEY}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          contents: [{
            parts: [{
              text: `Tu es un expert en migration COBOL vers ${language}.
              
Refactorise ce code COBOL avec:
1. Architecture moderne (classes, dataclasses)
2. Gestion d'erreurs
3. Configuration externalisée
4. Tests unitaires
5. Documentation en français

Code COBOL:
${cobol_code}

Année cible: ${year}

Retourne UNIQUEMENT le code ${language} sans commentaires supplémentaires.`
            }]
          }],
          generationConfig: {
            temperature: 0.2,
            topK: 1,
            topP: 1,
            maxOutputTokens: 2048,
          }
        })
      }
    )

    const data = await response.json()
    
    if (!response.ok) {
      throw new Error(data.error?.message || 'Gemini API error')
    }

    const pythonCode = data.candidates[0]?.content?.parts[0]?.text || ''

    return new Response(
      JSON.stringify({
        success: true,
        python_code: pythonCode,
        model: 'gemini-1.5-flash',
        tokens_used: data.usageMetadata?.totalTokenCount || 0
      }),
      {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 200,
      }
    )

  } catch (error) {
    return new Response(
      JSON.stringify({
        success: false,
        error: error.message
      }),
      {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status: 400,
      }
    )
  }

import os
import requests
from typing import Optional, Dict, Any
import supabase
from datetime import datetime

class CodeSwitchSupabase:
    """Client pour Supabase Edge Functions"""
    
    def __init__(self, supabase_url: Optional[str] = None, supabase_key: Optional[str] = None):
        # Récupère depuis les variables d'environnement ou utilise des valeurs par défaut
        self.supabase_url = supabase_url or os.getenv("SUPABASE_URL")
        self.supabase_key = supabase_key or os.getenv("SUPABASE_ANON_KEY")
        
        if not self.supabase_url or not self.supabase_key:
            raise ValueError("Supabase URL et Key requis")
        
        # Client Supabase
        self.client = supabase.create_client(self.supabase_url, self.supabase_key)
        
        # URL de l'edge function
        self.function_url = f"{self.supabase_url}/functions/v1/gemini-refactor"
    
    def refactor(self, cobol_code: str, language: str = "python", year: int = 2025) -> Dict[str, Any]:
        """
        Refactorise du code COBOL via Supabase
        """
        headers = {
            "Authorization": f"Bearer {self.supabase_key}",
            "Content-Type": "application/json"
        }
        
        payload = {
            "cobol_code": cobol_code,
            "language": language,
            "year": year
        }
        
        try:
            # 1. Log la requête (optionnel)
            self._log_request(len(cobol_code), language)
            
            # 2. Appel à l'edge function
            response = requests.post(
                self.function_url,
                headers=headers,
                json=payload,
                timeout=30
            )
            
            result = response.json()
            
            # 3. Log le résultat
            self._log_result(
                success=result.get("success", False),
                tokens_used=result.get("tokens_used", 0),
                error_message=result.get("error")
            )
            
            return result
            
        except Exception as e:
            # Log l'erreur
            self._log_result(success=False, error_message=str(e))
            raise
    
    def _log_request(self, code_length: int, language: str):
        """Log la requête dans la table audit"""
        try:
            # Récupère l'user ID si authentifié
            user = self.client.auth.get_user()
            user_id = user.user.id if user.user else None
            
            # Insère dans la table (via REST API car Edge Functions peut appeler)
            self.client.table("refactor_audit").insert({
                "user_id": user_id,
                "cobol_length": code_length,
                "language": language,
                "created_at": datetime.utcnow().isoformat()
            }).execute()
            
        except Exception as e:
            print(f"Logging error (non bloquant): {e}")
    
    def _log_result(self, success: bool, tokens_used: int = 0, error_message: str = None):
        """Met à jour le log avec le résultat"""
        try:
            # Met à jour le dernier log (simplifié)
            # En production, vous utiliseriez une table séparée ou une meilleure approche
            pass
            
        except Exception as e:
            print(f"Result logging error: {e}")
    
    def get_usage_stats(self, user_id: Optional[str] = None):
        """Récupère les statistiques d'utilisation"""
        query = self.client.table("refactor_audit").select("*")
        
        if user_id:
            query = query.eq("user_id", user_id)

  DATE(created_at) as date,
  COUNT(*) as total_requests,
  SUM(CASE WHEN success THEN 1 ELSE 0 END) as successful_requests,
  SUM(tokens_used) as total_tokens,

// Vérifie le rate limiting
const checkRateLimit = async (userId: string, ip: string) => {
  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000)
  
  const { count } = await supabase
    .from('refactor_audit')
    .select('*', { count: 'exact', head: true })
    .or(`user_id.eq.${userId},ip_address.eq.${ip}`)
    .gte('created_at', oneHourAgo.toISOString())
  
  return (count || 0) < 100 // 100 requêtes/heure max
}
