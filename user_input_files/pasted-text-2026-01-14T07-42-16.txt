pythondef estimate_effort(metrics):
    base_days = metrics.cobol_lines / 500  # 500 lignes/jour baseline
    
    # Facteurs multiplicateurs
    complexity_factor = {
        "LOW": 0.8,
        "MEDIUM": 1.0,
        "HIGH": 1.5
    }[metrics.complexity]
    
    external_calls_penalty = len(metrics.external_calls) * 0.5
    test_coverage_bonus = -0.3 if metrics.test_pass_rate == 100 else 0
    
    total_days = (base_days * complexity_factor + 
                  external_calls_penalty + 
                  test_coverage_bonus)
    
    return max(1, round(total_days, 1))  # Minimum 1 jour
