        """19200-execute_steps."""
        while self.ws_current_step <= self.ws_total_steps and self.ws_workflow_status != "FAILED":
            self.execute_current_step()
            self.ws_current_step += 1

        """19210-execute_current_step."""
        self.step_start_date[self.ws_current_step - 1] = str(date.today())
        self.step_status[self.ws_current_step - 1] = "in_progress"
        match self.step_name[self.ws_current_step - 1]:
            case "VALIDATION":
                self.validation_step()
            case "APPROVAL":
                self.approval_step()
            case "PROCESSING":
                self.processing_step()
            case "NOTIFICATION":
                self.notification_step()
            case _:
                self.generic_step()
        self.step_end_date[self.ws_current_step - 1] = str(date.today())

        """19220-validation_step."""

            self.step_status[self.ws_current_step - 1] = "COMPLETED"
            self.step_outcome[self.ws_current_step - 1] = "VALIDATED"
        else:
            self.step_status[self.ws_current_step - 1] = "FAILED"
            self.step_outcome[self.ws_current_step - 1] = "VALIDATION FAILED"
            self.ws_workflow_status = "FAILED"

        """19230-approval_step."""

            self.step_status[self.ws_current_step - 1] = "COMPLETED"
            self.step_outcome[self.ws_current_step - 1] = "APPROVED"
        elif self.ws_rejection_received == "Y":
            self.step_status[self.ws_current_step - 1] = "COMPLETED"
            self.step_outcome[self.ws_current_step - 1] = "REJECTED"
            self.ws_workflow_status = "FAILED"
        else:
            self.step_status[self.ws_current_step - 1] = "PENDING"
            self.ws_current_step -= 1
