class CobolConditionTranspiler:
    """Convertit les conditions COBOL en Python valide"""
    
    def parse_evaluate_when(self, cobol_lines):
        """
        COBOL:
            EVALUATE TRUE
            WHEN TRD-QUANTITY = ZERO
                ...
            WHEN TRD-PRICE <= ZERO
                ...
        
        Python:
            if self.trd_quantity == Decimal('0'):
                ...
            elif self.trd_price <= Decimal('0'):
                ...
        """
        conditions = []
        for line in cobol_lines:
            if 'WHEN' in line:
                # Parser la condition après WHEN
                condition = self.parse_cobol_condition(line)
                conditions.append(condition)
        return conditions
    
    def parse_cobol_condition(self, line):
        """Convertit une condition COBOL en Python"""
        # ZERO -> Decimal('0')
        line = line.replace('ZERO', "Decimal('0')")
        # = -> ==
        line = line.replace(' = ', ' == ')
        # SPACES -> ''
        line = line.replace('SPACES', "''")
        return line

class CobolArrayTranspiler:
    """Gère les accès aux tableaux COBOL (1-indexed)"""
    
    def parse_array_access(self, cobol_expr):
        """
        COBOL: WS-HOLD-SYMBOL(HOLD-IDX)
        Python: self.hold_symbol[int(self.hold_idx) - 1]
        """
        if '(' in cobol_expr:
            array_name, index = cobol_expr.split('(')
            index = index.rstrip(')')
            return f"self.{array_name.lower()}[int(self.{index.lower()}) - 1]"
        return cobol_expr
