python# debug_suite.py
import re
from api.transpile import generate_python_code

def find_all_issues(python_code):
    """Trouve tous les problèmes potentiels"""
    issues = []
    lines = python_code.split('\n')
    
    # 1. Decimal avec subscript
    for i, line in enumerate(lines, 1):
        if re.search(r'Decimal\[[^\]]+\]', line):
            issues.append({
                'type': 'decimal_subscript',
                'line': i,
                'content': line.strip()
            })
    
    # 2. Strings non fermées
    for i, line in enumerate(lines, 1):
        stripped = line.strip()
        if stripped.startswith('"') and not stripped.startswith('"""'):
            if stripped.count('"') % 2 == 1:
                issues.append({
                    'type': 'unterminated_string',
                    'line': i,
                    'content': line.strip()[:80]
                })
    
    # 3. Decimal() vide
    for i, line in enumerate(lines, 1):
        if re.search(r'Decimal\(\s*\)', line):
            issues.append({
                'type': 'empty_decimal',
                'line': i,
                'content': line.strip()
            })
    
    # 4. F-strings avec variables non définies
    for i, line in enumerate(lines, 1):
        if 'f"' in line or "f'" in line:
            # Chercher {var} qui pourrait ne pas exister
            vars_in_fstring = re.findall(r'\{([^}]+)\}', line)
            for var in vars_in_fstring:
                if var and not var[0].isdigit():  # Pas un index
                    if 'self.' not in var and var not in ['i', 'j', 'check']:
                        issues.append({
                            'type': 'undefined_fstring_var',
                            'line': i,
                            'var': var,
                            'content': line.strip()[:80]
                        })
    
    return issues

# Test avec le COBOL problématique
with open('user_input_files/pasted-text-2026-01-21T16-46-18.txt', 'r') as f:
    cobol_code = f.read()

result = generate_python_code(cobol_code, enhance=False)
if 'python_code' in result:
    issues = find_all_issues(result['python_code'])
    
    print(f"Found {len(issues)} issues:\n")
    
    # Grouper par type
    by_type = {}
    for issue in issues:
        t = issue['type']
        if t not in by_type:
            by_type[t] = []
        by_type[t].append(issue)
    
    for issue_type, items in by_type.items():
        print(f"\n{issue_type.upper()}: {len(items)} occurrences")
        for item in items[:5]:  # Montrer les 5 premiers
            print(f"  Line {item['line']}: {item['content']}")
else:
    print(f"Transpilation error: {result.get('error')}")
