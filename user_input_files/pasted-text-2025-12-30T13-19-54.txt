    def calc_auto_premium(self) -> None:
        """
        13220-CALC-AUTO-PREMIUM.
        """
        match self.WS_DRIVER_AGE:
            case age if 6 <= age <= 10:
                self.WS_BASE_PREMIUM += 100 # ADD 100 TO WS-BASE-PREMIUM
            case _:
                self.WS_BASE_PREMIUM += 50 # ADD 50 TO WS-BASE-PREMIUM
        

            self.WS_BASE_PREMIUM *= Decimal(1.5) # MULTIPLY 1.5 BY WS-BASE-PREMIUM
        

            self.WS_ACCIDENT_SURCHARGE = Decimal(self.WS_ACCIDENTS_3YR * 200) # COMPUTE WS-ACCIDENT-SURCHARGE = WS-ACCIDENTS-3YR * 200
            self.WS_BASE_PREMIUM += self.WS_ACCIDENT_SURCHARGE # ADD WS-ACCIDENT-SURCHARGE TO WS-BASE-PREMIUM
        

            self.WS_VIOLATION_SURCHARGE = Decimal(self.WS_VIOLATIONS_3YR * 100) # COMPUTE WS-VIOLATION-SURCHARGE = WS-VIOLATIONS-3YR * 100
            self.WS_BASE_PREMIUM += self.WS_VIOLATION_SURCHARGE # ADD WS-VIOLATION-SURCHARGE TO WS-BASE-PREMIUM
        
        self.WS_ANNUAL_PREMIUM = self.WS_BASE_PREMIUM # MOVE WS-BASE-PREMIUM TO WS-ANNUAL-PREMIUM
        self.WS_MONTHLY_PREMIUM = self.WS_ANNUAL_PREMIUM / Decimal(12) # COMPUTE WS-MONTHLY-PREMIUM = WS-ANNUAL-PREMIUM / 12.

    def calc_home_premium(self) -> None:
        """

def calculate_simple_interest(account_balance: Decimal, interest_rate: Decimal, days_in_period: int) -> Decimal:

    # COMPUTE WS-SIMPLE-INTEREST =

    return account_balance * interest_rate * Decimal(days_in_period) / Decimal('36500')

def calculate_compound_interest(account_balance: Decimal, interest_rate: Decimal, days_in_period: int) -> Decimal:
    """7300-CALCULATE-COMPOUND-INTEREST."""
    # COMPUTE WS-COMPOUND-FACTOR =
    # (1 + WS-INTEREST-RATE / 36500)
    # ** WS-DAYS-IN-PERIOD
    compound_factor = (1 + interest_rate / Decimal('36500')) ** days_in_period
    # COMPUTE WS-COMPOUND-INTEREST =
    # WS-ACCOUNT-BALANCE * (WS-COMPOUND-FACTOR - 1).
    return account_balance * (compound_factor - 1)

def calculate_next_run_date(ws_last_run_date: int, schedule_frequency: str) -> int:
    """
    Original COBOL:
    FUNCTION INTEGER-OF-DATE(WS-LAST-RUN-DATE) + 1
    WHEN 'WEEKLY'

        FUNCTION INTEGER-OF-DATE(WS-LAST-RUN-DATE) + 7
    WHEN 'MONTHLY'

        FUNCTION INTEGER-OF-DATE(WS-LAST-RUN-DATE) + 30
    WHEN 'QUARTERLY'

        FUNCTION INTEGER-OF-DATE(WS-LAST-RUN-DATE) + 90
    WHEN 'YEARLY'

        FUNCTION INTEGER-OF-DATE(WS-LAST-RUN-DATE) + 365
    END-EVALUATE.
    """
    match schedule_frequency:
        case 'WEEKLY':
            return ws_last_run_date + 7
        case 'MONTHLY':
            return ws_last_run_date + 30
        case 'QUARTERLY':
            return ws_last_run_date + 90
        case 'YEARLY':
            return ws_last_run_date + 365
        case _:
            return ws_last_run_date + 1

def collect_transaction_metrics(self):
    """
    Original COBOL:
    21110-COLLECT-TRANSACTION-METRICS.

                AT END

                NOT AT END

            END-READ
        END-PERFORM

                WS-TOTAL-TRANS-AMOUNT / WS-TOTAL-TRANS-COUNT
        END-IF

    """
    self.ws_total_trans_amount = Decimal("0")
    self.ws_total_trans_count = 0
    self.ws_avg_trans_amount = Decimal("0")
    self.ws_eof_flag = 'N'
    while self.ws_eof_flag == 'N':
        try:
            ws_trans_rec = self.transaction_file.read()
            self.ws_total_trans_count += 1
            self.ws_total_trans_amount += ws_trans_rec.trans_amount
        except EOFError:
            self.ws_eof_flag = 'Y'

        self.ws_avg_trans_amount = self.ws_total_trans_amount / self.ws_total_trans_count
    self.ws_eof_flag = 'N'

        END-PERFORM

    """
    self.ws_active_customers = 0

def calc_financial_kpi(self):
    """
    Original COBOL:
    21310-CALC-FINANCIAL-KPI.

                (WS-NET-INCOME / WS-TOTAL-ASSETS) * 100
        END-IF

                (WS-NET-INCOME / WS-TOTAL-EQUITY) * 100
        END-IF

                ((WS-INTEREST-INCOME - WS-INTEREST-EXPENSE) /
                WS-EARNING-ASSETS) * 100
        END-IF.
    """

        self.ws_roa = (self.ws_net_income / self.ws_total_assets) * 100

        self.ws_roe = (self.ws_net_income / self.ws_total_equity) * 100

        self.ws_nim = ((self.ws_interest_income - self.ws_interest_expense) /
                       self.ws_earning_assets) * 100

def calc_operational_kpi(self):
    """
    Original COBOL:
    21320-CALC-OPERATIONAL-KPI.

                (WS-ERROR-COUNT / WS-TOTAL-TRANS-COUNT) * 100
        END-IF

            (WS-WITHIN-SLA-COUNT / WS-TOTAL-CASES) * 100

            (WS-FCR-COUNT / WS-TOTAL-CALLS) * 100.
    """

        self.ws_error_rate = (self.ws_error_count / self.ws_total_trans_count) * 100
    self.ws_sla_compliance = (self.ws_within_sla_count / self.ws_total_cases) * 100
    self.ws_first_call_resolution = (self.ws_fcr_count / self.ws_total_calls) * 100

def calc_customer_kpi(self):
    """
    Original COBOL:
    21330-CALC-CUSTOMER-KPI.

                (WS-CHURNED-CUSTOMERS / WS-ACTIVE-CUSTOMERS) * 100
        END-IF

            WS-MARKETING-SPEND / WS-NEW-CUSTOMERS

            WS-AVG-REVENUE-PER-CUSTOMER * WS-AVG-CUSTOMER-TENURE.
    """

        self.ws_churn_rate = (self.ws_churned_customers / self.ws_active_customers) * 100
    self.ws_acquisition_cost = self.ws_marketing_spend / self.ws_new_customers
    self.ws_lifetime_value = self.ws_avg_revenue_per_customer * self.ws_avg_customer_tenure

    """

                        ',' DELIMITED SIZE

                        ',' DELIMITED SIZE
                        DAILY-TRANS-AMOUNT DELIMITED SIZE
                        ',' DELIMITED SIZE
                        DAILY-DEPOSITS DELIMITED SIZE
                        ',' DELIMITED SIZE
                        DAILY-WITHDRAWALS DELIMITED SIZE
                    INTO WS-CSV-LINE

            END-READ
        END-PERFORM

    """

    self.ws_csv_header = 'Date,TransCount,TransAmount,Deposits,Withdrawals'
