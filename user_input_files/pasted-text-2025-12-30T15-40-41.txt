def calculate_tax_C2() -> None:
    """8500-calculate_tax."""
    global WS_CALC_AMOUNT, WS_BRACKET_1_MAX, WS_BRACKET_1_RATE, WS_BRACKET_2_MAX, WS_BRACKET_2_RATE, WS_BRACKET_3_MAX, WS_BRACKET_3_RATE, WS_BRACKET_5_RATE, WS_CALC_TAX
    match True:  # EVALUATE TRUE
        case WS_CALC_AMOUNT <= WS_BRACKET_1_MAX:  # WHEN ws_calc_amount <= ws_bracket_1_max
            WS_CALC_TAX = WS_CALC_AMOUNT * WS_BRACKET_1_RATE  # COMPUTE ws_calc_tax = ws_calc_amount * ws_bracket_1_rate
        case WS_CALC_AMOUNT <= WS_BRACKET_2_MAX:  # WHEN ws_calc_amount <= ws_bracket_2_max
            WS_CALC_TAX = (WS_BRACKET_1_MAX * WS_BRACKET_1_RATE) + ((WS_CALC_AMOUNT - WS_BRACKET_1_MAX) * WS_BRACKET_2_RATE)  # COMPUTE ws_calc_tax = (ws_bracket_1_max * ws_bracket_1_rate) + ((ws_calc_amount - ws_bracket_1_max) * ws_bracket_2_rate)
        case WS_CALC_AMOUNT <= WS_BRACKET_3_MAX:  # WHEN ws_calc_amount <= ws_bracket_3_max
            WS_CALC_TAX = (WS_BRACKET_1_MAX * WS_BRACKET_1_RATE) + ((WS_BRACKET_2_MAX - WS_BRACKET_1_MAX) * WS_BRACKET_2_RATE) + ((WS_CALC_AMOUNT - WS_BRACKET_2_MAX) * WS_BRACKET_3_RATE)  # COMPUTE ws_calc_tax = (ws_bracket_1_max * ws_bracket_1_rate) + ((ws_bracket_2_max - ws_bracket_1_max) * ws_bracket_2_rate) + ((ws_calc_amount - ws_bracket_2_max) * ws_bracket_3_rate)
        case _:  # WHEN OTHER
            WS_CALC_TAX = WS_CALC_AMOUNT * WS_BRACKET_5_RATE  # COMPUTE ws_calc_tax = ws_calc_amount * ws_bracket_5_rate
