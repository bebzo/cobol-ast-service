       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
           SELECT MASTER-ACCOUNT-FILE
               ASSIGN TO DISK-ACCT-MASTER
               ORGANIZATION IS INDEXED
               ACCESS MODE IS DYNAMIC
               RECORD KEY IS ACCT-KEY-PRIMARY
               ALTERNATE RECORD KEY IS ACCT-KEY-SSN WITH DUPLICATES
               ALTERNATE RECORD KEY IS ACCT-KEY-BIOMETRIC
               FILE STATUS IS FS-MASTER
               LOCK MODE IS AUTOMATIC WITH LOCK ON MULTIPLE RECORDS
               WITH ROLLBACK.

           SELECT TRANSACTION-LOG-FILE
               ASSIGN TO DISK-TRANS-LOG
               ORGANIZATION IS SEQUENTIAL
               ACCESS MODE IS SEQUENTIAL
               FILE STATUS IS FS-TRANS-LOG
               ENCRYPTION KEY IS MASTER-ENCRYPTION-KEY
               WITH WRITE VERIFY
               WITH AUDIT TRAIL.

           SELECT ENCRYPTION-KEY-FILE
               ASSIGN TO HSM-KEY-VAULT
               ORGANIZATION IS INDEXED
               ACCESS MODE IS RANDOM
               RECORD KEY IS KEY-IDENTIFIER
               FILE STATUS IS FS-KEY-VAULT
               EXTERNAL SECURITY REQUIRED.

           SELECT BIOMETRIC-DATA-FILE
               ASSIGN TO SECURE-BIO-STORAGE
               ORGANIZATION IS RELATIVE
               ACCESS MODE IS DYNAMIC
               RELATIVE KEY IS BIO-RECORD-NUMBER
               FILE STATUS IS FS-BIOMETRIC
               ENCRYPTION LEVEL IS MAXIMUM.

           SELECT FRAUD-DETECTION-CONFIG
               ASSIGN TO ML-MODEL-STORAGE
               ORGANIZATION IS SEQUENTIAL
               ACCESS MODE IS SEQUENTIAL
               FILE STATUS IS FS-FRAUD-CONFIG.

           SELECT AUDIT-TRAIL-FILE
               ASSIGN TO IMMUTABLE-AUDIT-LOG
               ORGANIZATION IS SEQUENTIAL
               ACCESS MODE IS APPEND-ONLY
               FILE STATUS IS FS-AUDIT
               WITH BLOCKCHAIN-HASH
               WITH TIMESTAMP-VERIFICATION.

           SELECT GEO-LOCATION-DATABASE
               ASSIGN TO GEO-IP-DB
               ORGANIZATION IS INDEXED
               ACCESS MODE IS RANDOM
               RECORD KEY IS GEO-IP-KEY
               FILE STATUS IS FS-GEO.

           SELECT REGULATORY-RULES-FILE
               ASSIGN TO COMPLIANCE-ENGINE
               ORGANIZATION IS INDEXED
               ACCESS MODE IS DYNAMIC
               RECORD KEY IS RULE-ID
               FILE STATUS IS FS-COMPLIANCE.

   PROCEDURE DIVISION USING LS-TRANSACTION-REQUEST
                             LS-TRANSACTION-RESPONSE.

   000-MAIN-CONTROLLER SECTION.
       PERFORM 100-INITIALIZE-SYSTEM
       PERFORM 200-VALIDATE-SESSION-SECURITY
       IF WS-ERROR-CODE = SPACES
           PERFORM 300-AUTHENTICATE-USER
           IF WS-ERROR-CODE = SPACES
               PERFORM 400-CHECK-RATE-LIMITING
               IF RATE-LIMIT-OK
                   PERFORM 500-VALIDATE-GEO-LOCATION
                   IF WS-ERROR-CODE = SPACES
                       PERFORM 600-PROCESS-TRANSACTION
                       PERFORM 700-FRAUD-DETECTION-ANALYSIS
                       PERFORM 800-COMPLIANCE-VERIFICATION
                       IF WS-ERROR-CODE = SPACES
                           PERFORM 900-COMMIT-DISTRIBUTED-TRANSACTION
                       ELSE
                           PERFORM 950-ROLLBACK-TRANSACTION
                       END-IF
                   END-IF
               ELSE
                   MOVE 'RATE_LIMIT' TO WS-ERROR-CODE
                   MOVE 'Rate limit exceeded' TO WS-ERROR-MESSAGE
               END-IF
           END-IF
       END-IF
       
       PERFORM 990-FINALIZE-AND-RESPOND
       PERFORM 999-AUDIT-LOG-TRANSACTION
       STOP RUN.

   100-INITIALIZE-SYSTEM SECTION.
       MOVE FUNCTION CURRENT-DATE TO WS-CURRENT-DATE-TIME
       MOVE FUNCTION WHEN-COMPILED TO WS-TIMESTAMP-BUFFER
       
       OPEN INPUT ENCRYPTION-KEY-FILE
       OPEN INPUT FRAUD-DETECTION-CONFIG
       OPEN INPUT GEO-LOCATION-DATABASE
       OPEN INPUT REGULATORY-RULES-FILE
       OPEN INPUT BIOMETRIC-DATA-FILE
       
       OPEN I-O MASTER-ACCOUNT-FILE
       OPEN EXTEND TRANSACTION-LOG-FILE
       OPEN EXTEND AUDIT-TRAIL-FILE
       
       IF FS-MASTER NOT = '00'
           MOVE 'FILE_ERROR' TO WS-ERROR-CODE
           STRING 'Failed to open MASTER-ACCOUNT-FILE: '
                  FS-MASTER
                  DELIMITED BY SIZE
                  INTO WS-ERROR-MESSAGE
           PERFORM 9000-LOG-CRITICAL-ERROR
       END-IF
       
       PERFORM 110-LOAD-ENCRYPTION-KEYS
       PERFORM 120-LOAD-FRAUD-MODEL
       PERFORM 130-INITIALIZE-RATE-LIMITER.

   110-LOAD-ENCRYPTION-KEYS SECTION.
       MOVE 'MASTER-KEY-2024' TO KEY-IDENTIFIER
       READ ENCRYPTION-KEY-FILE KEY IS KEY-IDENTIFIER
           INVALID KEY
               MOVE 'KEY_NOT_FOUND' TO WS-ERROR-CODE
               MOVE 'Master encryption key not found in HSM' 
                    TO WS-ERROR-MESSAGE
               PERFORM 9000-LOG-CRITICAL-ERROR
       END-READ
       
       IF FS-KEY-VAULT = '00'
           PERFORM 111-DECRYPT-MASTER-KEY
           PERFORM 112-CACHE-SESSION-KEYS
       END-IF.

   111-DECRYPT-MASTER-KEY SECTION.
       MOVE 'DECRYPT' TO HSM-OPERATION
       MOVE KEY-IDENTIFIER TO HSM-KEY-ID
       MOVE KEY-ENCRYPTED-VALUE TO HSM-DATA-IN
       
       CALL 'HSM-CRYPTO-ENGINE' USING WS-HSM-REQUEST
       
       IF HSM-STATUS NOT = '00'
           MOVE 'HSM_ERROR' TO WS-ERROR-CODE
           MOVE 'Hardware Security Module failure' 
                TO WS-ERROR-MESSAGE
           PERFORM 9000-LOG-CRITICAL-ERROR
       ELSE
           MOVE HSM-DATA-OUT TO WS-AES-KEY
       END-IF.

   112-CACHE-SESSION-KEYS SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 UNTIL WS-I > 10
           MOVE KEY-IDENTIFIER TO CACHE-KEY-ID(WS-I)
           MOVE KEY-ENCRYPTED-VALUE TO CACHE-KEY-VALUE(WS-I)
           ADD 3600 TO WS-CURRENT-TIMESTAMP 
               GIVING CACHE-KEY-EXPIRY(WS-I)
       END-PERFORM.

   120-LOAD-FRAUD-MODEL SECTION.
       READ FRAUD-DETECTION-CONFIG
           AT END
               MOVE 'MODEL_ERROR' TO WS-ERROR-CODE
               MOVE 'Fraud detection model not loaded' 
                    TO WS-ERROR-MESSAGE
       END-READ.

   130-INITIALIZE-RATE-LIMITER SECTION.
       PERFORM VARYING BUCKET-IDX FROM 1 BY 1 UNTIL BUCKET-IDX > 1000
           MOVE SPACES TO BUCKET-USER-ID(BUCKET-IDX)
           MOVE ZERO TO BUCKET-REQUEST-COUNT(BUCKET-IDX)
           MOVE WS-CURRENT-TIMESTAMP TO BUCKET-WINDOW-START(BUCKET-IDX)
       END-PERFORM.

   200-VALIDATE-SESSION-SECURITY SECTION.
       IF LS-SESSION-TOKEN = SPACES
           MOVE 'AUTH_REQUIRED' TO WS-ERROR-CODE
           MOVE 'Session token is required' TO WS-ERROR-MESSAGE
           EXIT SECTION
       END-IF
       
       PERFORM 210-VERIFY-SESSION-TOKEN
       IF WS-ERROR-CODE = SPACES
           PERFORM 220-CHECK-SESSION-EXPIRY
           PERFORM 230-VERIFY-IP-ADDRESS
           PERFORM 240-CHECK-DEVICE-FINGERPRINT
       END-IF.

   210-VERIFY-SESSION-TOKEN SECTION.
       MOVE LS-USER-ID TO ACCT-KEY-PRIMARY
       READ MASTER-ACCOUNT-FILE KEY IS ACCT-KEY-PRIMARY
           INVALID KEY
               MOVE 'USER_NOT_FOUND' TO WS-ERROR-CODE
               MOVE 'User account not found' TO WS-ERROR-MESSAGE
       END-READ
       
       IF FS-MASTER = '00'
           IF SEC-SESSION-TOKEN NOT = LS-SESSION-TOKEN
               MOVE 'INVALID_SESSION' TO WS-ERROR-CODE
               MOVE 'Session token is invalid' TO WS-ERROR-MESSAGE
               PERFORM 9100-LOG-SECURITY-EVENT
           END-IF
       END-IF.

   220-CHECK-SESSION-EXPIRY SECTION.
       IF WS-CURRENT-TIMESTAMP > SEC-SESSION-EXPIRY
           MOVE 'SESSION_EXPIRED' TO WS-ERROR-CODE
           MOVE 'Session has expired' TO WS-ERROR-MESSAGE
           PERFORM 221-INVALIDATE-SESSION
       END-IF.

   221-INVALIDATE-SESSION SECTION.
       MOVE SPACES TO SEC-SESSION-TOKEN
       MOVE ZERO TO SEC-SESSION-EXPIRY
       REWRITE ACCOUNT-MASTER-RECORD
           INVALID KEY
               MOVE 'UPDATE_ERROR' TO WS-ERROR-CODE
       END-REWRITE.

   230-VERIFY-IP-ADDRESS SECTION.
       PERFORM 231-CHECK-IP-WHITELIST
       IF WS-ERROR-CODE = SPACES
           PERFORM 232-CHECK-GEO-RESTRICTIONS
       END-IF.

   231-CHECK-IP-WHITELIST SECTION.
       SET WS-I TO 1
       SEARCH WHITELIST-IP
           AT END
               CONTINUE
           WHEN WHITELIST-IP(WS-I) = LS-METADATA(1:45)
               CONTINUE
       END-SEARCH
       
       IF WS-I > 50
           ADD 10 TO WS-FRAUD-SCORE-CALCULATED
       END-IF.

   232-CHECK-GEO-RESTRICTIONS SECTION.
       PERFORM 2321-LOOKUP-GEO-DATABASE
       IF GEO-RISK-BLOCKED
           MOVE 'GEO_BLOCKED' TO WS-ERROR-CODE
           STRING 'Access from ' GEO-COUNTRY-NAME 
                  ' is not allowed'
                  DELIMITED BY SIZE
                  INTO WS-ERROR-MESSAGE
           PERFORM 9100-LOG-SECURITY-EVENT
       END-IF.

   2321-LOOKUP-GEO-DATABASE SECTION.
       MOVE 0 TO GEO-IP-START
       MOVE 4294967295 TO GEO-IP-END
       READ GEO-LOCATION-DATABASE KEY IS GEO-IP-KEY
           INVALID KEY
               MOVE 'UNKNOWN' TO GEO-COUNTRY-CODE
               MOVE 2 TO GEO-RISK-LEVEL
       END-READ.

   240-CHECK-DEVICE-FINGERPRINT SECTION.
       SET WS-I TO 1
       SEARCH DEVICE-ID
           AT END
               PERFORM 241-REGISTER-NEW-DEVICE
           WHEN DEVICE-ID(WS-I) = LS-METADATA(100:64)
               MOVE WS-CURRENT-TIMESTAMP TO DEVICE-LAST-SEEN(WS-I)
       END-SEARCH.

   241-REGISTER-NEW-DEVICE SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 UNTIL WS-I > 20
           IF DEVICE-ID(WS-I) = SPACES
               MOVE LS-METADATA(100:64) TO DEVICE-ID(WS-I)
               MOVE WS-CURRENT-TIMESTAMP TO DEVICE-LAST-SEEN(WS-I)
               EXIT PERFORM
           END-IF
       END-PERFORM
       
       REWRITE ACCOUNT-MASTER-RECORD
           INVALID KEY
               MOVE 'UPDATE_ERROR' TO WS-ERROR-CODE
       END-REWRITE.

   300-AUTHENTICATE-USER SECTION.
       PERFORM 310-VERIFY-PASSWORD-OR-BIOMETRIC
       IF WS-ERROR-CODE = SPACES
           IF TWO-FACTOR-ACTIVE
               PERFORM 320-VERIFY-2FA-TOKEN
           END-IF
           IF BIO-AUTH-REQUIRED AND WS-ERROR-CODE = SPACES
               PERFORM 330-VERIFY-BIOMETRIC-DATA
           END-IF
       END-IF
       
       IF WS-ERROR-CODE = SPACES
           PERFORM 340-UPDATE-LOGIN-SUCCESS
       ELSE
           PERFORM 350-HANDLE-LOGIN-FAILURE
       END-IF.

   310-VERIFY-PASSWORD-OR-BIOMETRIC SECTION.
       EVALUATE TRUE
           WHEN AUTH-PASSWORD
               PERFORM 311-VERIFY-PASSWORD-HASH
           WHEN AUTH-BIOMETRIC
               PERFORM 312-VERIFY-BIOMETRIC-AUTH
           WHEN OTHER
               MOVE 'INVALID_AUTH_METHOD' TO WS-ERROR-CODE
       END-EVALUATE.

   311-VERIFY-PASSWORD-HASH SECTION.
       MOVE LS-METADATA(200:256) TO WS-PASSWORD-HASH-OUTPUT
       
       CALL 'ARGON2-VERIFY' USING 
           WS-PASSWORD-HASH-OUTPUT
           SEC-PASSWORD-HASH
           SEC-PASSWORD-SALT
           SEC-PASSWORD-ITERATIONS
           HSM-STATUS
       
       IF HSM-STATUS NOT = '00'
           MOVE 'INVALID_PASSWORD' TO WS-ERROR-CODE
           MOVE 'Password verification failed' TO WS-ERROR-MESSAGE
       END-IF.

   312-VERIFY-BIOMETRIC-AUTH SECTION.
       MOVE LS-USER-ID TO BIO-USER-ID
       READ BIOMETRIC-DATA-FILE KEY IS BIO-USER-ID
           INVALID KEY
               MOVE 'BIO_NOT_ENROLLED' TO WS-ERROR-CODE
               MOVE 'Biometric data not enrolled' TO WS-ERROR-MESSAGE
       END-READ
       
       IF FS-BIOMETRIC = '00'
           PERFORM 3121-COMPARE-BIOMETRIC-DATA
       END-IF.

   3121-COMPARE-BIOMETRIC-DATA SECTION.
       CALL 'BIOMETRIC-MATCHER' USING
           LS-METADATA(500:2048)
           BIO-FINGERPRINT-DATA
           WS-BIOMETRIC-MATCH-SCORE
           HSM-STATUS
       
       IF WS-BIOMETRIC-MATCH-SCORE < 98.00
           MOVE 'BIO_MISMATCH' TO WS-ERROR-CODE
           MOVE 'Biometric authentication failed' TO WS-ERROR-MESSAGE
       END-IF.

   320-VERIFY-2FA-TOKEN SECTION.
       MOVE LS-METADATA(300:6) TO WS-CHALLENGE-RESPONSE
       
       CALL 'TOTP-VERIFY' USING
           WS-CHALLENGE-RESPONSE
           SEC-2FA-SECRET
           WS-CURRENT-TIMESTAMP
           HSM-STATUS
       
       IF HSM-STATUS NOT = '00'
           PERFORM 321-TRY-BACKUP-CODE
       END-IF.

   321-TRY-BACKUP-CODE SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 UNTIL WS-I > 10
           IF SEC-2FA-BACKUP-CODES(WS-I) = WS-CHALLENGE-RESPONSE
               MOVE SPACES TO SEC-2FA-BACKUP-CODES(WS-I)
               EXIT PERFORM
           END-IF
       END-PERFORM
       
       IF WS-I > 10
           MOVE 'INVALID_2FA' TO WS-ERROR-CODE
           MOVE '2FA token verification failed' TO WS-ERROR-MESSAGE
       END-IF.

   330-VERIFY-BIOMETRIC-DATA SECTION.
       PERFORM 312-VERIFY-BIOMETRIC-AUTH.

   340-UPDATE-LOGIN-SUCCESS SECTION.
       MOVE WS-CURRENT-TIMESTAMP TO SEC-LAST-LOGIN
       MOVE LS-METADATA(1:45) TO SEC-LAST-IP-ADDRESS
       MOVE LS-METADATA(100:64) TO SEC-LAST-DEVICE-ID
       MOVE ZERO TO SEC-FAILED-LOGIN-COUNT
       
       COMPUTE SEC-RISK-SCORE = WS-FRAUD-SCORE-CALCULATED
       
       REWRITE ACCOUNT-MASTER-RECORD
           INVALID KEY
               MOVE 'UPDATE_ERROR' TO WS-ERROR-CODE
       END-REWRITE
       
       PERFORM 9200-LOG-AUDIT-TRAIL.

   350-HANDLE-LOGIN-FAILURE SECTION.
       ADD 1 TO SEC-FAILED-LOGIN-COUNT
       
       IF SEC-FAILED-LOGIN-COUNT >= WS-MAX-LOGIN-ATTEMPTS
           PERFORM 351-LOCK-ACCOUNT
       END-IF
       
       REWRITE ACCOUNT-MASTER-RECORD
       PERFORM 9100-LOG-SECURITY-EVENT.

   351-LOCK-ACCOUNT SECTION.
       MOVE 'Y' TO SEC-ACCOUNT-LOCKED
       MOVE 'Maximum login attempts exceeded' TO SEC-LOCK-REASON
       MOVE WS-CURRENT-TIMESTAMP TO SEC-LOCK-TIMESTAMP
       
       PERFORM 9300-SEND-SECURITY-ALERT.

   400-CHECK-RATE-LIMITING SECTION.
       PERFORM 410-FIND-OR-CREATE-BUCKET
       PERFORM 420-EVALUATE-RATE-LIMIT.

   410-FIND-OR-CREATE-BUCKET SECTION.
       SET BUCKET-IDX TO 1
       SEARCH BUCKET-USER-ID
           AT END
               PERFORM 411-CREATE-NEW-BUCKET
           WHEN BUCKET-USER-ID(BUCKET-IDX) = LS-USER-ID
               PERFORM 412-UPDATE-BUCKET-WINDOW
       END-SEARCH.

   411-CREATE-NEW-BUCKET SECTION.
       PERFORM VARYING BUCKET-IDX FROM 1 BY 1 UNTIL BUCKET-IDX > 1000
           IF BUCKET-USER-ID(BUCKET-IDX) = SPACES
               MOVE LS-USER-ID TO BUCKET-USER-ID(BUCKET-IDX)
               MOVE 1 TO BUCKET-REQUEST-COUNT(BUCKET-IDX)
               MOVE WS-CURRENT-TIMESTAMP 
                    TO BUCKET-WINDOW-START(BUCKET-IDX)
               EXIT PERFORM
           END-IF
       END-PERFORM.

   412-UPDATE-BUCKET-WINDOW SECTION.
       COMPUTE WS-ELAPSED-MICROSECONDS =
           WS-CURRENT-TIMESTAMP - BUCKET-WINDOW-START(BUCKET-IDX)
       
       IF WS-ELAPSED-MICROSECONDS > 3600000000
           MOVE 1 TO BUCKET-REQUEST-COUNT(BUCKET-IDX)
           MOVE WS-CURRENT-TIMESTAMP 
                TO BUCKET-WINDOW-START(BUCKET-IDX)
       ELSE
           ADD 1 TO BUCKET-REQUEST-COUNT(BUCKET-IDX)
       END-IF.

   420-EVALUATE-RATE-LIMIT SECTION.
       IF BUCKET-REQUEST-COUNT(BUCKET-IDX) > 
          WS-MAX-TRANSACTIONS-PER-HOUR
           SET RATE-LIMIT-BLOCKED TO TRUE
           PERFORM 9100-LOG-SECURITY-EVENT
       ELSE
           SET RATE-LIMIT-OK TO TRUE
       END-IF.

   500-VALIDATE-GEO-LOCATION SECTION.
       PERFORM 510-EXTRACT-GEO-FROM-IP
       PERFORM 520-CHECK-GEO-RESTRICTIONS
       PERFORM 530-CALCULATE-GEO-RISK.

   510-EXTRACT-GEO-FROM-IP SECTION.
       MOVE LS-METADATA(1:45) TO WS-PLAINTEXT-BUFFER
       CALL 'IP-TO-GEO' USING
           WS-PLAINTEXT-BUFFER
           GEO-COUNTRY-CODE
           GEO-LATITUDE
           GEO-LONGITUDE
           HSM-STATUS.

   520-CHECK-GEO-RESTRICTIONS SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 UNTIL WS-I > 30
           IF GEO-BLOCKED-COUNTRIES(WS-I) = GEO-COUNTRY-CODE
               MOVE 'GEO_BLOCKED' TO WS-ERROR-CODE
               EXIT PERFORM
           END-IF
       END-PERFORM.

   530-CALCULATE-GEO-RISK SECTION.
       IF SEC-LAST-IP-ADDRESS NOT = SPACES
           PERFORM 531-CALCULATE-DISTANCE
           IF FEAT-LOCATION-DISTANCE > 1000.00
               ADD 15 TO WS-FRAUD-SCORE-CALCULATED
           END-IF
       END-IF.

   531-CALCULATE-DISTANCE SECTION.
       CALL 'HAVERSINE-DISTANCE' USING
           SEC-GEO-LOCATION
           GEO-LATITUDE
           GEO-LONGITUDE
           FEAT-LOCATION-DISTANCE.

   600-PROCESS-TRANSACTION SECTION.
       PERFORM 610-VALIDATE-TRANSACTION-PARAMETERS
       IF WS-ERROR-CODE = SPACES
           PERFORM 620-LOCK-ACCOUNTS
           PERFORM 630-VERIFY-SUFFICIENT-BALANCE
           IF WS-ERROR-CODE = SPACES
               PERFORM 640-CALCULATE-FEES-AND-TAXES
               PERFORM 650-ENCRYPT-TRANSACTION-DATA
               PERFORM 660-EXECUTE-TRANSACTION
           END-IF
           PERFORM 670-UNLOCK-ACCOUNTS
       END-IF.

   610-VALIDATE-TRANSACTION-PARAMETERS SECTION.
       IF LS-AMOUNT <= ZERO
           MOVE 'INVALID_AMOUNT' TO WS-ERROR-CODE
           MOVE 'Transaction amount must be positive' 
                TO WS-ERROR-MESSAGE
           EXIT SECTION
       END-IF
       
       IF LS-AMOUNT > FIN-DAILY-LIMIT
           MOVE 'LIMIT_EXCEEDED' TO WS-ERROR-CODE
           MOVE 'Transaction exceeds daily limit' TO WS-ERROR-MESSAGE
           EXIT SECTION
       END-IF
       
       COMPUTE WS-TRANS-TOTAL = FIN-DAILY-USED + LS-AMOUNT
       IF WS-TRANS-TOTAL > FIN-DAILY-LIMIT
           MOVE 'DAILY_LIMIT' TO WS-ERROR-CODE
           MOVE 'Daily transaction limit would be exceeded' 
                TO WS-ERROR-MESSAGE
       END-IF.

   620-LOCK-ACCOUNTS SECTION.
       MOVE LS-SOURCE-ACCOUNT TO ACCT-KEY-PRIMARY
       READ MASTER-ACCOUNT-FILE KEY IS ACCT-KEY-PRIMARY
           WITH LOCK
           INVALID KEY
               MOVE 'ACCOUNT_NOT_FOUND' TO WS-ERROR-CODE
           NOT INVALID KEY
               IF ACCOUNT-LOCKED OR ACCOUNT-SUSPENDED OR ACCOUNT-FROZEN
                   MOVE 'ACCOUNT_LOCKED' TO WS-ERROR-CODE
               END-IF
       END-READ
       
       IF LS-TRANSACTION-TYPE = 'TRF' AND WS-ERROR-CODE = SPACES
           PERFORM 621-LOCK-TARGET-ACCOUNT
       END-IF.

   621-LOCK-TARGET-ACCOUNT SECTION.
       MOVE LS-TARGET-ACCOUNT TO ACCT-KEY-PRIMARY
       READ MASTER-ACCOUNT-FILE KEY IS ACCT-KEY-PRIMARY
           WITH LOCK
           INVALID KEY
               MOVE 'TARGET_NOT_FOUND' TO WS-ERROR-CODE
       END-READ.

   630-VERIFY-SUFFICIENT-BALANCE SECTION.
       PERFORM 631-DECRYPT-BALANCE
       
       IF FIN-AVAILABLE-BALANCE < LS-AMOUNT
           IF FIN-AVAILABLE-BALANCE + FIN-OVERDRAFT-LIMIT >= LS-AMOUNT
               CONTINUE
           ELSE
               MOVE 'INSUFFICIENT_FUNDS' TO WS-ERROR-CODE
               MOVE 'Insufficient funds for transaction' 
                    TO WS-ERROR-MESSAGE
           END-IF
       END-IF.

   631-DECRYPT-BALANCE SECTION.
       MOVE 'DECRYPT' TO HSM-OPERATION
       MOVE ENC-KEY-ID TO HSM-KEY-ID
       MOVE FIN-BALANCE-ENCRYPTED TO HSM-DATA-IN
       
       CALL 'HSM-CRYPTO-ENGINE' USING WS-HSM-REQUEST
       
       IF HSM-STATUS = '00'
           MOVE HSM-DATA-OUT TO WS-DECRYPTED-BUFFER
           MOVE WS-DECRYPTED-BUFFER TO FIN-AVAILABLE-BALANCE
       ELSE
           MOVE 'DECRYPTION_ERROR' TO WS-ERROR-CODE
       END-IF.

   640-CALCULATE-FEES-AND-TAXES SECTION.
       EVALUATE LS-TRANSACTION-TYPE
           WHEN 'WDR'
               COMPUTE WS-TRANS-FEE = LS-AMOUNT * 0.015
               IF WS-TRANS-FEE < 5.00
                   MOVE 5.00 TO WS-TRANS-FEE
               END-IF
           WHEN 'TRF'
               COMPUTE WS-TRANS-FEE = LS-AMOUNT * 0.010
               IF WS-TRANS-FEE < 10.00
                   MOVE 10.00 TO WS-TRANS-FEE
               END-IF
           WHEN OTHER
               MOVE ZERO TO WS-TRANS-FEE
       END-EVALUATE
       
       COMPUTE WS-TRANS-TAX = WS-TRANS-FEE * 0.19
       COMPUTE WS-TRANS-TOTAL = LS-AMOUNT + WS-TRANS-FEE + WS-TRANS-TAX.

   650-ENCRYPT-TRANSACTION-DATA SECTION.
       MOVE LS-AMOUNT TO WS-PLAINTEXT-BUFFER
       
       PERFORM 651-GENERATE-TRANSACTION-KEY
       
       MOVE 'ENCRYPT' TO HSM-OPERATION
       MOVE ENC-KEY-ID TO HSM-KEY-ID
       MOVE WS-PLAINTEXT-BUFFER TO HSM-DATA-IN
       
       CALL 'HSM-CRYPTO-ENGINE' USING WS-HSM-REQUEST
       
       IF HSM-STATUS = '00'
           MOVE HSM-DATA-OUT TO TRANS-AMOUNT-ENCRYPTED
       ELSE
           MOVE 'ENCRYPTION_ERROR' TO WS-ERROR-CODE
       END-IF.

   651-GENERATE-TRANSACTION-KEY SECTION.
       CALL 'UUID-GENERATE-V4' USING WS-TRANS-ID
       
       MOVE WS-TRANS-ID TO ENC-TRANSACTION-KEY-ID
       
       PERFORM 652-GENERATE-NONCE.

   652-GENERATE-NONCE SECTION.
       CALL 'RANDOM-BYTES' USING
           16
           WS-AES-IV
           HSM-STATUS.

   660-EXECUTE-TRANSACTION SECTION.
       PERFORM 661-DEBIT-SOURCE-ACCOUNT
       IF WS-ERROR-CODE = SPACES AND LS-TRANSACTION-TYPE = 'TRF'
           PERFORM 662-CREDIT-TARGET-ACCOUNT
       END-IF
       
       IF WS-ERROR-CODE = SPACES
           PERFORM 663-LOG-TRANSACTION
           ADD WS-TRANS-TOTAL TO FIN-DAILY-USED
           REWRITE ACCOUNT-MASTER-RECORD
       END-IF.

   661-DEBIT-SOURCE-ACCOUNT SECTION.
       SUBTRACT WS-TRANS-TOTAL FROM FIN-AVAILABLE-BALANCE
       
       IF FIN-AVAILABLE-BALANCE < ZERO
           IF FIN-AVAILABLE-BALANCE + FIN-OVERDRAFT-LIMIT < ZERO
               MOVE 'INSUFFICIENT_FUNDS' TO WS-ERROR-CODE
               ADD WS-TRANS-TOTAL TO FIN-AVAILABLE-BALANCE
           END-IF
       END-IF
       
       PERFORM 631-DECRYPT-BALANCE.

   662-CREDIT-TARGET-ACCOUNT SECTION.
       ADD LS-AMOUNT TO FIN-AVAILABLE-BALANCE
       PERFORM 631-DECRYPT-BALANCE.

   663-LOG-TRANSACTION SECTION.
       MOVE WS-TRANS-ID TO TRANS-ID
       MOVE WS-CURRENT-TIMESTAMP TO TRANS-TIMESTAMP
       MOVE LS-TRANSACTION-TYPE TO TRANS-TYPE
       MOVE TRANS-AMOUNT-ENCRYPTED TO TRANS-AMOUNT-ENCRYPTED
       MOVE LS-SOURCE-ACCOUNT TO TRANS-SOURCE-ACCOUNT
       MOVE LS-TARGET-ACCOUNT TO TRANS-TARGET-ACCOUNT
       MOVE 'C' TO TRANS-STATUS
       
       MOVE LS-USER-ID TO SEC-USER-ID
       MOVE LS-SESSION-TOKEN TO SEC-SESSION-ID
       MOVE LS-METADATA(1:45) TO SEC-IP-ADDRESS
       MOVE LS-METADATA(100:64) TO SEC-DEVICE-FINGERPRINT
       
       WRITE TRANSACTION-LOG-RECORD
           INVALID KEY
               MOVE 'LOG_ERROR' TO WS-ERROR-CODE
       END-WRITE.

   670-UNLOCK-ACCOUNTS SECTION.
       UNLOCK MASTER-ACCOUNT-FILE ALL RECORDS.

        700-FRAUD-DETECTION-ANALYSIS SECTION.
       PERFORM 710-CALCULATE-FRAUD-FEATURES
       PERFORM 720-ML-FRAUD-PREDICTION
       PERFORM 730-EVALUATE-FRAUD-DECISION.

   710-CALCULATE-FRAUD-FEATURES SECTION.
       PERFORM 711-CALCULATE-AMOUNT-ZSCORE
       PERFORM 712-CALCULATE-FREQUENCY-PATTERN
       PERFORM 713-CALCULATE-TIME-ANOMALY
       PERFORM 714-CALCULATE-MERCHANT-RISK
       PERFORM 715-CALCULATE-DEVICE-TRUST
       PERFORM 716-CALCULATE-VELOCITY.

   711-CALCULATE-AMOUNT-ZSCORE SECTION.
       COMPUTE FEAT-AMOUNT-ZSCORE =
           (LS-AMOUNT - BEHAVIOR-AVG-TRANS-AMT) / 
           (BEHAVIOR-AVG-TRANS-AMT * 0.3).

   712-CALCULATE-FREQUENCY-PATTERN SECTION.
       COMPUTE FEAT-FREQUENCY-ZSCORE =
           (1 - BEHAVIOR-AVG-TRANS-FREQ) / 
           (BEHAVIOR-AVG-TRANS-FREQ * 0.2).

   713-CALCULATE-TIME-ANOMALY SECTION.
       MOVE WS-CURRENT-HOUR TO WS-I
       COMPUTE FEAT-TIME-ANOMALY =
           100 - (HOUR-FREQUENCY(WS-I) / 10).

   714-CALCULATE-MERCHANT-RISK SECTION.
       MOVE 50.00 TO FEAT-MERCHANT-RISK.

   715-CALCULATE-DEVICE-TRUST SECTION.
       SET WS-I TO 1
       SEARCH DEVICE-ID
           AT END
               MOVE 80.00 TO FEAT-DEVICE-TRUST
           WHEN DEVICE-ID(WS-I) = LS-METADATA(100:64)
               MOVE 10.00 TO FEAT-DEVICE-TRUST
       END-SEARCH.

   716-CALCULATE-VELOCITY SECTION.
       COMPUTE FEAT-VELOCITY-CHECK = FIN-DAILY-USED / FIN-DAILY-LIMIT * 100.

   720-ML-FRAUD-PREDICTION SECTION.
       STRING 
           FEAT-AMOUNT-ZSCORE ','
           FEAT-FREQUENCY-ZSCORE ','
           FEAT-LOCATION-DISTANCE ','
           FEAT-TIME-ANOMALY ','
           FEAT-MERCHANT-RISK ','
           FEAT-DEVICE-TRUST ','
           FEAT-VELOCITY-CHECK
           DELIMITED BY SIZE
           INTO ML-FEATURES
       
       CALL 'ML-FRAUD-MODEL' USING WS-ML-FRAUD-REQUEST
       
       IF ML-STATUS = '00'
           MOVE ML-PREDICTION-SCORE TO WS-FRAUD-SCORE-CALCULATED
       END-IF.

   730-EVALUATE-FRAUD-DECISION SECTION.
       EVALUATE TRUE
           WHEN WS-FRAUD-SCORE-CALCULATED < 25.00
               SET FRAUD-ALLOW TO TRUE
           WHEN WS-FRAUD-SCORE-CALCULATED < 75.00
               SET FRAUD-CHALLENGE TO TRUE
               PERFORM 731-SEND-CHALLENGE-REQUEST
           WHEN WS-FRAUD-SCORE-CALCULATED < 90.00
               SET FRAUD-REVIEW TO TRUE
               PERFORM 732-QUEUE-FOR-MANUAL-REVIEW
           WHEN OTHER
               SET FRAUD-BLOCK TO TRUE
               PERFORM 733-BLOCK-TRANSACTION
       END-EVALUATE.

   731-SEND-CHALLENGE-REQUEST SECTION.
       CALL 'SEND-2FA-CHALLENGE' USING
           LS-USER-ID
           SEC-2FA-SECRET
           WS-CHALLENGE-RESPONSE
           HSM-STATUS.

   732-QUEUE-FOR-MANUAL-REVIEW SECTION.
       MOVE 'R' TO TRANS-STATUS
       REWRITE TRANSACTION-LOG-RECORD.

   733-BLOCK-TRANSACTION SECTION.
       MOVE 'FRAUD_DETECTED' TO WS-ERROR-CODE
       MOVE 'Transaction blocked due to fraud detection' 
            TO WS-ERROR-MESSAGE
       PERFORM 9300-SEND-SECURITY-ALERT.

   800-COMPLIANCE-VERIFICATION SECTION.
       PERFORM 810-VERIFY-KYC-STATUS
       PERFORM 820-CHECK-AML-REQUIREMENTS
       PERFORM 830-CHECK-SANCTIONS-LIST
       PERFORM 840-VERIFY-PEP-STATUS
       PERFORM 850-CHECK-TAX-REPORTING.

   810-VERIFY-KYC-STATUS SECTION.
       IF NOT KYC-COMPLETE
           MOVE 'KYC_REQUIRED' TO WS-ERROR-CODE
           MOVE 'KYC verification required' TO WS-ERROR-MESSAGE
       END-IF.

   820-CHECK-AML-REQUIREMENTS SECTION.
       IF LS-AMOUNT > 10000.00
           PERFORM 821-RUN-AML-CHECK
       END-IF.

   821-RUN-AML-CHECK SECTION.
       CALL 'AML-SCREENING-SERVICE' USING
           ACCT-CUSTOMER-INFO
           LS-AMOUNT
           WS-COMPLIANCE-CHECKS
           HSM-STATUS
       
       IF WS-AML-RESULT = 'F'
           MOVE 'AML_FLAGGED' TO WS-ERROR-CODE
           PERFORM 9100-LOG-SECURITY-EVENT
       END-IF.

   830-CHECK-SANCTIONS-LIST SECTION.
       CALL 'SANCTIONS-SCREENING' USING
           CUST-FIRST-NAME
           CUST-LAST-NAME
           CUST-DATE-OF-BIRTH
           WS-SANCTIONS-RESULT
           HSM-STATUS
       
       IF WS-SANCTIONS-RESULT = 'M'
           MOVE 'SANCTIONS_MATCH' TO WS-ERROR-CODE
           MOVE 'Account matched sanctions list' TO WS-ERROR-MESSAGE
           PERFORM 9000-LOG-CRITICAL-ERROR
       END-IF.

   840-VERIFY-PEP-STATUS SECTION.
       IF PEP-FOREIGN OR PEP-INTERNATIONAL
           IF LS-AMOUNT > 50000.00
               PERFORM 841-ENHANCED-DUE-DILIGENCE
           END-IF
       END-IF.

   841-ENHANCED-DUE-DILIGENCE SECTION.
       CALL 'EDD-PROCESS' USING
           ACCT-CUSTOMER-INFO
           LS-TRANSACTION-REQUEST
           WS-COMPLIANCE-CHECKS
           HSM-STATUS.

   850-CHECK-TAX-REPORTING SECTION.
       IF LS-AMOUNT > 10000.00
           PERFORM 851-GENERATE-TAX-REPORT
       END-IF.

   851-GENERATE-TAX-REPORT SECTION.
       CALL 'TAX-REPORTING-SERVICE' USING
           ACCT-KEY-PRIMARY
           LS-AMOUNT
           WS-CURRENT-TIMESTAMP
           HSM-STATUS.

   900-COMMIT-DISTRIBUTED-TRANSACTION SECTION.
       PERFORM 910-INITIALIZE-2PC-COORDINATOR
       PERFORM 920-PREPARE-PHASE
       IF WS-ERROR-CODE = SPACES
           PERFORM 930-COMMIT-PHASE
       ELSE
           PERFORM 940-ABORT-PHASE
       END-IF.

   910-INITIALIZE-2PC-COORDINATOR SECTION.
       CALL 'UUID-GENERATE-V4' USING WS-TRANSACTION-COORDINATOR
       MOVE 'INITIALIZED' TO WS-TRANSACTION-STATE
       MOVE 3 TO WS-PARTICIPANT-COUNT
       
       MOVE 'ACCOUNT-SERVICE' TO PARTICIPANT-ID(1)
       MOVE 'LEDGER-SERVICE' TO PARTICIPANT-ID(2)
       MOVE 'AUDIT-SERVICE' TO PARTICIPANT-ID(3).

   920-PREPARE-PHASE SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 
               UNTIL WS-I > WS-PARTICIPANT-COUNT
           CALL 'PREPARE-TRANSACTION' USING
               PARTICIPANT-ID(WS-I)
               WS-TRANSACTION-COORDINATOR
               LS-TRANSACTION-REQUEST
               PARTICIPANT-STATUS(WS-I)
               HSM-STATUS
           
           IF PARTICIPANT-STATUS(WS-I) NOT = 'READY'
               MOVE 'PREPARE_FAILED' TO WS-ERROR-CODE
               EXIT PERFORM
           END-IF
       END-PERFORM
       
       IF WS-ERROR-CODE = SPACES
           MOVE 'PREPARED' TO WS-TRANSACTION-STATE
       END-IF.

   930-COMMIT-PHASE SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 
               UNTIL WS-I > WS-PARTICIPANT-COUNT
           CALL 'COMMIT-TRANSACTION' USING
               PARTICIPANT-ID(WS-I)
               WS-TRANSACTION-COORDINATOR
               HSM-STATUS
           
           MOVE 'COMMITTED' TO PARTICIPANT-STATUS(WS-I)
       END-PERFORM
       
       MOVE 'COMMITTED' TO WS-TRANSACTION-STATE
       MOVE 0 TO LS-STATUS-CODE
       MOVE 'Transaction successful' TO LS-STATUS-MESSAGE
       MOVE WS-TRANS-ID TO LS-TRANSACTION-ID.

   940-ABORT-PHASE SECTION.
       PERFORM VARYING WS-I FROM 1 BY 1 
               UNTIL WS-I > WS-PARTICIPANT-COUNT
           CALL 'ABORT-TRANSACTION' USING
               PARTICIPANT-ID(WS-I)
               WS-TRANSACTION-COORDINATOR
               HSM-STATUS
           
           MOVE 'ABORTED' TO PARTICIPANT-STATUS(WS-I)
       END-PERFORM
       
       MOVE 'ABORTED' TO WS-TRANSACTION-STATE.

   950-ROLLBACK-TRANSACTION SECTION.
       PERFORM 951-REVERSE-ACCOUNT-CHANGES
       PERFORM 952-LOG-ROLLBACK.

   951-REVERSE-ACCOUNT-CHANGES SECTION.
       ADD WS-TRANS-TOTAL TO FIN-AVAILABLE-BALANCE
       REWRITE ACCOUNT-MASTER-RECORD.

   952-LOG-ROLLBACK SECTION.
       MOVE 'REV' TO TRANS-TYPE
       MOVE 'X' TO TRANS-STATUS
       WRITE TRANSACTION-LOG-RECORD.

   990-FINALIZE-AND-RESPOND SECTION.
       IF WS-ERROR-CODE NOT = SPACES
           MOVE 4000 TO LS-STATUS-CODE
           MOVE WS-ERROR-MESSAGE TO LS-STATUS-MESSAGE
       ELSE
           MOVE FIN-AVAILABLE-BALANCE TO LS-BALANCE-AFTER
       END-IF
       
       PERFORM 991-CLOSE-FILES.

   991-CLOSE-FILES SECTION.
       CLOSE MASTER-ACCOUNT-FILE
       CLOSE TRANSACTION-LOG-FILE
       CLOSE ENCRYPTION-KEY-FILE
       CLOSE BIOMETRIC-DATA-FILE
       CLOSE FRAUD-DETECTION-CONFIG
       CLOSE AUDIT-TRAIL-FILE
       CLOSE GEO-LOCATION-DATABASE
       CLOSE REGULATORY-RULES-FILE.

   999-AUDIT-LOG-TRANSACTION SECTION.
       CALL 'UUID-GENERATE-V4' USING AUDIT-ID
       MOVE FUNCTION CURRENT-DATE TO AUDIT-TIMESTAMP
       
       STRING 'TRANSACTION: ' 
              LS-TRANSACTION-TYPE ' '
              LS-SOURCE-ACCOUNT ' -> '
              LS-TARGET-ACCOUNT ' '
              LS-AMOUNT
              DELIMITED BY SIZE
              INTO AUDIT-ACTION
       
       MOVE LS-USER-ID TO AUDIT-USER-ID
       MOVE LS-METADATA(1:45) TO AUDIT-IP-ADDRESS
       MOVE WS-ERROR-MESSAGE TO AUDIT-DETAILS
       
       PERFORM 9991-CALCULATE-BLOCKCHAIN-HASH
       
       WRITE AUDIT-TRAIL-RECORD.

   9000-LOG-CRITICAL-ERROR SECTION.
       MOVE 'CRITICAL' TO WS-ERROR-SEVERITY
       PERFORM 9999-LOG-ERROR.

   9100-LOG-SECURITY-EVENT SECTION.
       STRING 'SECURITY EVENT: ' 
              WS-ERROR-CODE ' - '
              WS-ERROR-MESSAGE
              DELIMITED BY SIZE
              INTO AUDIT-ACTION
       
       MOVE LS-USER-ID TO AUDIT-USER-ID
       WRITE AUDIT-TRAIL-RECORD.

   9200-LOG-AUDIT-TRAIL SECTION.
       STRING 'LOGIN SUCCESS: ' 
              LS-USER-ID ' from '
              SEC-LAST-IP-ADDRESS
              DELIMITED BY SIZE
              INTO AUDIT-ACTION
       
       WRITE AUDIT-TRAIL-RECORD.

   9300-SEND-SECURITY-ALERT SECTION.
       CALL 'SECURITY-ALERT-SERVICE' USING
           LS-USER-ID
           WS-ERROR-CODE
           WS-ERROR-MESSAGE
           HSM-STATUS.

   9991-CALCULATE-BLOCKCHAIN-HASH SECTION.
       CALL 'SHA256-HASH' USING
           AUDIT-TRAIL-RECORD
           AUDIT-HASH-CURRENT
           HSM-STATUS.

   9999-LOG-ERROR SECTION.
       ADD 1 TO WS-ERROR-COUNT
       
       IF WS-ERROR-COUNT <= 20
           STRING WS-ERROR-CODE ': ' WS-ERROR-MESSAGE
                  DELIMITED BY SIZE
                  INTO ERROR-ENTRY(WS-ERROR-COUNT)
       END-IF
       
       DISPLAY 'ERROR [' WS-ERROR-SEVERITY ']: ' 
               WS-ERROR-CODE ' - ' WS-ERROR-MESSAGE.

   END PROGRAM ENTERPRISE-SECURITY-FORTRESS.
