Ce programme COBOL calcule l'impôt sur le revenu en appliquant un barème fiscal à trois tranches, dont les limites et les taux sont définis en dur et datent de 1990.

[PROBLÈMES]
-   **Logique fiscale obsolète**: Le barème fiscal (limites et taux) est fixé en dur pour 1990, rendant le programme inutilisable pour les calculs actuels sans une modification manuelle du code source.
-   **Données d'entrée non dynamiques**: Le revenu annuel est une valeur fixe (`75000`) intégrée directement dans le code, exigeant une recompilation pour chaque nouveau calcul, ce qui est très peu pratique.
-   **Manque de modularité et de flexibilité**: La logique de calcul est monolithique, rendant difficile l'adaptation à de nouveaux barèmes (par exemple, un nombre différent de tranches, des taux variables, etc.) ou la réutilisation de la fonction de calcul dans un contexte plus large.
-   **Gestion des décimales**: Bien que COBOL gère les décimales via les clauses `PIC V999` et `9(7)V99`, la migration vers un autre langage nécessite une attention particulière à la précision des calculs monétaires.

[PYTHON]
```python
# Constants for 1990 tax brackets and rates
TRANCHE1_LIMITE = 30000.00
TRANCHE2_LIMITE = 100000.00
TAUX_TRANCHE1 = 0.150  # 15%
TAUX_TRANCHE2 = 0.280  # 28%
TAUX_TRANCHE3 = 0.400  # 40%

def calculer_impot_1990(revenu_annuel: float) -> float:
    """
    Calcule l'impôt sur le revenu selon le barème fiscal de 1990.
    Args:
        revenu_annuel (float): Le revenu annuel à calculer.
    Returns:
        float: Le montant de l'impôt calculé.
    """
    impot_calcule = 0.0

    if revenu_annuel <= TRANCHE1_LIMITE:
        impot_calcule = revenu_annuel * TAUX_TRANCHE1
    elif revenu_annuel <= TRANCHE2_LIMITE:
        # Impôt sur la première tranche + impôt sur la partie excédentaire de la deuxième tranche
        impot_calcule = (TRANCHE1_LIMITE * TAUX_TRANCHE1) + \
                        ((revenu_annuel - TRANCHE1_LIMITE) * TAUX_TRANCHE2)
    else:
        # Impôt sur la première tranche + impôt sur la deuxième tranche complète + impôt sur la partie excédentaire de la troisième tranche
        impot_calcule = (TRANCHE1_LIMITE * TAUX_TRANCHE1) + \
                        ((TRANCHE2_LIMITE - TRANCHE1_LIMITE) * TAUX_TRANCHE2) + \
                        ((revenu_annuel - TRANCHE2_LIMITE) * TAUX_TRANCHE3)
    
    # Arrondir à deux décimales pour la monnaie, comme attendu dans les calculs fiscaux
    return round(impot_calcule, 2)

# Valeur d'entrée hardcodée, pour simuler le comportement du COBOL original
ws_revenu_annuel = 75000.00

# Calcul de l'impôt
ws_impot_calcule = calculer_impot_1990(ws_revenu_annuel)

# Affichage des résultats
print(f"REVENU: {ws_revenu_annuel:.2f}")
print(f"IMPOT CALCULE: {ws_impot_calcule:.2f}")

```

from fastapi import FastAPI, UploadFile, File, HTTPException
from fastapi.middleware.cors import CORSMiddleware
import google.generativeai as genai
import os
from typing import Dict, Any
import json

app = FastAPI(title="CodeSwitch API")

# CORS pour le frontend
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# Configuration Gemini
GEMINI_API_KEY = os.getenv("GEMINI_API_KEY", "TA_CLE_API")  # Mets ta clé
genai.configure(api_key=GEMINI_API_KEY)
model = genai.GenerativeModel('gemini-2.5-flash')

PROMPT_BASE = """Tu es CodeSwitch, expert migration legacy. Analyse ce code COBOL:

{code}

Retourne JSON avec: business_analysis, problems_detected, python_migration, unit_tests, warnings."""

@app.post("/analyze")
async def analyze_code(file: UploadFile = File(...)):
    try:
        # Lit le fichier
        content = (await file.read()).decode('utf-8')
        
        # Appelle Gemini
        prompt = PROMPT_BASE.format(code=content)
        response = model.generate_content(prompt)
        
        # Nettoie et parse la réponse
        text = response.text.strip()
        if "```json" in text:
            text = text.split("```json")[1].split("```")[0].strip()
        
        result = json.loads(text)
        
        return {
            "success": True,
            "filename": file.filename,
            "analysis": result
        }
        
    except Exception as e:
        return {"success": False, "error": str(e)}

@app.get("/")
def root():
    return {"status": "CodeSwitch API running"}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)

HEURE 3 : Edge Function pour Gemini (1h)
Dans Supabase Dashboard → Edge Functions → "New Function" :

Nom : analyze-legacy-code

Code (index.ts - mais en réalité Python) :

python
# /supabase/functions/analyze-legacy-code/index.py
from supabase import create_client
import google.generativeai as genai
import os
import json

# Prompt Gemini (le même qu'avant)
PROMPT = """Tu es CodeSwitch... {code}..."""

def main(req):
    # 1. Récupère le code du body
    body = req.json()
    code = body.get('code', '')
    language = body.get('language', 'cobol')
    
    # 2. Appelle Gemini
    genai.configure(api_key=os.environ['GEMINI_API_KEY'])
    model = genai.GenerativeModel('gemini-2.5-flash')
    response = model.generate_content(PROMPT.format(code=code))
    
    # 3. Parse la réponse
    result = json.loads(response.text)
    
    # 4. Stocke dans la DB (optionnel)
    supabase = create_client(
        os.environ['SUPABASE_URL'],
        os.environ['SUPABASE_SERVICE_KEY']
    )
    
    supabase.table('analyses').insert({
        'original_code': code,
        'language': language,
        'gemini_response': result,
        'python_code': result.get('python_migration', {}).get('code', '')
    }).execute()
    
    # 5. Retourne le résultat
    return {
        'status': 'success',
        'data': result
    }

typescript
'use client'

import { useState } from 'react'
import { supabase } from '@/lib/supabase/client'

export default function Home() {
  const [code, setCode] = useState('')
  const [result, setResult] = useState<any>(null)
  
  const analyzeCode = async () => {
    // Appelle l'Edge Function Supabase
    const { data, error } = await supabase.functions.invoke('analyze-legacy-code', {
      body: { code, language: 'cobol' }
    })
    
    if (!error && data) {
      setResult(data.data)
      
      // Optionnel : enregistre dans l'historique
      await supabase.from('analyses').insert({
        original_code: code,
        gemini_response: data.data
      })
    }
  }
  
  return (
    <div className="p-8">
      <h1>CodeSwitch</h1>
      <textarea value={code} onChange={e => setCode(e.target.value)} />
      <button onClick={analyzeCode}>Analyser</button>
      
      {result && (
        <div>
          <h2>Python généré :</h2>
          <pre>{result.python_migration?.code}</pre>
        </div>
      )}
    </div>
  )
}
