def calculate_tax():
    """8500-calculate_tax."""
    global WS_CALC_AMOUNT, WS_CALC_TAX, WS_BRACKET_1_MAX, WS_BRACKET_2_MAX, WS_BRACKET_3_MAX, WS_BRACKET_5_RATE, WS_BRACKET_1_RATE, WS_BRACKET_2_RATE, WS_BRACKET_3_RATE
    match True:
        case WS_CALC_AMOUNT <= WS_BRACKET_1_MAX:
            WS_CALC_TAX = WS_CALC_AMOUNT * WS_BRACKET_1_RATE
        case WS_CALC_AMOUNT <= WS_BRACKET_2_MAX:
            WS_CALC_TAX = (WS_BRACKET_1_MAX * WS_BRACKET_1_RATE) + \
                         ((WS_CALC_AMOUNT - WS_BRACKET_1_MAX) * WS_BRACKET_2_RATE)
        case WS_CALC_AMOUNT <= WS_BRACKET_3_MAX:
            WS_CALC_TAX = (WS_BRACKET_1_MAX * WS_BRACKET_1_RATE) + \
                         ((WS_BRACKET_2_MAX - WS_BRACKET_1_MAX) * WS_BRACKET_2_RATE) + \
                         ((WS_CALC_AMOUNT - WS_BRACKET_2_MAX) * WS_BRACKET_3_RATE)
        case _:
            WS_CALC_TAX = WS_CALC_AMOUNT * WS_BRACKET_5_RATE

def calculate_pmi(loan_data: LoanData) -> None:
    """

           WHEN ws_ltv_ratio > 95

                 ws_loan_amount * 0.0125 / 12
           WHEN ws_ltv_ratio > 90

                 ws_loan_amount * 0.0100 / 12
           WHEN ws_ltv_ratio > 85

                 ws_loan_amount * 0.0075 / 12
           WHEN OTHER

                 ws_loan_amount * 0.0050 / 12
        end_evaluate.
    """
    match True:
        case loan_data.ws_ltv_ratio > 95:
            loan_data.ws_pmi_amount = loan_data.ws_loan_amount * Decimal('0.0125') / 12
        case loan_data.ws_ltv_ratio > 90:
            loan_data.ws_pmi_amount = loan_data.ws_loan_amount * Decimal('0.0100') / 12

            loan_data.ws_pmi_amount = loan_data.ws_loan_amount * Decimal('0.0075') / 12
        case _:
            loan_data.ws_pmi_amount = loan_data.ws_loan_amount * Decimal('0.0050') / 12

def generate_loan_terms(loan_data: LoanData, records: Records, term_months: int) -> None:
    """

           ws_loan_interest_rate / 1200

           (1 + ws_monthly_rate) ** ws_loan_term_months

           ws_loan_amount * ws_monthly_rate * 
           ws_compound_factor / (ws_compound_factor - 1)

    """
    loan_data.ws_loan_interest_rate = loan_data.ws_approved_rate
    loan_data.ws_monthly_rate = loan_data.ws_loan_interest_rate / 1200
    loan_data.ws_compound_factor = (1 + loan_data.ws_monthly_rate) ** term_months
    loan_data.ws_loan_monthly_pmt = (loan_data.ws_loan_amount * loan_data.ws_monthly_rate *
                                     loan_data.ws_compound_factor / (loan_data.ws_compound_factor - 1))
    loan_data.ws_loan_principal_bal = loan_data.ws_loan_amount

def calculate_payment_split(loan_data: LoanData) -> None:
    """

           ws_running_balance * ws_monthly_rate

           ws_loan_monthly_pmt - amort_interest(ws_amort_idx)

           FROM ws_running_balance

           TO amort_balance(ws_amort_idx)

           TO amort_payment_amt(ws_amort_idx)

              (ws_property_tax + ws_insurance_premium) / 12

              ws_loan_monthly_pmt + 
              amort_escrow(ws_amort_idx) + ws_pmi_amount
        ELSE

              TO amort_total_pmt(ws_amort_idx)
        end_if

    """
    loan_data.amort_interest[loan_data.ws_amort_idx - 1] = loan_data.ws_running_balance * loan_data.ws_monthly_rate
    loan_data.amort_principal[loan_data.ws_amort_idx - 1] = loan_data.ws_loan_monthly_pmt - loan_data.amort_interest[loan_data.ws_amort_idx - 1]
    loan_data.ws_running_balance -= loan_data.amort_principal[loan_data.ws_amort_idx - 1]
    loan_data.amort_balance[loan_data.ws_amort_idx - 1] = loan_data.ws_running_balance
    loan_data.amort_payment_num[loan_data.ws_amort_idx - 1] = loan_data.ws_amort_idx
    loan_data.amort_payment_amt[loan_data.ws_amort_idx - 1] = loan_data.ws_loan_monthly_pmt

        loan_data.amort_escrow[loan_data.ws_amort_idx - 1] = (loan_data.ws_property_tax + loan_data.ws_insurance_premium) / 12
        loan_data.amort_total_pmt[loan_data.ws_amort_idx - 1] = (loan_data.ws_loan_monthly_pmt +
                                                                 loan_data.amort_escrow[loan_data.ws_amort_idx - 1] + loan_data.ws_pmi_amount)
    else:
        loan_data.amort_total_pmt[loan_data.ws_amort_idx - 1] = loan_data.ws_loan_monthly_pmt

    advance_payment_date(loan_data)
