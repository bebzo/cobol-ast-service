def calculate_tax_C2() -> None:
    """8500-calculate_tax."""
    global WS_CALC_AMOUNT, WS_BRACKET_1_MAX, WS_BRACKET_1_RATE, WS_CALC_TAX, WS_BRACKET_2_MAX, WS_BRACKET_2_RATE, WS_BRACKET_3_MAX, WS_BRACKET_3_RATE, WS_BRACKET_5_RATE
    if WS_CALC_AMOUNT <= WS_BRACKET_1_MAX:
        WS_CALC_TAX = WS_CALC_AMOUNT * WS_BRACKET_1_RATE
    elif WS_CALC_AMOUNT <= WS_BRACKET_2_MAX:
        WS_CALC_TAX = (WS_BRACKET_1_MAX * WS_BRACKET_1_RATE) + ((WS_CALC_AMOUNT - WS_BRACKET_1_MAX) * WS_BRACKET_2_RATE)
    elif WS_CALC_AMOUNT <= WS_BRACKET_3_MAX:
        WS_CALC_TAX = (WS_BRACKET_1_MAX * WS_BRACKET_1_RATE) + ((WS_BRACKET_2_MAX - WS_BRACKET_1_MAX) * WS_BRACKET_2_RATE) + ((WS_CALC_AMOUNT - WS_BRACKET_2_MAX) * WS_BRACKET_3_RATE)
    else:
        WS_CALC_TAX = WS_CALC_AMOUNT * WS_BRACKET_5_RATE

def calculate_payment_split_C6(ws_running_balance: Decimal, ws_monthly_rate: Decimal, ws_loan_monthly_pmt: Decimal, ws_property_tax: Decimal, ws_insurance_premium: Decimal, ws_pmi_amount: Decimal, loan_mortgage: bool, ws_amort_idx: int, ws_payment_month: int, ws_payment_year: int) -> tuple[Decimal, Decimal, Decimal, Decimal, int, Decimal, Decimal, Decimal]:
    """10650-calculate_payment_split."""
    amort_interest: Decimal = ws_running_balance * ws_monthly_rate
    amort_principal: Decimal = ws_loan_monthly_pmt - amort_interest
    ws_running_balance -= amort_principal
    amort_balance: Decimal = ws_running_balance
    amort_payment_num: int = ws_amort_idx
    amort_payment_amt: Decimal = ws_loan_monthly_pmt

    if loan_mortgage:
        amort_escrow: Decimal = (ws_property_tax + ws_insurance_premium) / 12
        amort_total_pmt: Decimal = ws_loan_monthly_pmt + amort_escrow + ws_pmi_amount
    else:
        amort_escrow: Decimal = Decimal(0) #Not used when loan_mortgage is false
        amort_total_pmt: Decimal = ws_loan_monthly_pmt
    
    return amort_interest, amort_principal, ws_running_balance, amort_balance, amort_payment_num, amort_payment_amt, amort_escrow, amort_total_pmt

    """11410-calculate_current_allocation."""
    ws_stocks_value: Decimal = Decimal(0)
    ws_bonds_value: Decimal = Decimal(0)
    ws_cash_value: Decimal = Decimal(0)

    for ws_hold_idx in range(1, ws_holdings_count + 1):
        hold_type: str = get_hold_type_C6(ws_holding[ws_hold_idx])
        hold_market_value: Decimal = get_hold_market_value_C6(ws_holding[ws_hold_idx])
        match hold_type:

            case 'BND':
                ws_bonds_value += hold_market_value
            case 'CSH':
                ws_cash_value += hold_market_value

    ws_stocks_pct: Decimal = (ws_stocks_value / ws_total_value) * 100
    ws_bonds_pct: Decimal = (ws_bonds_value / ws_total_value) * 100
    ws_cash_pct: Decimal = (ws_cash_value / ws_total_value) * 100
    
    return ws_stocks_pct, ws_bonds_pct, ws_cash_pct

def calculate_stress_impact_C12() -> None:
    """34450-calculate_stress_impact."""
    global WS_CREDIT_LOSSES, WS_MARKET_LOSSES, WS_STRESS_LOSSES, WS_STRESSED_CAPITAL, WS_STRESSED_RATIO, WS_STRESS_PASS_FAIL
    WS_CREDIT_LOSSES = WS_LOAN_PORTFOLIO * WS_STRESS_LGD * WS_STRESS_PD  # COBOL: COMPUTE ws_credit_losses = ws_loan_portfolio * ws_stress_lgd * ws_stress_pd
    WS_MARKET_LOSSES = WS_TRADING_ASSETS * WS_RATE_SHOCK / 100  # COBOL: COMPUTE ws_market_losses = ws_trading_assets * ws_rate_shock / 100
    WS_STRESS_LOSSES = WS_CREDIT_LOSSES + WS_MARKET_LOSSES  # COBOL: COMPUTE ws_stress_losses = ws_credit_losses + ws_market_losses
    WS_STRESSED_CAPITAL = WS_TOTAL_CAPITAL - WS_STRESS_LOSSES  # COBOL: COMPUTE ws_stressed_capital = ws_total_capital - ws_stress_losses
    WS_STRESSED_RATIO = (WS_STRESSED_CAPITAL / WS_RISK_WEIGHTED_ASSETS) * 100  # COBOL: COMPUTE ws_stressed_ratio = (ws_stressed_capital / ws_risk_weighted_assets) * 100
    if WS_STRESSED_RATIO >= WS_MIN_CAPITAL_RATIO:  # COBOL: IF ws_stressed_ratio >= ws_min_capital_ratio
        WS_STRESS_PASS_FAIL = 'PASS'  # COBOL: MOVE 'PASS' TO ws_stress_pass_fail
    else:  # COBOL: ELSE
        WS_STRESS_PASS_FAIL = 'FAIL'  # COBOL: MOVE 'FAIL' TO ws_stress_pass_fail
    # COBOL: end_if.

def reencrypt_data_C13() -> None:
    """41215-reencrypt_data."""
    global WS_EOF_FLAG, WS_ENC_RECORD, ENC_DATA, WS_DECRYPTED_DATA, WS_REENCRYPTED_DATA
    WS_EOF_FLAG = 'N'
    while WS_EOF_FLAG != 'Y':
        try:
            WS_ENC_RECORD = read_encrypted_data_file()
            ENC_DATA = WS_ENC_RECORD.ENC_DATA
            WS_DECRYPTED_DATA = aes256dec(ENC_DATA, WS_OLD_KEY, WS_DECRYPTED_DATA)
            WS_REENCRYPTED_DATA = aes256enc(WS_DECRYPTED_DATA, WS_ENCRYPTION_KEY, WS_REENCRYPTED_DATA)
            ENC_DATA = WS_REENCRYPTED_DATA
            WS_ENC_RECORD = WsEncRecord(ENC_DATA=ENC_DATA)
            rewrite_encrypted_data_record(WS_ENC_RECORD)
        except EOFError:
            WS_EOF_FLAG = 'Y'
    WS_EOF_FLAG = 'N'
