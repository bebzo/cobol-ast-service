class IndexedFileManager(FileManager):
    """Gestionnaire pour fichiers INDEXED COBOL"""
    
    def __init__(self, indexed_files: Dict[str, Dict]):
        super().__init__()
        self.indexed_files = indexed_files  # {file: {key_field: index}}
    
    def read_by_key(self, file_name: str, key_value: Any) -> Optional[str]:
        """READ file KEY IS key_value"""
        if file_name not in self.indexed_files:
            return self.read_record(file_name)
        
        # Implémentation réelle avec index
        index = self.indexed_files[file_name]
        if key_value in index:
            return index[key_value]
        return None  # INVALID KEY
