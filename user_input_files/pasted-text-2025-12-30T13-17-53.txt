    def calculate_tax(self):
        """8500-CALCULATE-TAX.

               WHEN WS-CALC-AMOUNT <= WS-BRACKET-1-MAX

                       WS-CALC-AMOUNT * WS-BRACKET-1-RATE
               WHEN WS-CALC-AMOUNT <= WS-BRACKET-2-MAX

                       (WS-BRACKET-1-MAX * WS-BRACKET-1-RATE) +
                       ((WS-CALC-AMOUNT - WS-BRACKET-1-MAX) * 
                        WS-BRACKET-2-RATE)
               WHEN WS-CALC-AMOUNT <= WS-BRACKET-3-MAX

                       (WS-BRACKET-1-MAX * WS-BRACKET-1-RATE) +
                       ((WS-BRACKET-2-MAX - WS-BRACKET-1-MAX) * 
                        WS-BRACKET-2-RATE) +
                       ((WS-CALC-AMOUNT - WS-BRACKET-2-MAX) * 
                        WS-BRACKET-3-RATE)
               WHEN OTHER

                       WS-CALC-AMOUNT * WS-BRACKET-5-RATE
           END-EVALUATE.
        """
        match True:
            case self.WS_CALC_AMOUNT <= self.WS_BRACKET_1_MAX:
                self.WS_CALC_TAX = self.WS_CALC_AMOUNT * self.WS_BRACKET_1_RATE
            case self.WS_CALC_AMOUNT <= self.WS_BRACKET_2_MAX:
                self.WS_CALC_TAX = (self.WS_BRACKET_1_MAX * self.WS_BRACKET_1_RATE) + ((self.WS_CALC_AMOUNT - self.WS_BRACKET_1_MAX) * self.WS_BRACKET_2_RATE)
            case self.WS_CALC_AMOUNT <= self.WS_BRACKET_3_MAX:
                self.WS_CALC_TAX = (self.WS_BRACKET_1_MAX * self.WS_BRACKET_1_RATE) + ((self.WS_BRACKET_2_MAX - self.WS_BRACKET_1_MAX) * self.WS_BRACKET_2_RATE) + ((self.WS_CALC_AMOUNT - self.WS_BRACKET_2_MAX) * self.WS_BRACKET_3_RATE)
            case _:
                self.WS_CALC_TAX = self.WS_CALC_AMOUNT * self.WS_BRACKET_5_RATE

def calculate_simple_interest():
    """
    7200-CALCULATE-SIMPLE-INTEREST.

              WS-ACCOUNT-BALANCE * WS-INTEREST-RATE 
              * WS-DAYS-IN-PERIOD / 36500.
    """

        ws_account_balance * ws_interest_rate * ws_days_in_period / Decimal('36500')
    )

def calculate_compound_interest():
    """
    7300-CALCULATE-COMPOUND-INTEREST.

              (1 + WS-INTEREST-RATE / 36500) 
              ** WS-DAYS-IN-PERIOD

              WS-ACCOUNT-BALANCE * (WS-COMPOUND-FACTOR - 1).
    """
    global ws_compound_factor, ws_compound_interest
    ws_compound_factor = (
        1 + ws_interest_rate / Decimal('36500')
    ) ** ws_days_in_period
    ws_compound_interest = (
        ws_account_balance * (ws_compound_factor - 1)
    )
