# ------------------------------------------------------------------
# 3. Layout ACH (94 bytes)
# ------------------------------------------------------------------
@dataclass
class AchRecord:
    """ACH record layout (94 bytes) - Auto-generated."""
    routing_number: str = ""      # pos 1-8
    account_number: str = ""      # pos 9-26
    amount: Decimal = Decimal("0")  # pos 27-37 (implied 2 decimals)
    transaction_code: str = ""    # pos 38-39
    company_name: str = ""        # pos 40-55

    @classmethod
    def from_line(cls, line: str) -> "AchRecord":
        if len(line) != 94:
            raise ValueError("ACH record must be exactly 94 bytes")
        return cls(
            routing_number=line[0:8].strip(),
            account_number=line[8:26].strip(),
            amount=Decimal(line[26:37]) / 100,
            transaction_code=line[37:39],
            company_name=line[39:55].strip(),
        )
