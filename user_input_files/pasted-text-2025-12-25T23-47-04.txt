def calculer_impot(revenu: Revenu, tax_brackets: dict = TAX_BRACKETS_1990) -> Impot:
    """Calcule l'impôt sur le revenu en utilisant les tranches d'imposition de 1990."""
    warnings.warn("Attention : Utilisation des taux d'imposition de 1990. Les résultats ne sont pas applicables à l'année en cours.", DeprecationWarning)

    revenu_annuel = revenu.montant
    tranche1_limit = tax_brackets["tranche1_limit"]
    tranche2_limit = tax_brackets["tranche2_limit"]
    taux_tranche1 = tax_brackets["taux_tranche1"]
    taux_tranche2 = tax_brackets["taux_tranche2"]
    taux_tranche3 = tax_brackets["taux_tranche3"]

    if revenu_annuel <= tranche1_limit:
        impot_calcule = revenu_annuel * taux_tranche1
    elif revenu_annuel <= tranche2_limit:
        impot_calcule = tranche1_limit * taux_tranche1 + (revenu_annuel - tranche1_limit) * taux_tranche2
    else:
        impot_calcule = (
            tranche1_limit * taux_tranche1
            + (tranche2_limit - tranche1_limit) * taux_tranche2
            + (revenu_annuel - tranche2_limit) * taux_tranche3
        )

    return Impot(montant=impot_calcule.quantize(Decimal("0.00")))

@dataclass
class ConfigFiscale:
    annee: int
    tranches: List[dict]
    derniere_mise_a_jour: str
    
    @classmethod
    def charger(cls, fichier: str = "config_fiscale.json"):
        with open(fichier, 'r') as f:
            data = json.load(f)
        return cls(**data)

import logging
import csv
from datetime import datetime

class AuditFiscal:
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.fichier_audit = "audit_calculs.csv"
        
    def enregistrer_calcul(self, revenu: Decimal, impot: Decimal, 
                          annee_fiscale: int, utilisateur: str):
        with open(self.fichier_audit, 'a', newline='') as f:
            writer = csv.writer(f)
            writer.writerow([
                datetime.now().isoformat(),
                utilisateur,
                str(revenu),
                str(impot),
                annee_fiscale
            ])
        self.logger.info(f"Calcul enregistré: {revenu} -> {impot}")
