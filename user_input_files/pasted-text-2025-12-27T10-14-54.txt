async def analyze_modules(cobol_code: str):
    prompt = """
    Analyze this COBOL and identify 3-6 logical modules.
    For each module provide:
    - Name
    - Line range
    - Responsibilities
    - Dependencies
    
    Output JSON format.
    """
    
    response = await gemini.generate(prompt + cobol_code)
    return json.loads(response.text)

async def generate_modular_python(analysis, cobol_code):
    modules = {}
    
    for module in analysis['modules']:
        module_cobol = extract_lines(cobol_code, module['line_range'])
        
        prompt = f"""
        Convert to Python class: {module['name']}
        Responsibilities: {module['responsibilities']}
        Dependencies: {module['dependencies']}
        
        Use dependency injection.
        
        Code: {module_cobol}
        """
        
        modules[module['name']] = await gemini.generate(prompt)
    
    return modules
