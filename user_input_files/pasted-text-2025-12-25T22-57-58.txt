"""

from decimal import Decimal, ROUND_HALF_UP
from dataclasses import dataclass
from typing import List
import warnings

# ============================================================================
# DÉFINITION DES STRUCTURES DE DONNÉES
# ============================================================================

@dataclass(frozen=True)  # Immutable pour la sécurité
class TaxBracket:
    """
    Représente une tranche du barème fiscal progressif.
    Une tranche est définie par un seuil supérieur et un taux d'imposition.
    """
    upper_limit: Decimal
    rate: Decimal

    def __post_init__(self):
        """Validation des données après création."""
        if self.upper_limit < Decimal('0'):
            raise ValueError("Le seuil de la tranche ne peut être négatif.")
        if not (Decimal('0') <= self.rate <= Decimal('1')):
            raise ValueError("Le taux doit être compris entre 0 et 1 (0% et 100%).")


@dataclass
class TaxCalculationResult:
    """
    Conteneur pour le résultat détaillé d'un calcul d'impôt.
    """
    annual_income: Decimal
    total_tax: Decimal
    marginal_rate: Decimal  # Taux marginal d'imposition
    average_rate: Decimal   # Taux moyen d'imposition
    calculation_year: int = 1990

    def __str__(self) -> str:
        return (f"Revenu : {self.annual_income:,.2f} € | "
                f"Impôt {self.calculation_year} : {self.total_tax:,.2f} € | "
                f"Taux moyen : {self.average_rate:.1%}")

# BARÈME FISCAL HISTORIQUE (1990) - À NE PAS UTILISER EN PRODUCTION
# ============================================================================

# 🔴 ATTENTION : Ces valeurs sont fournies à titre historique.
# Source : Loi de finances pour 1990 - Article 197 du Code Général des Impôts
_TAX_BRACKETS_1990: List[TaxBracket] = [
    TaxBracket(Decimal('30000'), Decimal('0.150')),   # Tranche 1 : jusqu'à 30 000 FRF

    TaxBracket(Decimal('1_000_000'), Decimal('0.400')),  # Tranche 3+ : taux par défaut

def calculate_income_tax_1990(
    annual_income: Decimal,
    brackets: List[TaxBracket] = _TAX_BRACKETS_1990,
    verbose: bool = False
) -> TaxCalculationResult:
    """
    Calcule l'impôt sur le revenu selon un barème progressif historique (1990).

    PRINCIPLE : L'impôt est calculé par tranches.
    Le revenu est divisé selon les seuils, chaque partie étant imposée au taux de sa tranche.

    Args:
        annual_income: Revenu annuel brut. Doit être positif ou nul.
        brackets: Liste des tranches fiscales, ordonnées par seuil croissant.
        verbose: Si True, affiche les étapes détaillées du calcul.

    Returns:
        Un objet TaxCalculationResult contenant le résultat détaillé.

    Raises:
        ValueError: Si `annual_income` est négatif ou si `brackets` est vide/mal formé.
        RuntimeWarning: Avertissement sur l'obsolescence des données.

    Example:
        >>> result = calculate_income_tax_1990(Decimal('75000'))
        >>> print(f"Impôt dû : {result.total_tax:.2f} €")
        Impôt dû : 17100.00 €
    """
    # --- VALIDATION ET AVERTISSEMENTS ---
    if annual_income < Decimal('0'):
        raise ValueError(f"Le revenu annuel ne peut être négatif. Reçu : {annual_income}")

    # Avertissement sur l'obsolescence des données
    warning_msg = (
        "Ce calcul utilise le barème fiscal français de 1990, qui est OBSOLÈTE. "
        "Pour un calcul conforme à la législation actuelle, consultez impots.gouv.fr "
        "ou utilisez un module fiscal à jour."
    )
    warnings.warn(warning_msg, DeprecationWarning, stacklevel=2)
    if verbose:
        print(f"⚠️  {warning_msg}\n")

    # --- INITIALISATION ---
    total_tax = Decimal('0')
    remaining_income = annual_income
    previous_limit = Decimal('0')

    if verbose:
        print(f"Calcul de l'impôt sur {annual_income:,.2f} € (barème 1990)")
        print("-" * 50)

    # --- CALCUL PAR TRANCHES ---
    for i, bracket in enumerate(brackets, start=1):
        if remaining_income <= Decimal('0'):
            break  # Le revenu est entièrement imposé

        # Montant imposable dans cette tranche
        income_in_this_bracket = min(
            (bracket.upper_limit - previous_limit) if bracket.upper_limit > previous_limit else remaining_income,
            remaining_income
        )

        if income_in_this_bracket > Decimal('0'):
            tax_for_bracket = income_in_this_bracket * bracket.rate
            total_tax += tax_for_bracket

            if verbose:
                print(f"Tranche {i} (≤ {bracket.upper_limit:,.0f} € à {bracket.rate:.1%}) :")
                print(f"  • Part imposable : {income_in_this_bracket:,.2f} €")
                print(f"  • Impôt tranche : {tax_for_bracket:,.2f} €")
                print(f"  • Cumul impôt : {total_tax:,.2f} €\n")

        remaining_income -= income_in_this_bracket
        previous_limit = bracket.upper_limit

    # --- PRÉPARATION DU RÉSULTAT ---
    # Arrondi à l'euro près (méthode bancaire)
    total_tax_rounded = total_tax.quantize(Decimal('1'), rounding=ROUND_HALF_UP)

    # Taux marginal (taux de la dernière tranche utilisée)
    marginal_rate = _TAX_BRACKETS_1990[-1].rate  # Par défaut, la dernière tranche
    for bracket in _TAX_BRACKETS_1990:
        if annual_income <= bracket.upper_limit:
            marginal_rate = bracket.rate
            break

    # Taux moyen d'imposition
    average_rate = (total_tax_rounded / annual_income) if annual_income > Decimal('0') else Decimal('0')

    result = TaxCalculationResult(
        annual_income=annual_income,
        total_tax=total_tax_rounded,
        marginal_rate=marginal_rate,
        average_rate=average_rate
    )

    if verbose:
        print("=" * 50)
        print(f"RÉSULTAT : {result}")
        print("=" * 50)

    return result

class CalculateurImpot:
    """Calculateur d'impôt sur le revenu avec système de tranches"""
    
    def __init__(self, config: ConfigurationFiscale = None):
        self.config = config or ConfigurationFiscale()
    
    def calculer_tranche1_seule(self, revenu: float) -> Dict[str, float]:
        """Calcul pour revenu dans la première tranche uniquement"""
        impot_tranche1 = revenu * self.config.taux_tranche1
        return {
            'impot_tranche1': impot_tranche1,
            'impot_tranche2': 0.0,
            'impot_tranche3': 0.0,
            'impot_total': impot_tranche1
        }
    
    def calculer_jusqu_a_tranche2(self, revenu: float) -> Dict[str, float]:
        """Calcul pour revenu dans les tranches 1 et 2"""
        # Tranche 1: revenu jusqu'à la limite
        impot_tranche1 = self.config.tranche1_limite * self.config.taux_tranche1
        
        # Tranche 2: revenu au-delà de la tranche 1
        revenu_imposable_t2 = revenu - self.config.tranche1_limite
        impot_tranche2 = revenu_imposable_t2 * self.config.taux_tranche2
        
        return {
            'impot_tranche1': impot_tranche1,
            'impot_tranche2': impot_tranche2,
            'impot_tranche3': 0.0,
            'impot_total': impot_tranche1 + impot_tranche2
        }
    
    def calculer_toutes_tranches(self, revenu: float) -> Dict[str, float]:
        """Calcul pour revenu dans les trois tranches"""
        # Tranche 1
        impot_tranche1 = self.config.tranche1_limite * self.config.taux_tranche1
        
        # Tranche 2
        revenu_imposable_t2 = (self.config.tranche2_limite - 
                               self.config.tranche1_limite)
        impot_tranche2 = revenu_imposable_t2 * self.config.taux_tranche2
        
        # Tranche 3
        revenu_imposable_t3 = revenu - self.config.tranche2_limite
        impot_tranche3 = revenu_imposable_t3 * self.config.taux_tranche3
        
        return {
            'impot_tranche1': impot_tranche1,
            'impot_tranche2': impot_tranche2,
            'impot_tranche3': impot_tranche3,
            'impot_total': impot_tranche1 + impot_tranche2 + impot_tranche3
        }
    
    def calculer_impot(self, revenu_annuel: float) -> ResultatCalcul:
        """
        Calcule l'impôt sur le revenu selon le système de tranches
        
        Args:
            revenu_annuel: Revenu annuel en EUR
            
        Returns:
            ResultatCalcul avec le détail du calcul
        """
        if revenu_annuel < 0:
            raise ValueError("Le revenu ne peut pas être négatif")
        
        # Sélection de la méthode de calcul selon la tranche
        if revenu_annuel <= self.config.tranche1_limite:
            calculs = self.calculer_tranche1_seule(revenu_annuel)
        elif revenu_annuel <= self.config.tranche2_limite:
            calculs = self.calculer_jusqu_a_tranche2(revenu_annuel)
        else:
            calculs = self.calculer_toutes_tranches(revenu_annuel)
        
        # Calcul du taux effectif
        taux_effectif = (calculs['impot_total'] / revenu_annuel * 100 
                        if revenu_annuel > 0 else 0.0)
        
        return ResultatCalcul(
            revenu_annuel=revenu_annuel,
            impot_tranche1=calculs['impot_tranche1'],
            impot_tranche2=calculs['impot_tranche2'],
            impot_tranche3=calculs['impot_tranche3'],
            impot_total=calculs['impot_total'],
            taux_effectif=taux_effectif
        )

    exemples_revenus = [20000, 75000, 150000]
    
    for revenu in exemples_revenus:
        resultat = calculateur.calculer_impot(revenu)
