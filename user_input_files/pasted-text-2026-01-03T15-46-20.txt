python"""Analyseur AST pour code Python migré de COBOL."""
import ast
import json
from typing import Dict, List, Set, Any
from dataclasses import dataclass, asdict
from pathlib import Path


@dataclass
class MethodInfo:
    """Informations sur une méthode."""
    name: str
    line_start: int
    line_end: int
    args: List[str]
    returns: str
    calls: List[str]  # Méthodes appelées
    variables: Set[str]  # Variables utilisées
    complexity: int
    docstring: str
    has_errors: bool
    error_types: List[str]


@dataclass
class ClassInfo:
    """Informations sur une classe."""
    name: str
    line_start: int
    line_end: int
    methods: List[MethodInfo]
    attributes: List[str]
    base_classes: List[str]


class ASTAnalyzer(ast.NodeVisitor):
    """Analyseur AST pour extraction d'informations."""
    
    def __init__(self, source_code: str):
        """Initialize analyzer.
        
        Args:
            source_code: Code source Python
        """
        self.source_code = source_code
        self.source_lines = source_code.split('\n')
        self.classes: List[ClassInfo] = []
        self.current_class: Optional[ClassInfo] = None
        self.current_method: Optional[MethodInfo] = None
        self.errors: List[Dict[str, Any]] = []
    
    def analyze(self) -> Dict[str, Any]:
        """Analyse le code source.
        
        Returns:
            Dictionnaire d'analyse complète
        """
        try:
            tree = ast.parse(self.source_code)
            self.visit(tree)
            
            return {
                'classes': [asdict(c) for c in self.classes],
                'errors': self.errors,
                'statistics': self._compute_statistics()
            }
        except SyntaxError as e:
            self.errors.append({
                'type': 'SyntaxError',
                'line': e.lineno,
                'message': str(e),
                'text': e.text
            })
            return {'classes': [], 'errors': self.errors, 'statistics': {}}
    
    def visit_ClassDef(self, node: ast.ClassDef) -> None:
        """Visite une définition de classe."""
        class_info = ClassInfo(
            name=node.name,
            line_start=node.lineno,
            line_end=node.end_lineno or node.lineno,
            methods=[],
            attributes=self._extract_attributes(node),
            base_classes=[base.id if isinstance(base, ast.Name) else str(base) 
                         for base in node.bases]
        )
        
        self.current_class = class_info
        self.generic_visit(node)
        self.classes.append(class_info)
        self.current_class = None
    
    def visit_FunctionDef(self, node: ast.FunctionDef) -> None:
        """Visite une définition de fonction."""
        if self.current_class is None:
            return
        
        method_info = MethodInfo(
            name=node.name,
            line_start=node.lineno,
            line_end=node.end_lineno or node.lineno,
            args=[arg.arg for arg in node.args.args],
            returns=self._extract_return_type(node),
            calls=self._extract_calls(node),
            variables=self._extract_variables(node),
            complexity=self._calculate_complexity(node),
            docstring=ast.get_docstring(node) or "",
            has_errors=self._detect_errors(node),
            error_types=self._detect_error_types(node)
        )
        
        self.current_class.methods.append(method_info)
        self.current_method = method_info
        self.generic_visit(node)
        self.current_method = None
    
    def _extract_attributes(self, node: ast.ClassDef) -> List[str]:
        """Extrait les attributs d'une classe."""
        attributes = set()
        
        for item in ast.walk(node):
            if isinstance(item, ast.Assign):
                for target in item.targets:
                    if isinstance(target, ast.Attribute):
                        if isinstance(target.value, ast.Name) and target.value.id == 'self':
                            attributes.add(target.attr)
        
        return sorted(list(attributes))
    
    def _extract_calls(self, node: ast.FunctionDef) -> List[str]:
        """Extrait les appels de méthodes."""
        calls = set()
        
        for item in ast.walk(node):
            if isinstance(item, ast.Call):
                if isinstance(item.func, ast.Attribute):
                    calls.add(item.func.attr)
                elif isinstance(item.func, ast.Name):
                    calls.add(item.func.id)
        
        return sorted(list(calls))
    
    def _extract_variables(self, node: ast.FunctionDef) -> Set[str]:
        """Extrait les variables utilisées."""
        variables = set()
        
        for item in ast.walk(node):
            if isinstance(item, ast.Name):
                variables.add(item.id)
        
        return variables
    
    def _calculate_complexity(self, node: ast.FunctionDef) -> int:
        """Calcule la complexité cyclomatique."""
        complexity = 1  # Base
        
        for item in ast.walk(node):
            if isinstance(item, (ast.If, ast.While, ast.For, ast.ExceptHandler)):
                complexity += 1
            elif isinstance(item, ast.BoolOp):
                complexity += len(item.values) - 1
        
        return complexity
    
    def _extract_return_type(self, node: ast.FunctionDef) -> str:
        """Extrait le type de retour."""
        if node.returns:
            return ast.unparse(node.returns)
        return "None"
    
    def _detect_errors(self, node: ast.FunctionDef) -> bool:
        """Détecte des erreurs potentielles."""
        # Vérifier variables non définies, appels invalides, etc.
        for item in ast.walk(node):
            # TODO placeholders
            if isinstance(item, ast.Expr):
                if isinstance(item.value, ast.Constant):
                    if isinstance(item.value.value, str) and "TODO" in item.value.value:
                        return True
        
        return False
    
    def _detect_error_types(self, node: ast.FunctionDef) -> List[str]:
        """Détecte les types d'erreurs."""
        errors = []
        
        # Variables non définies
        defined_vars = set()
        for item in ast.walk(node):
            if isinstance(item, ast.Assign):
                for target in item.targets:
                    if isinstance(target, ast.Name):
                        defined_vars.add(target.id)
        
        used_vars = self._extract_variables(node)
        undefined = used_vars - defined_vars - {'self', 'True', 'False', 'None'}
        
        if undefined:
            errors.append(f"UndefinedVariables: {', '.join(list(undefined)[:5])}")
        
        # Méthodes sans return
        has_return = any(isinstance(item, ast.Return) for item in ast.walk(node))
        if not has_return and node.returns and node.returns.id != 'None':
            errors.append("MissingReturn")
        
        return errors
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Calcule des statistiques."""
        total_methods = sum(len(c.methods) for c in self.classes)
        total_complexity = sum(
            m.complexity for c in self.classes for m in c.methods
        )
        methods_with_errors = sum(
            1 for c in self.classes for m in c.methods if m.has_errors
        )
        
        return {
            'total_classes': len(self.classes),
            'total_methods': total_methods,
            'total_lines': len(self.source_lines),
            'average_complexity': total_complexity / total_methods if total_methods > 0 else 0,
            'methods_with_errors': methods_with_errors,
            'error_rate': methods_with_errors / total_methods if total_methods > 0 else 0,
            'total_attributes': sum(len(c.attributes) for c in self.classes)
        }


def analyze_file(filepath: str) -> Dict[str, Any]:
    """Analyse un fichier Python.
    
    Args:
        filepath: Chemin du fichier
        
    Returns:
        Résultats d'analyse
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        source_code = f.read()
    
    analyzer = ASTAnalyzer(source_code)
    return analyzer.analyze()


if __name__ == "__main__":
    import sys
    
    if len(sys.argv) < 2:
        print("Usage: python ast_analyzer.py <file.py>")
        sys.exit(1)
    
    filepath = sys.argv[1]
    results = analyze_file(filepath)
    
    # Sauvegarder résultats
    output_file = Path(filepath).stem + "_ast_analysis.json"
    with open(output_file, 'w', encoding='utf-8') as f:
        json.dump(results, f, indent=2, default=str)
    
    print(f"✅ Analyse terminée: {output_file}")
    print(f"📊 Classes: {results['statistics']['total_classes']}")
    print(f"📊 Méthodes: {results['statistics']['total_methods']}")
    print(f"⚠️  Erreurs: {results['statistics']['methods_with_errors']}")

python"""Refactoring intelligent avec Gemini."""
import google.generativeai as genai
import json
import os
from typing import Dict, List, Any
from pathlib import Path


class GeminiRefactorer:
    """Refactorise le code avec Gemini."""
    
    def __init__(self, api_key: str = None):
        """Initialize Gemini refactorer.
        
        Args:
            api_key: Clé API Gemini
        """
        self.api_key = api_key or os.getenv("GEMINI_API_KEY")
        if not self.api_key:
            raise ValueError("GEMINI_API_KEY required")
        
        genai.configure(api_key=self.api_key)
        self.model = genai.GenerativeModel('gemini-1.5-pro')
    
    def refactor_method(
        self,
        method_code: str,
        method_info: Dict[str, Any],
        context: Dict[str, Any]
    ) -> Dict[str, Any]:
        """Refactorise une méthode avec Gemini.
        
        Args:
            method_code: Code de la méthode
            method_info: Informations AST
            context: Contexte (classe, autres méthodes)
            
        Returns:
            Code refactorisé + explications
        """
        prompt = self._build_refactor_prompt(method_code, method_info, context)
        
        try:
            response = self.model.generate_content(prompt)
            return self._parse_response(response.text)
        except Exception as e:
            return {
                'success': False,
                'error': str(e),
                'original_code': method_code
            }
    
    def _build_refactor_prompt(
        self,
        method_code: str,
        method_info: Dict[str, Any],
        context: Dict[str, Any]
    ) -> str:
        """Construit le prompt pour Gemini."""
        prompt = f"""Tu es un expert en refactoring Python et migration COBOL.

# CONTEXTE
Cette méthode a été migrée automatiquement depuis COBOL et contient des erreurs.

# INFORMATIONS AST
- Nom: {method_info['name']}
- Complexité: {method_info['complexity']}
- Erreurs détectées: {', '.join(method_info['error_types'])}
- Variables utilisées: {', '.join(list(method_info['variables'])[:10])}
- Appels de méthodes: {', '.join(method_info['calls'][:10])}

# CONTEXTE CLASSE
- Classe: {context.get('class_name', 'Unknown')}
- Attributs disponibles: {', '.join(context.get('attributes', [])[:20])}

# CODE ORIGINAL
```python
{method_code}
```

# TÂCHES
1. Corriger TOUTES les erreurs (syntaxe, logique, noms)
2. Simplifier la complexité si > 10
3. Ajouter type hints complets
4. Ajouter docstring Google style
5. Gérer les exceptions
6. Optimiser les performances

# FORMAT DE RÉPONSE (JSON)
Réponds UNIQUEMENT avec un JSON valide:
{{
    "refactored_code": "code corrigé complet",
    "changes": ["liste des modifications"],
    "remaining_issues": ["problèmes non résolus"],
    "complexity_before": {method_info['complexity']},
    "complexity_after": nombre,
    "confidence": 0.0-1.0
}}

⚠️ IMPORTANT: Le code doit être COMPLET et EXÉCUTABLE."""

        return prompt
    
    def _parse_response(self, response_text: str) -> Dict[str, Any]:
        """Parse la réponse de Gemini."""
        try:
            # Extraire JSON de la réponse
            json_start = response_text.find('{')
            json_end = response_text.rfind('}') + 1
            
            if json_start == -1 or json_end == 0:
                raise ValueError("No JSON found in response")
            
            json_str = response_text[json_start:json_end]
            result = json.loads(json_str)
            
            result['success'] = True
            return result
        except Exception as e:
            return {
                'success': False,
                'error': f"Parse error: {str(e)}",
                'raw_response': response_text
            }
    
    def refactor_class(
        self,
        class_info: Dict[str, Any],
        source_code: str
    ) -> Dict[str, Any]:
        """Refactorise une classe complète.
        
        Args:
            class_info: Informations AST de la classe
            source_code: Code source complet
            
        Returns:
            Résultats du refactoring
        """
        results = {
            'class_name': class_info['name'],
            'methods': {},
            'statistics': {
                'total_methods': len(class_info['methods']),
                'refactored_successfully': 0,
                'failed': 0
            }
        }
        
        context = {
            'class_name': class_info['name'],
            'attributes': class_info['attributes'],
            'base_classes': class_info['base_classes']
        }
        
        for method_info in class_info['methods']:
            # Extraire code de la méthode
            lines = source_code.split('\n')
            method_code = '\n'.join(
                lines[method_info['line_start']-1:method_info['line_end']]
            )
            
            # Refactoriser
            result = self.refactor_method(method_code, method_info, context)
            
            results['methods'][method_info['name']] = result
            
            if result.get('success'):
                results['statistics']['refactored_successfully'] += 1
            else:
                results['statistics']['failed'] += 1
        
        return results
    
    def suggest_module_split(
        self,
        class_info: Dict[str, Any]
    ) -> Dict[str, List[str]]:
        """Suggère comment diviser une grosse classe en modules.
        
        Args:
            class_info: Informations sur la classe
            
        Returns:
            Dictionnaire module -> liste de méthodes
        """
        methods = class_info['methods']
        method_names = [m['name'] for m in methods]
        
        prompt = f"""Analyse ces {len(methods)} méthodes Python et suggère comment les organiser en modules cohérents.

# MÉTHODES
{json.dumps(method_names[:100], indent=2)}

# RÈGLES
- Max 30 méthodes par module
- Grouper par domaine fonctionnel
- Noms de modules clairs (ex: deposit_service, loan_application)

# FORMAT RÉPONSE (JSON)
{{
    "suggested_modules": {{
        "module_name_1": ["method1", "method2"],
        "module_name_2": ["method3", "method4"]
    }},
    "rationale": "Explication des choix"
}}
"""
        
        try:
            response = self.model.generate_content(prompt)
            result = self._parse_response(response.text)
            return result.get('suggested_modules', {})
        except Exception as e:
            print(f"Error suggesting modules: {e}")
            return {}

python"""Orchestrateur hybride AST + Gemini."""
import json
import os
from pathlib import Path
from typing import Dict, List, Any
import logging

from ast_analyzer import ASTAnalyzer, analyze_file
from gemini_refactor import GeminiRefactorer


class HybridMigrator:
    """Orchestrateur de migration hybride."""
    
    def __init__(self, gemini_api_key: str = None):
        """Initialize hybrid migrator.
        
        Args:
            gemini_api_key: Clé API Gemini
        """
        self.gemini = GeminiRefactorer(gemini_api_key)
        self.logger = logging.getLogger(__name__)
        
        # Configuration logging
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
    
    def migrate_file(
        self,
        input_file: str,
        output_dir: str = "migrated"
    ) -> Dict[str, Any]:
        """Migre un fichier complet.
        
        Args:
            input_file: Fichier source
            output_dir: Répertoire de sortie
            
        Returns:
            Rapport de migration
        """
        self.logger.info(f"📂 Migration de: {input_file}")
        
        # Créer dossier sortie
        output_path = Path(output_dir)
        output_path.mkdir(exist_ok=True)
        
        # PHASE 1: Analyse AST
        self.logger.info("1️⃣ Analyse AST...")
        ast_analysis = analyze_file(input_file)
        
        # Sauvegarder analyse
        ast_file = output_path / f"{Path(input_file).stem}_ast_analysis.json"
        with open(ast_file, 'w') as f:
            json.dump(ast_analysis, f, indent=2, default=str)
        
        self.logger.info(f"   ✓ {ast_analysis['statistics']['total_methods']} méthodes analysées")
        
        # PHASE 2: Suggestions de modules
        self.logger.info("2️⃣ Suggestion de modules...")
        
        with open(input_file, 'r') as f:
            source_code = f.read()
        
        module_suggestions = {}
        for class_info in ast_analysis['classes']:
            suggestions = self.gemini.suggest_module_split(class_info)
            module_suggestions[class_info['name']] = suggestions
        
        # PHASE 3: Refactoring par module
        self.logger.info("3️⃣ Refactoring avec Gemini...")
        
        refactored_modules = {}
        
        for class_info in ast_analysis['classes']:
            class_name = class_info['name']
            self.logger.info(f"   🔧 Refactoring: {class_name}")
            
            # Refactoriser classe
            results = self.gemini.refactor_class(class_info, source_code)
            
            # Organiser par modules suggérés
            suggestions = module_suggestions.get(class_name, {})
            
            for module_name, method_names in suggestions.items():
                if module_name not in refactored_modules:
                    refactored_modules[module_name] = {
                        'methods': {},
                        'imports': set(),
                        'classes': []
                    }
                
                # Ajouter méthodes refactorisées
                for method_name in method_names:
                    if method_name in results['methods']:
                        method_result = results['methods'][method_name]
                        if method_result.get('success'):
                            refactored_modules[module_name]['methods'][method_name] = \
                                method_result['refactored_code']
        
        # PHASE 4: Générer fichiers de modules
        self.logger.info("4️⃣ Génération des modules...")
        
        generated_files = []
        for module_name, module_data in refactored_modules.items():
            module_file = output_path / f"{module_name}.py"
            
            with open(module_file, 'w') as f:
                # Header
                f.write(f'"""Module {module_name} - Migré automatiquement."""\n')
                f.write('from decimal import Decimal\n')
                f.write('from datetime import datetime\n')
                f.write('from typing import Optional, Dict, Any\n\n')
                f.write('from src.core.processor_base import BaseProcessor\n')
                f.write('from src.core.file_operations import FileOperations\n\n\n')
                
                # Classe
                f.write(f'class {module_name.title().replace("_", "")}:\n')
                f.write('    """Service généré automatiquement."""\n\n')
                f.write('    def __init__(self, file_ops: FileOperations):\n')
                f.write('        self.file_ops = file_ops\n\n')
                
                # Méthodes
                for method_name, method_code in module_data['methods'].items():
                    f.write(f'    {method_code}\n\n')
            
            generated_files.append(str(module_file))
            self.logger.info(f"   ✓ Créé: {module_file}")
        
        # PHASE 5: Générer tests
        self.logger.info("5️⃣ Génération des tests...")
        self._generate_tests(refactored_modules, output_path)
        
        # Rapport final
        report = {
            'input_file': input_file,
            'output_dir': output_dir,
            'ast_analysis': ast_file,
            'modules_generated': len(generated_files),
            'files': generated_files,
            'statistics': ast_analysis['statistics']
        }
        
        # Sauvegarder rapport
        report_file = output_path / "migration_report.json"
        with open(report_file, 'w') as f:
            json.dump(report, f, indent=2, default=str)
        
        self.logger.info(f"\n✅ Migration terminée!")
        self.logger.info(f"📊 Modules générés: {len(generated_files)}")
        self.logger.info(f"📄 Rapport: {report_file}")
        
        return report
    
    def _generate_tests(
        self,
        modules: Dict[str, Any],
        output_path: Path
    ) -> None:
        """Génère des tests unitaires automatiquement."""
        tests_dir = output_path / "tests"
        tests_dir.mkdir(exist_ok=True)
        
        for module_name, module_data in modules.items():
            test_file = tests_dir / f"test_{module_name}.py"
            
            with open(test_file, 'w') as f:
                f.write(f'"""Tests pour {module_name}."""\n')
                f.write('import pytest\n')
                f.write('from decimal import Decimal\n\n')
                f.write(f'from migrated.{module_name} import {module_name.title().replace("_", "")}\n\n\n')
                
                # Générer tests basiques
                f.write(f'class Test{module_name.title().replace("_", "")}:\n')
                f.write('    """Tests générés automatiquement."""\n\n')
                f.write('    def test_initialization(self):\n')
                f.write('        """Test initialisation."""\n')
                f.write('        # TODO: Implémenter test\n')
                f.write('        pass\n')
            
            self.logger.info(f"   ✓ Test créé: {test_file}")


# CLI
def main():
    """Point d'entrée CLI."""
    import argparse
    
    parser = argparse.ArgumentParser(description="Migration hybride AST + Gemini")
    parser.add_argument("input_file", help="Fichier Python source")
    parser.add_argument("--output-dir", default="migrated", help="Dossier de sortie")
    parser.add_argument("--api-key", help="Clé API Gemini (ou GEMINI_API_KEY env)")
    
    args = parser.parse_args()
    
    migrator = HybridMigrator(args.api_key)
    report = migrator.migrate_file(args.input_file, args.output_dir)

    print(f"Modules générés: {report['modules_generated']}")

    print(f"Taux d'erreurs: {report['statistics']['error_rate']:.1%}")
