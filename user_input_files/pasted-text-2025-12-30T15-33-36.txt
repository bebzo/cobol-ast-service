def search_account(ws_search_key: str, master_file: dict, ws_account_rec: dict) -> tuple[str, Decimal, str, str]:
    """5000-search_account."""
    ws_found_flag = 'N'  # MOVE 'N' TO ws_found_flag

    if acct_id in master_file:  # READ master_file INTO ws_account_rec KEY IS acct_id
        ws_found_flag = 'Y'  # MOVE 'Y' TO ws_found_flag
        ws_account_balance = master_file[acct_id]['acct_balance']  # MOVE acct_balance TO ws_account_balance
        ws_account_type = master_file[acct_id]['acct_type']  # MOVE acct_type TO ws_account_type
        ws_account_status = master_file[acct_id]['acct_status']  # MOVE acct_status TO ws_account_status
    else:  # INVALID KEY

def binary_search(ws_search_key: str, tbl_key: list[str], ws_table_size: int) -> tuple[str, int]:
    """5100-binary_search."""
    ws_low = 1  # SET ws_low TO 1
    ws_high = ws_table_size  # SET ws_high TO ws_table_size
    ws_found_flag = 'N'  # MOVE 'N' TO ws_found_flag
    ws_found_index = 0

    while ws_low <= ws_high:  # PERFORM UNTIL ws_low > ws_high
        ws_mid = (ws_low + ws_high) // 2  # COMPUTE ws_mid = (ws_low + ws_high) / 2
        if tbl_key[ws_mid - 1] == ws_search_key:  # IF tbl_key(ws_mid) = ws_search_key
            ws_found_flag = 'Y'  # MOVE 'Y' TO ws_found_flag
            ws_found_index = ws_mid  # MOVE ws_mid TO ws_found_index
            break  # EXIT PERFORM
        elif tbl_key[ws_mid - 1] < ws_search_key:  # ELSE IF tbl_key(ws_mid) < ws_search_key
            ws_low = ws_mid + 1  # ADD 1 TO ws_mid GIVING ws_low
        else:  # ELSE
            ws_high = ws_mid - 1  # SUBTRACT 1 FROM ws_mid GIVING ws_high
    return ws_found_flag, ws_found_index

def hash_lookup(ws_search_key: str, hash_key: list[str], hash_value: list[int], ws_hash_table_size: int) -> tuple[str, int]:
    """5200-hash_lookup."""
    ws_hash_value = (ord(ws_search_key[0]) * 31 + ord(ws_search_key[1])) % ws_hash_table_size  # COMPUTE ws_hash_value = FUNCTION MOD(FUNCTION ORD(ws_search_key(1:1)) * 31 + FUNCTION ORD(ws_search_key(2:1)), ws_hash_table_size)
    ws_hash_value += 1  # ADD 1 TO ws_hash_value
    ws_lookup_result = 0
    ws_found_flag = 'N'
    if hash_key[ws_hash_value - 1] == ws_search_key:  # IF hash_key(ws_hash_value) = ws_search_key
        ws_found_flag = 'Y'  # MOVE 'Y' TO ws_found_flag
        ws_lookup_result = hash_value[ws_hash_value - 1]  # MOVE hash_value(ws_hash_value) TO ws_lookup_result
    else:  # ELSE
        ws_found_flag, ws_lookup_result = probe_hash_table(ws_search_key, hash_key, hash_value, ws_hash_value, ws_hash_table_size)  # PERFORM 5250-probe_hash_table
    return ws_found_flag, ws_lookup_result

def probe_hash_table(ws_search_key: str, hash_key: list[str], hash_value: list[int], ws_hash_value: int, ws_hash_table_size: int) -> tuple[str, int]:
    """5250-probe_hash_table."""
    ws_probe_start = ws_hash_value  # MOVE ws_hash_value TO ws_probe_start
    ws_hash_value += 1  # ADD 1 TO ws_hash_value
    ws_lookup_result = 0
    ws_found_flag = 'N'
    while ws_hash_value != ws_probe_start:  # PERFORM UNTIL ws_hash_value = ws_probe_start
        if ws_hash_value > ws_hash_table_size:  # IF ws_hash_value > ws_hash_table_size
            ws_hash_value = 1  # MOVE 1 TO ws_hash_value

        if hash_key[ws_hash_value - 1] == ws_search_key:  # IF hash_key(ws_hash_value) = ws_search_key
            ws_found_flag = 'Y'  # MOVE 'Y' TO ws_found_flag
            ws_lookup_result = hash_value[ws_hash_value - 1]  # MOVE hash_value(ws_hash_value) TO ws_lookup_result
            break  # EXIT PERFORM

        if hash_key[ws_hash_value - 1] == ' ':  # IF hash_key(ws_hash_value) = SPACES
            break  # EXIT PERFORM

        ws_hash_value += 1  # ADD 1 TO ws_hash_value
    return ws_found_flag, ws_lookup_result

def interest_calculation(ws_account_balance: Decimal, ws_days_in_period: int, ws_interest_method: str) -> Decimal:
    """7000-interest_calculation."""
    ws_simple_interest = Decimal("0.00")
    ws_compound_interest = Decimal("0.00")

def determine_rate_tier(ws_account_balance: Decimal) -> Decimal:

    match True:  # EVALUATE TRUE
        case ws_account_balance < 1000:  # WHEN ws_account_balance < 1000

        case ws_account_balance < 10000:  # WHEN ws_account_balance < 10000
            ws_interest_rate = Decimal("1.0")  # MOVE 1.0 TO ws_interest_rate
        case ws_account_balance < 50000:  # WHEN ws_account_balance < 50000
            ws_interest_rate = Decimal("1.5")  # MOVE 1.5 TO ws_interest_rate
        case ws_account_balance < 100000:  # WHEN ws_account_balance < 100000
            ws_interest_rate = Decimal("2.0")  # MOVE 2.0 TO ws_interest_rate
        case _:  # WHEN OTHER
            ws_interest_rate = Decimal("2.5")  # MOVE 2.5 TO ws_interest_rate

def calculate_simple_interest(ws_account_balance: Decimal, ws_interest_rate: Decimal, ws_days_in_period: int) -> Decimal:
    """7200-calculate_simple_interest."""
    ws_simple_interest = ws_account_balance * ws_interest_rate * ws_days_in_period / 36500  # COMPUTE ws_simple_interest = ws_account_balance * ws_interest_rate * ws_days_in_period / 36500.
    return ws_simple_interest

def calculate_compound_interest(ws_account_balance: Decimal, ws_interest_rate: Decimal, ws_days_in_period: int) -> Decimal:
    """7300-calculate_compound_interest."""
    ws_compound_factor = (1 + ws_interest_rate / 36500) ** ws_days_in_period  # COMPUTE ws_compound_factor = (1 + ws_interest_rate / 36500) ** ws_days_in_period
    ws_compound_interest = ws_account_balance * (ws_compound_factor - 1)  # COMPUTE ws_compound_interest = ws_account_balance * (ws_compound_factor - 1).
    return ws_compound_interest

def apply_interest(ws_account_balance: Decimal, ws_simple_interest: Decimal, ws_compound_interest: Decimal, ws_interest_method: str) -> Decimal:
    """7400-apply_interest."""
    if ws_interest_method == 'S':  # IF ws_interest_method = 'S'
        ws_account_balance += ws_simple_interest  # ADD ws_simple_interest TO ws_account_balance
    else:  # ELSE
        ws_account_balance += ws_compound_interest  # ADD ws_compound_interest TO ws_account_balance
    update_account(ws_account_balance)  # PERFORM 2350-update_account
    return ws_account_balance

def fee_processing(ws_account_type: str, ws_trans_count: int, ws_free_trans_limit: int, ws_per_trans_fee: Decimal, ws_account_balance: Decimal, ws_min_balance_waiver: Decimal, ws_customer_tier: str) -> tuple[Decimal, Decimal]:
    """8000-fee_processing."""
    ws_monthly_fee = calculate_monthly_fee(ws_account_type)  # PERFORM 8100-calculate_monthly_fee
    ws_trans_fee = calculate_transaction_fees(ws_trans_count, ws_free_trans_limit, ws_per_trans_fee)  # PERFORM 8200-calculate_transaction_fees
    ws_monthly_fee, ws_trans_fee = apply_fee_waivers(ws_account_balance, ws_min_balance_waiver, ws_customer_tier, ws_monthly_fee, ws_trans_fee)  # PERFORM 8300-apply_fee_waivers
    ws_account_balance = deduct_fees(ws_account_balance, ws_monthly_fee, ws_trans_fee)  # PERFORM 8400-deduct_fees
    return ws_account_balance, ws_monthly_fee

def calculate_monthly_fee(ws_account_type: str) -> Decimal:
    """8100-calculate_monthly_fee."""
    match ws_account_type:  # EVALUATE ws_account_type
        case 'CHK':  # WHEN 'CHK'
            ws_monthly_fee = Decimal("12.00")  # MOVE 12.00 TO ws_monthly_fee
        case 'SAV':  # WHEN 'SAV'
            ws_monthly_fee = Decimal("5.00")  # MOVE 5.00 TO ws_monthly_fee
        case 'PRM':  # WHEN 'PRM'
            ws_monthly_fee = Decimal("25.00")  # MOVE 25.00 TO ws_monthly_fee
        case _:  # WHEN OTHER
            ws_monthly_fee = Decimal("0.00")  # MOVE 0.00 TO ws_monthly_fee
    return ws_monthly_fee

def calculate_transaction_fees(ws_trans_count: int, ws_free_trans_limit: int, ws_per_trans_fee: Decimal) -> Decimal:
    """8200-calculate_transaction_fees."""
    if ws_trans_count > ws_free_trans_limit:  # IF ws_trans_count > ws_free_trans_limit
        ws_excess_trans = ws_trans_count - ws_free_trans_limit  # COMPUTE ws_excess_trans = ws_trans_count - ws_free_trans_limit
        ws_trans_fee = ws_excess_trans * ws_per_trans_fee  # COMPUTE ws_trans_fee = ws_excess_trans * ws_per_trans_fee
    else:  # ELSE
        ws_trans_fee = Decimal("0.00")  # MOVE ZEROES TO ws_trans_fee
    return ws_trans_fee

def apply_fee_waivers(ws_account_balance: Decimal, ws_min_balance_waiver: Decimal, ws_customer_tier: str, ws_monthly_fee: Decimal, ws_trans_fee: Decimal) -> tuple[Decimal, Decimal]:
    """8300-apply_fee_waivers."""
    if ws_account_balance >= ws_min_balance_waiver:  # IF ws_account_balance >= ws_min_balance_waiver
        ws_monthly_fee = Decimal("0.00")  # MOVE ZEROES TO ws_monthly_fee

    if ws_customer_tier == 'GOLD' or ws_customer_tier == 'PLATINUM':  # IF ws_customer_tier = 'GOLD' OR 'PLATINUM'
        ws_trans_fee = ws_trans_fee * Decimal("0.5")  # COMPUTE ws_trans_fee = ws_trans_fee * 0.5
    return ws_monthly_fee, ws_trans_fee

           (1 + ws_monthly_rate) ** ws_loan_term_months

    data.WS_LOAN_MONTHLY_PMT = data.WS_LOAN_AMOUNT * data.WS_MONTHLY_RATE * data.WS_COMPOUND_FACTOR / (data.WS_COMPOUND_FACTOR - 1)
    data.WS_LOAN_PRINCIPAL_BAL = data.WS_LOAN_AMOUNT

def create_amortization(data: Data, loan_mortgage: bool, ws_property_tax: Decimal, ws_insurance_premium: Decimal, ws_loan_term_months: int) -> None:
    """
        COBOL Translation:
        10600-create_amortization.
        MOVE ws_loan_amount TO ws_running_balance
        MOVE FUNCTION current_date TO ws_payment_date
        PERFORM VARYING ws_amort_idx FROM 1 BY 1
           UNTIL ws_amort_idx > ws_loan_term_months
           PERFORM 10650-calculate_payment_split
        end_perform.
    """
    data.WS_RUNNING_BALANCE = data.WS_LOAN_AMOUNT
    data.WS_PAYMENT_DATE = 20240101 # replace with function to current date int value

    for data.WS_AMORT_IDX in range(1, ws_loan_term_months + 1):
        calculate_payment_split(data, loan_mortgage, ws_property_tax, ws_insurance_premium)
    data.WS_AMORT_IDX = 0 # reset the index
    

def calculate_payment_split(data: Data, loan_mortgage: bool, ws_property_tax: Decimal, ws_insurance_premium: Decimal) -> None:
    """
        COBOL Translation:
        10650-calculate_payment_split.
        COMPUTE amort_interest(ws_amort_idx) = 
           ws_running_balance * ws_monthly_rate
        COMPUTE amort_principal(ws_amort_idx) = 
           ws_loan_monthly_pmt - amort_interest(ws_amort_idx)
        SUBTRACT amort_principal(ws_amort_idx) 
           FROM ws_running_balance
        MOVE ws_running_balance 
           TO amort_balance(ws_amort_idx)
        MOVE ws_amort_idx TO amort_payment_num(ws_amort_idx)
        MOVE ws_loan_monthly_pmt 
           TO amort_payment_amt(ws_amort_idx)
        IF loan_mortgage
           COMPUTE amort_escrow(ws_amort_idx) = 
              (ws_property_tax + ws_insurance_premium) / 12
           COMPUTE amort_total_pmt(ws_amort_idx) = 
              ws_loan_monthly_pmt + 
              amort_escrow(ws_amort_idx) + ws_pmi_amount
        ELSE
           MOVE ws_loan_monthly_pmt 
              TO amort_total_pmt(ws_amort_idx)
        end_if
        PERFORM 10660-advance_payment_date.
    """
    data.AMORT_INTEREST[data.WS_AMORT_IDX - 1] = data.WS_RUNNING_BALANCE * data.WS_MONTHLY_RATE
    data.AMORT_PRINCIPAL[data.WS_AMORT_IDX - 1] = data.WS_LOAN_MONTHLY_PMT - data.AMORT_INTEREST[data.WS_AMORT_IDX - 1]
    data.WS_RUNNING_BALANCE -= data.AMORT_PRINCIPAL[data.WS_AMORT_IDX - 1]
    data.AMORT_BALANCE[data.WS_AMORT_IDX - 1] = data.WS_RUNNING_BALANCE
    data.AMORT_PAYMENT_NUM[data.WS_AMORT_IDX - 1] = data.WS_AMORT_IDX
    data.AMORT_PAYMENT_AMT[data.WS_AMORT_IDX - 1] = data.WS_LOAN_MONTHLY_PMT

    if loan_mortgage:
        data.AMORT_ESCROW[data.WS_AMORT_IDX - 1] = (ws_property_tax + ws_insurance_premium) / 12
        data.AMORT_TOTAL_PMT[data.WS_AMORT_IDX - 1] = data.WS_LOAN_MONTHLY_PMT + data.AMORT_ESCROW[data.WS_AMORT_IDX - 1] + data.WS_PMI_AMOUNT
    else:
        data.AMORT_TOTAL_PMT[data.WS_AMORT_IDX - 1] = data.WS_LOAN_MONTHLY_PMT

    advance_payment_date(data)

def advance_payment_date(data: Data) -> None:
    """
        COBOL Translation:
        10660-advance_payment_date.

           ADD 1 TO ws_payment_year
        end_if
        COMPUTE amort_payment_date(ws_amort_idx) = 
           ws_payment_year * 10000 + 
           ws_payment_month * 100 + 01.
    """
    data.WS_PAYMENT_MONTH += 1
    if data.WS_PAYMENT_MONTH > 12:
        data.WS_PAYMENT_MONTH = 1
        data.WS_PAYMENT_YEAR += 1
    data.AMORT_PAYMENT_DATE[data.WS_AMORT_IDX - 1] = data.WS_PAYMENT_YEAR * 10000 + data.WS_PAYMENT_MONTH * 100 + 1

    ws_luhn_check = calculate_luhn_check(ws_card_number_temp) # PERFORM 23115-calculate_luhn_check

def calculate_luhn_check(ws_card_number_temp: str) -> int:
    """
    23115-calculate_luhn_check.
    """
    ws_luhn_sum = 0  # MOVE ZEROES TO ws_luhn_sum
    for ws_luhn_idx in range(15, 0, -1): # PERFORM VARYING ws_luhn_idx FROM 15 BY -1 UNTIL ws_luhn_idx < 1
        ws_luhn_digit = int(ws_card_number_temp[ws_luhn_idx-1]) # MOVE ws_card_number_temp(ws_luhn_idx:1) TO ws_luhn_digit
        if (16 - ws_luhn_idx) % 2 == 0: # IF FUNCTION MOD(16 - ws_luhn_idx, 2) = 0
            ws_luhn_digit *= 2  # MULTIPLY 2 BY ws_luhn_digit
            if ws_luhn_digit > 9: # IF ws_luhn_digit > 9
                ws_luhn_digit -= 9  # SUBTRACT 9 FROM ws_luhn_digit
        ws_luhn_sum += ws_luhn_digit  # ADD ws_luhn_digit TO ws_luhn_sum
    ws_luhn_check = (10 - (ws_luhn_sum % 10)) % 10  # COMPUTE ws_luhn_check = FUNCTION MOD(10 - FUNCTION MOD(ws_luhn_sum, 10), 10)
    return ws_luhn_check

    """28100-calculate_daily_interest.

        WHEN 'SAV'

        WHEN 'MMA'
            PERFORM 28120-money_market_interest
        WHEN 'CD'
            PERFORM 28130-cd_interest
        WHEN 'CHK'
            IF acct_interest_bearing = 'Y'
                PERFORM 28140-checking_interest
            end_if
    end_evaluate."""
    global ACCT_TYPE, ACCT_INTEREST_BEARING
    match ACCT_TYPE:
        case 'SAV':
            savings_interest()
        case 'MMA':
            money_market_interest()
        case 'CD':

        case 'CHK':
            if ACCT_INTEREST_BEARING == 'Y':
                checking_interest()
