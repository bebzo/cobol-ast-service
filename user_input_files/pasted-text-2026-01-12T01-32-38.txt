from dataclasses import dataclass
from decimal import Decimal
from typing import Optional
from datetime import datetime

@dataclass
class Account:
    account_id: str
    customer_id: str
    account_number: str
    account_type: AccountType
    balance: Decimal
    currency: str
    status: AccountStatus
    opening_date: datetime
    closing_date: Optional[datetime] = None
    
    def can_withdraw(self, amount: Decimal) -> bool:
        return self.balance - amount >= self.get_minimum_balance()
    
    def get_minimum_balance(self) -> Decimal:
        return ACCOUNT_MINIMUMS[self.account_type]

@dataclass
class Transaction:
    transaction_id: str
    account_id: str
    amount: Decimal
    transaction_type: TransactionType
    timestamp: datetime
    reference: str
    status: TransactionStatus
    
    def is_valid(self) -> bool:
        return self.amount > 0 and len(self.reference) <= 100

# application/use_cases/process_transaction.py
from abc import ABC, abstractmethod
from typing import Protocol

class TransactionRepository(Protocol):
    def save(self, transaction: Transaction) -> None: ...
    def get_by_id(self, transaction_id: str) -> Transaction: ...

class AccountRepository(Protocol):
    def get_by_id(self, account_id: str) -> Account: ...
    def update_balance(self, account_id: str, new_balance: Decimal) -> None: ...

class ProcessTransactionUseCase:
    def __init__(
        self,
        transaction_repo: TransactionRepository,
        account_repo: AccountRepository,
        event_bus: EventBus,
        logger: Logger
    ):
        self.transaction_repo = transaction_repo
        self.account_repo = account_repo
        self.event_bus = event_bus
        self.logger = logger
    
    async def execute(self, command: ProcessTransactionCommand) -> TransactionResult:
        try:
            # Validation
            account = self.account_repo.get_by_id(command.account_id)
            if not account.can_withdraw(command.amount):
                return TransactionResult.insufficient_funds()
            
            # Business logic
            transaction = Transaction.create(
                account_id=command.account_id,
                amount=command.amount,
                type=command.transaction_type
            )
            
            # Update account
            new_balance = account.balance - command.amount
            self.account_repo.update_balance(account.account_id, new_balance)
            
            # Save transaction
            self.transaction_repo.save(transaction)
            
            # Publish events
            await self.event_bus.publish(TransactionCompletedEvent(
                transaction_id=transaction.transaction_id,
                account_id=account.account_id,
                new_balance=new_balance
            ))
            
            return TransactionResult.success(transaction)
            
        except Exception as e:
            self.logger.error(f"Transaction failed: {e}")
            return TransactionResult.failure(str(e))

from sqlalchemy.orm import Session
from infrastructure.models import AccountORM

class PostgresAccountRepository(AccountRepository):
    def __init__(self, session: Session):
        self.session = session
    
    def get_by_id(self, account_id: str) -> Account:
        orm_account = self.session.query(AccountORM).filter_by(
            account_id=account_id
        ).first()
        
        if not orm_account:
            raise AccountNotFoundError(account_id)
            
        return Account(
            account_id=orm_account.account_id,
            customer_id=orm_account.customer_id,
            account_number=orm_account.account_number,
            account_type=AccountType(orm_account.account_type),
            balance=orm_account.balance,
            currency=orm_account.currency,
            status=AccountStatus(orm_account.status),
            opening_date=orm_account.opening_date
        )
    
    def update_balance(self, account_id: str, new_balance: Decimal) -> None:
        orm_account = self.session.query(AccountORM).filter_by(
            account_id=account_id
        ).first()
        
        if not orm_account:
            raise AccountNotFoundError(account_id)
            
        orm_account.balance = new_balance
        orm_account.last_updated = datetime.utcnow()
        
        self.session.commit()

from sqlalchemy.orm import Session

app = FastAPI(title="Transaction Service", version="1.0.0")

@app.post("/api/v1/transactions")
async def process_transaction(
    request: TransactionRequest,
    use_case: ProcessTransactionUseCase = Depends(get_transaction_use_case)
):
    command = ProcessTransactionCommand(
        account_id=request.account_id,
        amount=request.amount,
        transaction_type=request.transaction_type
    )
    
    result = await use_case.execute(command)
    
    if result.is_success:
        return TransactionResponse(
            transaction_id=result.transaction.transaction_id,
            status="completed",
            new_balance=result.new_balance
        )
    else:
        raise HTTPException(
            status_code=400, 
            detail=result.error_message
        )

-- accounts table
CREATE TABLE accounts (
    account_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    customer_id UUID NOT NULL REFERENCES customers(id),
    account_number VARCHAR(20) UNIQUE NOT NULL,
    account_type VARCHAR(20) NOT NULL CHECK (account_type IN ('CHECKING', 'SAVINGS', 'LOAN', 'CREDIT_CARD')),
    balance DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    currency VARCHAR(3) NOT NULL DEFAULT 'USD',
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    opening_date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    closing_date TIMESTAMP,
    minimum_balance DECIMAL(15,2) DEFAULT 0.00,
    overdraft_limit DECIMAL(15,2) DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- transactions table with partitioning
CREATE TABLE transactions (
    transaction_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    account_id UUID NOT NULL REFERENCES accounts(account_id),
    amount DECIMAL(15,2) NOT NULL,
    transaction_type VARCHAR(50) NOT NULL,
    reference VARCHAR(100),
    status VARCHAR(20) NOT NULL DEFAULT 'PENDING',
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) PARTITION BY RANGE (timestamp);

-- Create monthly partitions
CREATE TABLE transactions_2024_01 PARTITION OF transactions
    FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');

from kafka import KafkaProducer, KafkaConsumer
import json

class KafkaEventBus(EventBus):
    def __init__(self, brokers: List[str]):
        self.producer = KafkaProducer(
            bootstrap_servers=brokers,
            value_serializer=lambda v: json.dumps(v).encode('utf-8')
        )
        
    async def publish(self, event: DomainEvent) -> None:
        topic = self.get_topic_for_event(event)
        
        await self.producer.send(
            topic=topic,
            value={
                'event_id': str(uuid.uuid4()),
                'event_type': event.__class__.__name__,
                'aggregate_id': event.aggregate_id,
                'timestamp': datetime.utcnow().isoformat(),
                'data': event.to_dict()
            }
        )
        
    def subscribe(self, event_type: str, handler: Callable) -> None:
        consumer = KafkaConsumer(
            self.get_topic_for_event_type(event_type),
            bootstrap_servers=self.brokers,
            value_deserializer=lambda m: json.loads(m.decode('utf-8'))
        )
        
        for message in consumer:
            event_data = message.value
            event = self.reconstruct_event(event_data)
            handler(event)

# infrastructure/security/jwt_handler.py
from jose import jwt, JWTError
from passlib.context import CryptContext

class JWTHandler:
    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
    
    def create_access_token(self, data: dict, expires_delta: timedelta = None) -> str:
        to_encode = data.copy()
        expire = datetime.utcnow() + (expires_delta or timedelta(minutes=15))
        to_encode.update({"exp": expire})
        
        encoded_jwt = jwt.encode(to_encode, self.secret_key, algorithm=self.algorithm)
        return encoded_jwt
    
    def verify_token(self, token: str) -> Optional[dict]:
        try:
            payload = jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
            return payload
        except JWTError:
            return None
    
    def verify_password(self, plain_password: str, hashed_password: str) -> bool:
        return self.pwd_context.verify(plain_password, hashed_password)
    
    def get_password_hash(self, password: str) -> str:
        return self.pwd_context.hash(password)

# infrastructure/security/encryption_service.py
from cryptography.fernet import Fernet

class EncryptionService:
    def __init__(self, encryption_key: bytes):
        self.cipher_suite = Fernet(encryption_key)
    
    def encrypt_sensitive_data(self, data: str) -> str:
        return self.cipher_suite.encrypt(data.encode()).decode()
    
    def decrypt_sensitive_data(self, encrypted_data: str) -> str:
        return self.cipher_suite.decrypt(encrypted_data.encode()).decode()
    
    def encrypt_account_number(self, account_number: str) -> str:
        return self.encrypt_sensitive_data(account_number)
    
    def encrypt_ssn(self, ssn: str) -> str:
        return self.encrypt_sensitive_data(ssn)

# infrastructure/observability/tracing.py
from opentelemetry import trace
from opentelemetry.exporter.jaeger.thrift import JaegerExporter
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor

def setup_tracing(service_name: str, jaeger_endpoint: str) -> None:
    trace.set_tracer_provider(TracerProvider())
    
    jaeger_exporter = JaegerExporter(
        agent_host_name=jaeger_endpoint.split(":")[0],
        agent_port=int(jaeger_endpoint.split(":")[1]),
    )
    
    span_processor = BatchSpanProcessor(jaeger_exporter)
    trace.get_tracer_provider().add_span_processor(span_processor)
    
    return trace.get_tracer(service_name)

# Usage in services
tracer = setup_tracing("account-service", "jaeger:6831")

@app.get("/api/v1/accounts/{account_id}")
async def get_account(account_id: str):
    with tracer.start_as_current_span("get_account") as span:
        span.set_attribute("account.id", account_id)
        
        account = account_service.get_account(account_id)
        
        span.set_attribute("account.balance", float(account.balance))
        span.set_attribute("account.type", account.account_type)
        
        return account
