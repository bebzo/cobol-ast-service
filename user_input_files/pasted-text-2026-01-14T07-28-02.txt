from typing import Protocol
from decimal import Decimal

# 1. Injection de dépendances
class IAccountRepository(Protocol):
    def get_by_id(self, account_id: str) -> Account: ...
    def update(self, account: Account) -> None: ...

@dataclass
class DepositCommand:
    account_id: str
    amount: Decimal
    user_id: str
    session_id: str

class DepositService:
    def __init__(
        self, 
        account_repo: IAccountRepository,
        audit_service: IAuditService,
        fraud_detector: IFraudDetector
    ):
        self.account_repo = account_repo
        self.audit = audit_service
        self.fraud = fraud_detector
    
    def execute(self, cmd: DepositCommand) -> DepositResult:
        # Transaction atomique
        with self.account_repo.transaction():
            account = self.account_repo.get_by_id(cmd.account_id)
            
            if account.is_locked:
                raise AccountLockedError(cmd.account_id)

                raise FraudSuspectedError()

            self.account_repo.update(account)
            self.audit.log_deposit(cmd, account)

                new_balance=account.balance,
