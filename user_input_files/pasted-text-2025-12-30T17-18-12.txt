def _10600_create_amortization_C7(ws_loan_amount: Decimal, ws_loan_term_months: int, ws_monthly_rate: Decimal, ws_loan_monthly_pmt: Decimal, loan_mortgage: LoanMortgage, ws_pmi_amount: Decimal) -> list[AmortEntry]:
    """10600-create_amortization."""
    ws_running_balance: Decimal = Decimal("0.00")
    ws_payment_date: date = date.today()
    ws_payment_month: int = ws_payment_date.month
    ws_payment_year: int = ws_payment_date.year

    amort_entries: list[AmortEntry] = []
    ws_running_balance = ws_loan_amount

    for ws_amort_idx in range(1, ws_loan_term_months + 1):
        amort_entry = _10650_calculate_payment_split_C7(ws_amort_idx, ws_running_balance, ws_monthly_rate, ws_loan_monthly_pmt, loan_mortgage, ws_payment_month, ws_payment_year, ws_pmi_amount)
        ws_running_balance = amort_entry.balance

def _10650_calculate_payment_split_C7(ws_amort_idx: int, ws_running_balance: Decimal, ws_monthly_rate: Decimal, ws_loan_monthly_pmt: Decimal, loan_mortgage: LoanMortgage, ws_payment_month: int, ws_payment_year: int, ws_pmi_amount: Decimal) -> AmortEntry:
    """10650-calculate_payment_split."""
    amort_interest: Decimal = ws_running_balance * ws_monthly_rate
    amort_principal: Decimal = ws_loan_monthly_pmt - amort_interest
    ws_running_balance -= amort_principal

    amort_escrow: Decimal = Decimal("0.00")
    if loan_mortgage.loan_mortgage:
        amort_escrow = (loan_mortgage.property_tax + loan_mortgage.insurance_premium) / 12
        amort_total_pmt: Decimal = ws_loan_monthly_pmt + amort_escrow + ws_pmi_amount
    else:
        amort_total_pmt = ws_loan_monthly_pmt

    amort_entry = AmortEntry()
    amort_entry.interest = amort_interest
    amort_entry.principal = amort_principal
    amort_entry.balance = ws_running_balance
    amort_entry.payment_num = ws_amort_idx
    amort_entry.payment_amt = ws_loan_monthly_pmt
    amort_entry.escrow = amort_escrow
    amort_entry.total_pmt = amort_total_pmt

    amort_entry.payment_date = _10660_advance_payment_date_C7(ws_payment_month, ws_payment_year)
    return amort_entry

def _10660_advance_payment_date_C7(ws_payment_month: int, ws_payment_year: int) -> int:

    return amort_payment_date
