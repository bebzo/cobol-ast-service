def validate_python_code(code_str, filename="generated.py"):
    """Vérifie si le code compile et retourne les erreurs"""

    try:
        ast.parse(code_str)
    except SyntaxError as e:
        errors.append(f"SYNTAX ERROR: {e}")

    temp_file.write_text(code_str)
    
    # 3. Vérification avec python -m py_compile
    result = subprocess.run(
        [sys.executable, "-m", "py_compile", str(temp_file)],
        capture_output=True,
        text=True
    )
    
    if result.returncode != 0:
        errors.append(f"COMPILATION ERROR: {result.stderr}")
    
    # 4. Nettoyage
    temp_file.unlink(missing_ok=True)
    
    return len(errors) == 0, errors

# correcteur_rapide.py - Corrections regex rapides
import re

def quick_fixes(code):
    """Corrections rapides pour les patterns connus"""
    
    fixes = [
        # 1. PERFORM → appels de fonction
        (r'PERFORM (\d+-[A-Z-]+)', r'self.\1()'),
        
        # 2. PIC 9 → Decimal
        (r'PIC 9\(\d+\)V9\(\d+\)', r'Decimal'),
        
        # 3. MOVE → assignation
        (r'MOVE ([A-Z-]+) TO ([A-Z-]+)', r'\2 = \1'),
        
        # 4. ADD ... TO
        (r'ADD ([A-Z-]+) TO ([A-Z-]+)', r'\2 += \1'),
        
        # 5. IF sans THEN
        (r'IF ([^:]+):', r'if \1:'),
        
        # 6. Variables COBOL avec tirets → underscores
        (r'(WS-[A-Z]+)-([A-Z]+)', r'\1_\2'),
        
        # 7. COMPUTE → assignation simple
        (r'COMPUTE ([A-Z-]+) =', r'\1 ='),
        
        # 8. DISPLAY → print
        (r'DISPLAY "([^"]+)"', r'print("\1")'),
        
        # 9. END-IF/END-PERFORM → dé-indentation
        (r'^\s*(END-[A-Z]+)', r''),
        
        # 10. Division headers → commentaires
        (r'^([A-Z]+ DIVISION)\.', r'# \1'),
    ]
    
    for pattern, replacement in fixes:
        code = re.sub(pattern, replacement, code, flags=re.MULTILINE)
    
    return code

# correction_pipeline.py
from enum import Enum
from dataclasses import dataclass
from typing import List, Optional
import ast
import subprocess
import tempfile
from pathlib import Path

class CorrectionLevel(Enum):
    SYNTAX = 1          # Problèmes de syntaxe basique
    IMPORTS = 2         # Modules manquants
    SEMANTIC = 3        # Logique métier incorrecte
    OPTIMIZATION = 4    # Améliorations de performance

@dataclass
class CodeIssue:
    level: CorrectionLevel
    description: str
    line: int
    column: int
    suggestion: Optional[str] = None

class HierarchicalCorrector:
    def __init__(self):
        self.correction_strategies = {
            CorrectionLevel.SYNTAX: self._fix_syntax,
            CorrectionLevel.IMPORTS: self._fix_imports,
            CorrectionLevel.SEMANTIC: self._fix_semantic,
            CorrectionLevel.OPTIMIZATION: self._optimize_code
        }
    
    def correct_python(self, python_code: str, cobol_context: dict = None) -> str:
        """Pipeline de correction hiérarchique"""
        
        corrected_code = python_code
        iteration = 0
        max_iterations = 5
        
        while iteration < max_iterations:
            iteration += 1
            
            # 1. Détection des problèmes
            issues = self._detect_issues(corrected_code)
            
            if not issues:
                print(f"✅ Code validé après {iteration} itération(s)")
                return corrected_code
            
            # 2. Tri par priorité (syntaxe d'abord)
            issues.sort(key=lambda x: x.level.value)
            
            # 3. Application des corrections
            for issue in issues:
                if issue.level in self.correction_strategies:
                    print(f"🔄 Correction niveau {issue.level.name}: {issue.description}")
                    corrected_code = self.correction_strategies[issue.level](
                        corrected_code, issue, cobol_context
                    )
            
            # 4. Vérification intermédiaire
            if self._compiles(corrected_code):
                print(f"✓ Compilation réussie à l'itération {iteration}")
            else:
                print(f"⚠️ Problèmes persistants à l'itération {iteration}")
        
        return corrected_code
    
    def _detect_issues(self, code: str) -> List[CodeIssue]:
        """Détecte tous les problèmes dans le code"""
        issues = []
        
        # Analyse syntaxique
        try:
            tree = ast.parse(code)
        except SyntaxError as e:
            issues.append(CodeIssue(
                level=CorrectionLevel.SYNTAX,
                description=str(e),
                line=e.lineno or 1,
                column=e.offset or 1
            ))
            return issues  # Syntax error first
        
        # Vérification des imports
        issues.extend(self._check_imports(tree, code))
        
        # Vérification sémantique
        issues.extend(self._check_semantics(tree, code))
        
        return issues
    
    def _fix_syntax(self, code: str, issue: CodeIssue, context=None) -> str:
        """Correction syntaxique avec IA"""
        prompt = f"""
        Corrige cette erreur de syntaxe Python:
        
        Erreur: {issue.description}
        Ligne {issue.line}, colonne {issue.column}
        
        Code problématique:
        ```python
        {self._extract_context(code, issue.line)}
        ```
        
        Contexte (généré depuis COBOL):
        {context.get('original_cobol_snippet', '') if context else ''}
        
        Fournis UNIQUEMENT le code corrigé.
        """
        
        return self._call_ai_corrector(prompt, code, issue.line)
    
    def _fix_imports(self, code: str, issue: CodeIssue, context=None) -> str:
        """Ajoute les imports manquants basés sur le contexte COBOL"""
        
        # Mapping COBOL → Python imports
        cobol_to_python_imports = {
            'COMPUTE': ['from decimal import Decimal', 'import math'],
            'MOVE': [],  # Pas d'import spécial
            'PERFORM': [],  # Déjà en Python
            'DISPLAY': ['import sys', 'from datetime import datetime'],
            'ACCEPT': ['import sys'],
            'FILE-CONTROL': ['import os', 'import json', 'from pathlib import Path'],
            'CALL': ['import subprocess'],
        }
        
        # Analyse le code COBOL d'origine pour déterminer les imports nécessaires
        required_imports = set()
        if context and 'cobol_keywords' in context:
            for keyword in context['cobol_keywords']:
                if keyword in cobol_to_python_imports:
                    required_imports.update(cobol_to_python_imports[keyword])
        
        # Ajoute les imports au début du fichier
        lines = code.split('\n')
        import_section = []
        
        # Trouve où insérer les imports
        insert_position = 0
        for i, line in enumerate(lines):
            if line.strip() and not line.strip().startswith(('import ', 'from ')):
                insert_position = i
                break
        
        # Crée la section d'imports
        for imp in sorted(required_imports):
            if imp not in code:
                import_section.append(imp)
        
        if import_section:
            lines.insert(insert_position, '\n'.join(import_section))
        
        return '\n'.join(lines)
    
    def _compiles(self, code: str) -> bool:
        """Vérifie si le code compile"""
        try:
            with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
                f.write(code)
                temp_path = f.name
            
            result = subprocess.run(
                ['python', '-m', 'py_compile', temp_path],
                capture_output=True,
                text=True
            )
            
            Path(temp_path).unlink(missing_ok=True)
            return result.returncode == 0
            
        except Exception:
            return False
    
    def _extract_context(self, code: str, line: int, context_lines=3) -> str:
        """Extrait le contexte autour d'une ligne"""
        lines = code.split('\n')
        start = max(0, line - context_lines - 1)
        end = min(len(lines), line + context_lines)
        
        context = []
        for i in range(start, end):
            prefix = '>>> ' if i == line - 1 else '    '
            context.append(f"{prefix}{lines[i]}")
        
        return '\n'.join(context)

import asyncio
from concurrent.futures import ThreadPoolExecutor
import redis
import pickle
from datetime import timedelta

class ProductionPipeline:
    def __init__(self, redis_url="redis://localhost:6379"):
        self.corrector = HierarchicalCorrector()
        self.knowledge_base = COBOLPythonKnowledgeBase()
        self.redis = redis.from_url(redis_url)
        self.executor = ThreadPoolExecutor(max_workers=4)
        
        # Cache pour les corrections fréquentes
        self.correction_cache = {}
    
    async def process_cobol_file(self, cobol_content: str, 
                               file_metadata: Dict = None) -> Dict:
        """Traite un fichier COBOL complet"""
        
        # Clé de cache basée sur le hash du contenu
        cache_key = f"cobol:{hashlib.md5(cobol_content.encode()).hexdigest()}"
        
        # Vérifie le cache
        cached_result = self.redis.get(cache_key)
        if cached_result:
            print("📦 Résultat récupéré du cache")
            return pickle.loads(cached_result)
        
        # 1. Analyse COBOL
        print("🔍 Analyse du COBOL...")
        cobol_analysis = await self._analyze_cobol(cobol_content)
        
        # 2. Génération Python initiale
        print("🔄 Génération Python...")
        raw_python = await self._generate_initial_python(
            cobol_content, cobol_analysis
        )
        
        # 3. Correction en parallèle (par module/section)
        print("🔧 Correction du code...")
        corrected_modules = await self._parallel_correction(
            raw_python, cobol_analysis
        )
        
        # 4. Assemblage final
        print("🧩 Assemblage final...")
        final_code = self._assemble_final_code(corrected_modules, cobol_analysis)
        
        # 5. Validation complète
        print("✅ Validation...")
        validation_result = await self._validate_final_code(
            final_code, cobol_content, cobol_analysis
        )
        
        result = {
            "python_code": final_code,
            "validation": validation_result,
            "metrics": self._calculate_metrics(cobol_content, final_code),
            "warnings": validation_result.get("warnings", []),
            "corrections_applied": len(corrected_modules)
        }
        
        # Mise en cache (24h)
        self.redis.setex(cache_key, timedelta(hours=24), pickle.dumps(result))
        
        return result
    
    async def _parallel_correction(self, python_code: str, 
                                 cobol_analysis: Dict) -> List[Dict]:
        """Corrige différentes sections en parallèle"""
        
        # Découpe le code en sections logiques
        sections = self._split_into_sections(python_code, cobol_analysis)
        
        # Corrections parallèles
        loop = asyncio.get_event_loop()
        tasks = []
        
        for section in sections:
            task = loop.run_in_executor(
                self.executor,
                self._correct_section,
                section,
                cobol_analysis
            )
            tasks.append(task)
        
        # Attend toutes les corrections
        corrected_sections = await asyncio.gather(*tasks)
        
        return corrected_sections
    
    def _correct_section(self, section: Dict, cobol_analysis: Dict) -> Dict:
        """Corrige une section spécifique"""
        section_code = section["code"]
        
        # Vérifie d'abord le cache local
        cache_key = f"section:{hashlib.md5(section_code.encode()).hexdigest()}"
        if cache_key in self.correction_cache:
            return self.correction_cache[cache_key]
        
        # Recherche des corrections similaires
        similar = self.knowledge_base.find_similar_corrections(
            section.get("cobol_source", "")
        )
        
        # Applique les corrections connues si disponibles
        if similar and similar[0]["similarity"] > 0.9:
            corrected = similar[0]["python_after"]
        else:
            # Correction hiérarchique
            corrected = self.corrector.correct_python(
                section_code,
                {
                    **cobol_analysis,
                    "section_type": section["type"],
                    "original_cobol_snippet": section.get("cobol_source", "")
                }
            )
        
        # Met en cache
        self.correction_cache[cache_key] = {
            **section,
            "corrected_code": corrected,
            "original_hash": cache_key
        }
        
        return self.correction_cache[cache_key]

# monitoring.py
from prometheus_client import Counter, Histogram, Gauge
import time
from datetime import datetime
import logging

# Métriques Prometheus
CORRECTIONS_TOTAL = Counter('codeswitch_corrections_total', 
                           'Total corrections attempted', ['level', 'status'])
CORRECTION_DURATION = Histogram('codeswitch_correction_duration_seconds',
                               'Time spent correcting code', ['level'])
CODE_QUALITY = Gauge('codeswitch_code_quality', 
                    'Quality score of generated code', ['file_type'])

class CodeSwitchMonitor:
    def __init__(self):
        self.logger = logging.getLogger("codeswitch")
        
    def track_correction(self, level: str, duration: float, 
                        success: bool, issues_fixed: int):
        """Track une correction"""
        CORRECTIONS_TOTAL.labels(level=level, 
                               status="success" if success else "failure").inc()
        CORRECTION_DURATION.labels(level=level).observe(duration)
        
        self.logger.info(
            f"Correction {level}: success={success}, "
            f"duration={duration:.2f}s, fixed={issues_fixed}"
        )
    
    def track_conversion(self, cobol_loc: int, python_loc: int, 
                        quality_score: float):
        """Track une conversion complète"""
        ratio = python_loc / max(cobol_loc, 1)
        CODE_QUALITY.labels(file_type="generated").set(quality_score)
        
        metrics = {
            "cobol_loc": cobol_loc,
            "python_loc": python_loc,
            "ratio": ratio,
            "quality_score": quality_score,
            "timestamp": datetime.now().isoformat()
        }
        
        # Log détaillé
        self.logger.info(f"Conversion completed: {metrics}")
        
        return metrics
