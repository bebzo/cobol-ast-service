class ContextAwareGemini:
    def __init__(self):
        self.current_context = {
            "cobol_line": 120,
            "python_line": 494,
            "error_message": "unterminated string literal",
            "file_section": "WORKING-STORAGE",
            "variable_name": "WS-TAX-AMOUNT"
        }
    
    def get_enhanced_response(self, question):
        # Injecte le contexte automatiquement
        prompt = f"""
        CONTEXTE ACTUEL:
        - Ligne COBOL: {self.current_context['cobol_line']}
        - Variable: {self.current_context['variable_name']}
        - Erreur: {self.current_context.get('error_message', 'None')}
        
        QUESTION UTILISATEUR: {question}
        
        Réponds avec:
        1. Référence à la ligne spécifique
        2. Explication technique précise
        3. Solution concrète si applicable
        """
        return gemini.generate(prompt)

**3. Auto-Fix Wizard
python
class AutoFixWizard:
    def __init__(self):
        self.common_fixes = {
            "unterminated_string": self.fix_string_literal,
            "decimal_rounding": self.fix_rounding,
            "cobol_call": self.wrap_external_call,
        }
    
    def suggest_fix(self, error_type, context):
        """Gemini suggère, wizard implémente"""
        suggestion = gemini.suggest_fix(error_type, context)
        
        # Génère le code de correction
        fix_code = self.generate_fix_code(suggestion)
        
        # Propose à l'utilisateur
        return {
            "description": suggestion,
            "code": fix_code,
            "apply_button": f"applyFix({fix_code})",
            "preview_button": f"previewFix({fix_code})"
        }
