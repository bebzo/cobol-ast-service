class DepositService:
    """Service métier pour les dépôts."""
    
    def __init__(
        self,
        customer_repo: CustomerRepository,
        fraud_detector: FraudDetector,
        config: BankingConfig
    ):
        self.customer_repo = customer_repo
        self.fraud_detector = fraud_detector
        self.config = config
    
    def process_deposit(
        self,
        customer_id: str,
        amount: Decimal
    ) -> DepositResult:
        """Traite un dépôt avec toutes validations."""
        
        # 1. Récupérer client
        customer = self.customer_repo.find_by_id(customer_id)
        if customer is None:
            raise CustomerNotFoundError(customer_id)
        
        # 2. Vérifier verrouillage
        if customer.is_locked:
            raise AccountLockedError(customer_id)
        
        # 3. Valider montant
        self._validate_amount(amount)
        
        # 4. Vérifier fraude
        fraud_score = self.fraud_detector.calculate_score(
            customer, amount, TransactionType.DEPOSIT
        )
        if fraud_score > self.config.fraud_threshold:
            raise InvalidTransactionError(f"Fraud suspected: {fraud_score}")
        
        # 5. Calculer montant net (après taxes)
        tax = amount * self.config.tax_rate
        net_amount = amount - tax
        
        # 6. Mettre à jour soldes
        customer.account_balance += net_amount
        customer.available_balance += net_amount
        
        # 7. Sauvegarder
        self.customer_repo.save(customer)
        
        return DepositResult(
            success=True,
            new_balance=customer.account_balance,
            tax_amount=tax,
            net_amount=net_amount
        )
    
    def _validate_amount(self, amount: Decimal) -> None:
        if amount <= 0:
            raise InvalidTransactionError("Amount must be positive")
        if amount > self.config.max_single_transaction:
            raise InvalidTransactionError("Exceeds transaction limit")
