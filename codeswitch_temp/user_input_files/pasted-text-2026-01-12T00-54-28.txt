# Architecture Clean Architecture
from abc import ABC, abstractmethod

class DomainEntity(ABC):
    """Entités métier pures"""
    
class LoanAccount(DomainEntity):
    def __init__(self, principal: Decimal, rate: Decimal, term_months: int):
        self.principal = principal
        self.rate = rate
        self.term_months = term_months
    
    def calculate_monthly_payment(self) -> Decimal:
        """Formule d'amortissement réelle"""
        monthly_rate = self.rate / 12
        return self.principal * (monthly_rate * (1 + monthly_rate)**self.term_months) / ((1 + monthly_rate)**self.term_months - 1)

class FeeRepository(ABC):
    @abstractmethod
    def get_fee_schedule(self, account_type: str) -> Dict[str, Decimal]:
        pass

# Wrapper COBOL → Python
class COBOLCompatibilityWrapper:
    """Permet migration progressive"""
    
    def __init__(self, python_service, cobol_fallback):
        self.python_service = python_service
        self.cobol_fallback = cobol_fallback
        
    def calculate_interest(self, account_id: str) -> Decimal:
        try:
            # Essayer Python d'abord
            return self.python_service.calculate_interest(account_id)
        except NotImplementedError:
            # Fallback COBOL temporaire
            return self.cobol_fallback.calculate_interest(account_id)

# Service métier complet
class BankingService:
    def __init__(self, account_repo: AccountRepository, 
                 fee_repo: FeeRepository,
                 audit_service: AuditService):
        self.account_repo = account_repo
        self.fee_repo = fee_repo
        self.audit_service = audit_service
    
    def process_withdrawal(self, account_id: str, amount: Decimal) -> TransactionResult:
        """Logique métier complète"""
        account = self.account_repo.get(account_id)
        
        # Validation métier
        if account.balance - amount < account.minimum_balance:
            fee = self.fee_repo.get_overdraft_fee()
            raise InsufficientFundsException(f"Frais de ${fee} appliqués")
        
        # Traitement
        account.balance -= amount
        self.account_repo.update(account)
        
        # Audit
        self.audit_service.log_transaction(account_id, -amount, "WITHDRAWAL")
        
        return TransactionResult(success=True, new_balance=account.balance)
