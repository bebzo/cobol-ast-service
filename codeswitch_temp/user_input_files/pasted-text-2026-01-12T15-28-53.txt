"""Calculinterets - Clean Architecture Python Code
Version corrigée et optimisée

from __future__ import annotations
from decimal import Decimal, ROUND_HALF_UP
from dataclasses import dataclass
import logging
from typing import Optional

# ========== DOMAIN MODELS ==========
@dataclass
class InterestCalculation:
    """Objet métier pour le calcul d'intérêts"""
    initial_balance: Decimal
    annual_rate: Decimal
    period_days: int
    calculated_interest: Decimal = Decimal('0')
    final_balance: Decimal = Decimal('0')
    
    # Constante métier
    DAYS_IN_YEAR: int = 365
    
    def calculate(self) -> None:
        """Calcule les intérêts proportionnels aux jours"""
        if self.period_days <= 0:
            raise ValueError("La période doit être positive")
        
        daily_rate = self.annual_rate / Decimal(self.DAYS_IN_YEAR)
        self.calculated_interest = (
            self.initial_balance * daily_rate * Decimal(self.period_days)
        ).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)
        
        self.final_balance = (
            self.initial_balance + self.calculated_interest
        ).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP)

# ========== SERVICES ==========
class InterestCalculator:
    """Service de calcul d'intérêts"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
    
    def calculate_interest(
        self,
        balance: Decimal,
        rate: Decimal,
        days: int
    ) -> InterestCalculation:
        """Exécute le calcul et retourne l'objet métier"""
        calculation = InterestCalculation(
            initial_balance=balance,
            annual_rate=rate,
            period_days=days
        )
        
        calculation.calculate()
        self.logger.info(f"Intérêts calculés: {calculation.calculated_interest}")
        
        return calculation

class ReportGenerator:
    """Génère les rapports d'affichage"""
    
    @staticmethod
    def format_amount(amount: Decimal) -> str:
        """Formate un montant avec séparateurs"""
        return f"{amount:,.2f}"
    
    @staticmethod
    def format_percentage(rate: Decimal) -> str:
        """Formate un taux en pourcentage"""
        return f"{rate * 100:.3f}%"
    
    def generate(self, calculation: InterestCalculation) -> str:
        """Génère le rapport complet"""
        lines = [
            "=" * 33,
            "CALCUL D'INTERETS BANCAIRES",
            "=" * 33,
            f"Solde initial: {self.format_amount(calculation.initial_balance)} €",
            f"Taux annuel: {self.format_percentage(calculation.annual_rate)}",
            f"Période: {calculation.period_days} jours",
            "-" * 33,
            f"Intérêts calculés: {self.format_amount(calculation.calculated_interest)} €",
            f"Nouveau solde: {self.format_amount(calculation.final_balance)} €",
            "=" * 33
        ]
        return "\n".join(lines)
