pythondef p_310_process_deposit(self) -> None:
    """Process deposit with exception-based error handling."""
    try:
        self.cm_customer_id = self.ls_from_account
        _record = self.file_manager.read_record('customer_master_file')
        
        if _record is None:
            raise CustomerNotFoundError(str(self.cm_customer_id))
        
        self.customer_master_file_record = _record
        
        if self.account_locked:
            raise AccountLockedError(str(self.cm_customer_id))
        
        self.p_311_validate_deposit()
        
        if not self.data_valid:
            raise InvalidTransactionError(self.error_message)
        
        self.p_312_execute_deposit()
        self.p_313_update_record()
        
    except CobolBusinessError as e:
        self.ls_return_code = e.error_code
        self.ls_error_msg = str(e)
        self.log_error()
        # Mode compat COBOL : pas de re-raise
