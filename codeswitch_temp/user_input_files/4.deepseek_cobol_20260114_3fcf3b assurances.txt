       DECLARATIVES.

       POLICE-FILE-ERROR SECTION.
           USE AFTER ERROR PROCEDURE ON POLICE-FILE.
       POL-ERR-PROC.
           EVALUATE WS-POL-STATUS
               WHEN '10'
                   MOVE 'EOF POLICE FILE' TO WS-ERROR-MESSAGE
               WHEN '22'
                   MOVE 'DUPLICATE KEY' TO WS-ERROR-MESSAGE
                   PERFORM LOG-AUDIT-ERROR
               WHEN '23'
                   MOVE 'KEY NOT FOUND' TO WS-ERROR-MESSAGE
               WHEN '24'
                   MOVE 'INVALID KEY' TO WS-ERROR-MESSAGE
                   PERFORM LOG-SECURITY-EVENT
               WHEN '30'
                   MOVE 'PERMANENT ERROR' TO WS-ERROR-MESSAGE
                   MOVE 9999 TO WS-RETURN-CODE
               WHEN OTHER
                   MOVE 'UNKNOWN FILE ERROR' TO WS-ERROR-MESSAGE
           END-EVALUATE.
           MOVE 'POL' TO WS-ERROR-CODE.

       SINISTRE-FILE-ERROR SECTION.
           USE AFTER ERROR PROCEDURE ON SINISTRE-FILE.
       SIN-ERR-PROC.
           EVALUATE WS-SIN-STATUS
               WHEN '10'
                   CONTINUE
               WHEN '23'
                   MOVE 'RELATIVE KEY ERROR' TO WS-ERROR-MESSAGE
               WHEN OTHER
                   MOVE 'SINISTRE FILE ERROR' TO WS-ERROR-MESSAGE
           END-EVALUATE.

       END DECLARATIVES.

       MAIN-PROGRAM SECTION.
       000-MAIN-CONTROL.
           PERFORM 100-INITIALIZATION
           PERFORM 200-AUTHENTICATION
           IF WS-AUTH-LEVEL > 0
               PERFORM 300-PROCESS-REQUEST
               PERFORM 400-CALCULATIONS
               PERFORM 500-UPDATE-DATABASE
               PERFORM 600-GENERATE-REPORTS
           ELSE
               PERFORM 800-SECURITY-FAILURE
           END-IF
           PERFORM 900-TERMINATION
           GOBACK.

       100-INITIALIZATION.
           MOVE FUNCTION CURRENT-DATE TO WS-DATE-COURANTE
           MOVE FUNCTION TIME-OF-DAY TO WS-HEURE-COURANTE
           MOVE FUNCTION WHEN-COMPILED TO WS-TIMESTAMP
           
           OPEN I-O POLICE-FILE
           OPEN I-O SINISTRE-FILE
           OPEN INPUT TARIFICATION-FILE
           OPEN OUTPUT HISTORIQUE-FILE
           OPEN I-O REASSURANCE-FILE
           OPEN OUTPUT TEMP-WORK-FILE
           
           IF NOT POL-OK OR NOT WS-TAR-STATUS = '00'
               MOVE 9001 TO WS-RETURN-CODE
               PERFORM 900-TERMINATION
               GOBACK
           END-IF
           
           PERFORM 110-LOAD-TARIFF-TABLES
           PERFORM 120-LOAD-REASSURANCE-TABLES
           PERFORM 130-INIT-STATISTICS
           MOVE 'N' TO WS-POL-EOF
           MOVE 'V' TO WS-VALID-POLICE.

       110-LOAD-TARIFF-TABLES.
           MOVE 1 TO WS-I
           MOVE '001' TO TAR-CLASS
           START TARIFICATION-FILE KEY >= TAR-CLASS
           PERFORM UNTIL WS-TAR-EOF = 'Y' OR WS-I > 100
               READ TARIFICATION-FILE NEXT
                   AT END MOVE 'Y' TO WS-TAR-EOF
                   NOT AT END
                       MOVE TAR-CLASS TO WS-TAR-CLASS(WS-I)
                       MOVE TAR-ZONE TO WS-TAR-ZONE(WS-I)
                       MOVE TAR-BASE TO WS-TAR-BASE(WS-I)
                       PERFORM VARYING WS-J FROM 1 BY 1
                               UNTIL WS-J > 5
                           MOVE TAR-MAJ-CAUSE(WS-J)
                             TO WS-MAJ-CAUSE(WS-I, WS-J)
                           MOVE TAR-MAJ-TAUX(WS-J)
                             TO WS-MAJ-TAUX(WS-I, WS-J)
                       END-PERFORM
                       PERFORM VARYING WS-J FROM 1 BY 1
                               UNTIL WS-J > 3
                           MOVE TAR-RED-CAUSE(WS-J)
                             TO WS-RED-CAUSE(WS-I, WS-J)
                           MOVE TAR-RED-TAUX(WS-J)
                             TO WS-RED-TAUX(WS-I, WS-J)
                       END-PERFORM
                       ADD 1 TO WS-I
               END-READ
           END-PERFORM.

       120-LOAD-REASSURANCE-TABLES.
           MOVE 1 TO WS-I
           MOVE LOW-VALUES TO REA-COMPAGNIE
           START REASSURANCE-FILE KEY >= REA-COMPAGNIE
           PERFORM UNTIL WS-REA-STATUS NOT = '00' OR WS-I > 50
               READ REASSURANCE-FILE NEXT
                   AT END CONTINUE
                   NOT AT END
                       MOVE REA-COMPAGNIE
                         TO WS-REA-COMPAGNIE(WS-I)
                       MOVE REA-CAPACITE
                         TO WS-REA-CAPACITE(WS-I)
                       MOVE REA-TAUX TO WS-REA-TAUX(WS-I)
                       MOVE ZERO TO WS-REA-UTILISE(WS-I)
                       ADD 1 TO WS-I
               END-READ
           END-PERFORM.

       130-INIT-STATISTICS.
           MOVE ZERO TO WS-STAT-POLICES
                        WS-STAT-SINISTRES
                        WS-STAT-MONTANT-SINISTRES
                        WS-STAT-FREQUENCE
                        WS-STAT-COUT-MOYEN
                        WS-STAT-SEVERITE
                        WS-STAT-LOSS-RATIO.

       200-AUTHENTICATION.
           MOVE LS-USER-DATA(1:20) TO WS-USER-ID
           MOVE LS-USER-DATA(21:15) TO WS-IP-ADDRESS
           
           CALL 'AUTH-SYSTEM' USING WS-USER-ID
                                    WS-IP-ADDRESS
                                    WS-AUTH-LEVEL
                                    WS-SESSION-ID
           
           EVALUATE WS-AUTH-LEVEL
               WHEN 1 MOVE 'LECTURE' TO WS-BUFFER-AUDIT
               WHEN 2 MOVE 'ECRITURE' TO WS-BUFFER-AUDIT
               WHEN 3 MOVE 'ADMIN' TO WS-BUFFER-AUDIT
               WHEN OTHER
                   MOVE 0 TO WS-AUTH-LEVEL
                   MOVE 'AUTH FAILED' TO WS-ERROR-MESSAGE
           END-EVALUATE
           
           PERFORM LOG-AUDIT-ENTRY
               USING 'AUTHENTICATION' WS-USER-ID WS-AUTH-LEVEL.

       300-PROCESS-REQUEST.
           EVALUATE LS-FUNCTION-CODE
               WHEN 'NEWP' PERFORM 310-NEW-POLICY
               WHEN 'UPDT' PERFORM 320-UPDATE-POLICY
               WHEN 'SINR' PERFORM 330-NEW-CLAIM
               WHEN 'SINU' PERFORM 340-UPDATE-CLAIM
               WHEN 'CALC' PERFORM 350-CALCULATE-PREMIUM
               WHEN 'RAPP' PERFORM 360-GENERATE-REPORT
               WHEN 'RISK' PERFORM 370-RISK-ANALYSIS
               WHEN 'REAS' PERFORM 380-REINSURANCE-PROCESS
               WHEN 'COMP' PERFORM 390-COMPLEX-CALCULATION
               WHEN OTHER
                   MOVE 9002 TO WS-RETURN-CODE
                   MOVE 'INVALID FUNCTION' TO WS-ERROR-MESSAGE
           END-EVALUATE.

       310-NEW-POLICY.
           PERFORM 311-VALIDATE-POLICY-DATA
           IF POL-VALIDE
               PERFORM 312-CALCULATE-INITIAL-PREMIUM
               PERFORM 313-ASSESS-RISK
               PERFORM 314-APPLY-TARIFF
               PERFORM 315-CREATE-POLICY-RECORD
               PERFORM 316-UPDATE-STATISTICS
               PERFORM 317-GENERATE-CONTRACT
               MOVE 0 TO WS-RETURN-CODE
               MOVE 'POLICY CREATED' TO WS-ERROR-MESSAGE
           ELSE
               MOVE 9003 TO WS-RETURN-CODE
           END-IF.

       311-VALIDATE-POLICY-DATA.
           MOVE 'V' TO WS-VALID-POLICE
           MOVE LS-DATA-CONTENT(1:300) TO WS-FORM-POLICE
           
           IF WS-FORM-CLIENT = ZERO
               MOVE 'I' TO WS-VALID-POLICE
               MOVE 'INVALID CLIENT' TO WS-ERROR-MESSAGE
           END-IF
           
           IF WS-FORM-GARANTIE < 1000.00
               MOVE 'I' TO WS-VALID-POLICE
               MOVE 'GUARANTEE TOO LOW' TO WS-ERROR-MESSAGE
           END-IF
           
           IF WS-FORM-EFFET < WS-DATE-COMPACT
               MOVE 'I' TO WS-VALID-POLICE
               MOVE 'INVALID EFFECTIVE DATE' TO WS-ERROR-MESSAGE
           END-IF
           
           PERFORM CHECK-FRAUD-INDICATORS
           IF FRAUD-DETECTED
               MOVE 'I' TO WS-VALID-POLICE
               MOVE 'FRAUD SUSPECTED' TO WS-ERROR-MESSAGE
           END-IF.

       312-CALCULATE-INITIAL-PREMIUM.
           COMPUTE WS-PRIME-PURE =
               WS-FORM-GARANTIE * WS-TAUX-BASE / 100
           
           COMPUTE WS-TAUX-SINISTRALITE =
               (WS-STAT-MONTANT-SINISTRES / WS-STAT-POLICES)
               / WS-FORM-GARANTIE
           
           COMPUTE WS-PRIME-COMMERCIALE =
               WS-PRIME-PURE * (1 + WS-TAUX-SINISTRALITE
                              + WS-TAUX-TECHNIQUE
                              + WS-MARGIN-SECURITE)
           
           COMPUTE WS-PRIME-NETTE =
               WS-PRIME-COMMERCIALE * (1 - WS-CHARGES-FIXES)
           
           COMPUTE WS-PRIME-TOTALE = WS-PRIME-NETTE
           
           PERFORM APPLY-MAJORATIONS-REDUCTIONS
           PERFORM CALCULATE-VARIANCE.

       313-ASSESS-RISK.
           MOVE 50 TO POL-SCORE-RISQUE
           
           EVALUATE WS-FORM-BRANCHE
               WHEN 01 ADD 10 TO POL-SCORE-RISQUE  *> Auto
               WHEN 02 ADD 20 TO POL-SCORE-RISQUE  *> Santé
               WHEN 03 ADD 15 TO POL-SCORE-RISQUE  *> Habitation
               WHEN 04 ADD 30 TO POL-SCORE-RISQUE  *> Professionnelle
               WHEN OTHER ADD 25 TO POL-SCORE-RISQUE
           END-EVALUATE
           
           IF WS-FORM-GARANTIE > 1000000.00
               ADD 20 TO POL-SCORE-RISQUE
           END-IF
           
           EVALUATE TRUE
               WHEN POL-SCORE-RISQUE < 50
                   MOVE 'F' TO POL-CLASS-RISQUE
               WHEN POL-SCORE-RISQUE < 75
                   MOVE 'M' TO POL-CLASS-RISQUE
               WHEN POL-SCORE-RISQUE < 90
                   MOVE 'E' TO POL-CLASS-RISQUE
               WHEN OTHER
                   MOVE 'C' TO POL-CLASS-RISQUE
           END-EVALUATE.

       314-APPLY-TARIFF.
           PERFORM VARYING IDX-TARIF FROM 1 BY 1
                   UNTIL IDX-TARIF > 100
               IF WS-TAR-CLASS(IDX-TARIF) =
                  WS-FORM-BRANCHE OF WS-FORM-POLICE
                   MOVE WS-TAR-BASE(IDX-TARIF)
                     TO WS-TAUX-BASE
                   PERFORM VARYING IDX-MAJ FROM 1 BY 1
                           UNTIL IDX-MAJ > 5
                       IF WS-MAJ-CAUSE(IDX-TARIF, IDX-MAJ) =
                          POL-CLASS-RISQUE
                           COMPUTE WS-PRIME-TOTALE =
                               WS-PRIME-TOTALE *
                               (1 + WS-MAJ-TAUX(IDX-TARIF, IDX-MAJ))
                       END-IF
                   END-PERFORM
                   PERFORM VARYING IDX-RED FROM 1 BY 1
                           UNTIL IDX-RED > 3
                       IF WS-RED-CAUSE(IDX-TARIF, IDX-RED) =
                          'LOYAL'
                           COMPUTE WS-PRIME-TOTALE =
                               WS-PRIME-TOTALE *
                               (1 - WS-RED-TAUX(IDX-TARIF, IDX-RED))
                       END-IF
                   END-PERFORM
               END-IF
           END-PERFORM.

       315-CREATE-POLICY-RECORD.
           MOVE '001' TO POL-COMPAGNIE
           MOVE WS-FORM-BRANCHE TO POL-BRANCHE
           MOVE FUNCTION RANDOM(999999999999) TO POL-NUMERO
           MOVE WS-FORM-CLIENT TO POL-NUM-CLIENT
           MOVE WS-FORM-CONTRAT TO POL-NUM-C
