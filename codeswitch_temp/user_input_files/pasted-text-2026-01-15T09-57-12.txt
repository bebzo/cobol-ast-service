    def add_with_overflow_check(a: Decimal, b: Decimal, pic_length: int, pic_decimal_places: int) -> Decimal:
        """Adds two Decimal numbers and checks for overflow based on PIC length and decimal places."""
        result = a + b
        max_value = Decimal('9' * pic_length + '.' + '9' * pic_decimal_places)
        min_value = Decimal('-' + '9' * pic_length + '.' + '9' * pic_decimal_places)

        if result > max_value:
            raise OverflowError(f"Overflow: Result {result} exceeds maximum value {max_value}")
        if result < min_value:
