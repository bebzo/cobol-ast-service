def _parse_customer_record(self, record: str) -> None:
    """Parse un enregistrement COBOL de 512 caractères"""
    if len(record) >= 512:
        # CM-CUSTOMER-ID PIC 9(10) : positions 1-10
        self.cm_customer_id = record[0:10].strip()
        
        # CM-ACCOUNT-BALANCE PIC S9(9)V99 : positions ?-?
        # À calculer selon la structure COBOL
        balance_str = record[100:112]  # Exemple
        try:
            self.cm_account_balance = Decimal(balance_str)
        except:
            self.cm_account_balance = Decimal('0')
