from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from mega_bank.domain.repositories.abstract import AbstractAccountRepository
from mega_bank.domain.entities.account import Account

class SQLAccountRepository(AbstractAccountRepository):
    def __init__(self, session: AsyncSession) -> None:
        self.session = session

    async def get(self, acct_id: str) -> Account | None:
        return await self.session.get(Account, acct_id)

    async def save(self, account: Account) -> None:
        self.session.add(account)
        await self.session.flush()

    async def list_by_customer(self, cust_id: str) -> list[Account]:
        stmt = select(Account).where(Account.cust_id == cust_id)
        res = await self.session.execute(stmt)
        return res.scalars().all()

from decimal import Decimal, ROUND_HALF_UP
from mega_bank.domain.entities.account import Account, AccountType

RATES = {
    AccountType.CHECKING: Decimal("0.0050"),
    AccountType.SAVINGS:  Decimal("0.0225"),
    AccountType.MONEY_MARKET: Decimal("0.0350"),
    AccountType.CD:       Decimal("0.0425"),
}

def calculate_daily_interest(account: Account) -> Decimal:
    rate = RATES.get(account.acct_type, Decimal("0"))
    daily = account.balance * rate / Decimal("360")
    return daily.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession
from mega_bank.infrastructure.db import get_session
from mega_bank.domain.repositories.sqlalchemy import SQLAccountRepository
from mega_bank.domain.services.interest import calculate_daily_interest
from mega_bank.domain.schemas import AccountResponse

router = APIRouter()

@router.post("/{acct_id}/credit-interest", response_model=AccountResponse)
async def credit_interest(
    acct_id: str,
    db: AsyncSession = Depends(get_session),
):
    repo = SQLAccountRepository(db)
    account = await repo.get(acct_id)
    if not account:
        raise HTTPException(404, "Account not found")
    interest = calculate_daily_interest(account)
    account.balance += interest
    await repo.save(account)
    await db.commit()
    return AccountResponse.from_orm(account)
