import aiofiles

class AccountFileGateway:
    def __init__(self, path: str):
        self.path = path
    async def read_next(self) -> Account | None:
        async with aiofiles.open(self.path, "r") as f:
            line = await f.readline()
            return Account.from_cobol_line(line) if line else None
    async def rewrite(self, account: Account) -> None:

# infrastructure/gateways/account_file_gateway.py
import aiofiles
from pathlib import Path
from mega_bank.domain.entities.account import Account, AccountType
from decimal import Decimal

class AccountFileGateway:
    def __init__(self, file_path: Path):
        self.file_path = file_path
        self._fd: aiofiles.threadpool.AsyncTextIOWrapper | None = None

    async def __aenter__(self):
        self._fd = await aiofiles.open(self.file_path, mode="r")
        return self

    async def __aexit__(self, exc_type, exc, tb):
        if self._fd:
            await self._fd.close()

    async def read_next(self) -> Account | None:
        line = await self._fd.readline()  # type: ignore
        if not line:
            return None
        # COBOL layout: ACCT-ID(1-16), ACCT-TYPE(17-18), BALANCE(19-31), ...
        return Account(
            acct_id=line[0:16].strip(),
            acct_type=AccountType(line[16:18].strip()),
            balance=Decimal(line[18:31].strip()) / 100,
            open_date=datetime.strptime(line[31:39], "%Y%m%d").date(),
            status=line[39],
        )

# extract_and_replace_todos.py
import re, pathlib, vertexai
from vertexai.preview.language_models import TextGenerationModel

model = TextGenerationModel.from_pretrained("text-bison@001")
FILE = "mega_enterprise_clean.py"
BACKUP = FILE + ".bak"

code = pathlib.Path(FILE).read_text()
todo_pattern = re.compile(r"self\.logger\.debug\(r?'TODO: (.+?)'\)")

def gemini_replace(todo: str) -> str:
    prompt = f"""
Replace this COBOL-style TODO by modern **async Python code** (aiofiles, Decimal, no Y/N).
Keep business logic identical. Output **only** the new code (no comments).
TODO: {todo}
"""
    return model.predict(prompt, temperature=0.0).text.strip()

new_code = todo_pattern.sub(lambda m: gemini_replace(m.group(1)), code)
pathlib.Path(FILE).write_text(new_code)
print("✅ All TODOs replaced by Gemini-generated async code.")
