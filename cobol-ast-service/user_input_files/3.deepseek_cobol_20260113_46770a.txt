       PROCEDURE DIVISION 
           USING LS-CONTROL-BLOCK
                 LS-TRANSACTION-DATA.

       DECLARATIVES.

       CUSTOMER-FILE-ERROR SECTION.
           USE AFTER ERROR PROCEDURE ON CUSTOMER-MASTER-FILE.
       CUST-ERR-PROC.
           DISPLAY 'CUSTOMER FILE ERROR: ' CM-STATUS
           MOVE 'CFER' TO WS-ERROR-CODE
           PERFORM LOG-ERROR
           IF CM-STATUS = '30' OR '35'
               MOVE 'CRITICAL' TO WS-ERROR-SEVERITY
               MOVE 9999 TO LS-RETURN-CODE
           ELSE
               MOVE 'ERROR' TO WS-ERROR-SEVERITY
               MOVE 9000 TO LS-RETURN-CODE.

       TRANSACTION-FILE-ERROR SECTION.
           USE AFTER ERROR PROCEDURE ON TRANSACTION-FILE.
       TRANS-ERR-PROC.
           DISPLAY 'TRANSACTION FILE ERROR: ' TRANS-STATUS
           MOVE 'TFER' TO WS-ERROR-CODE
           PERFORM LOG-ERROR.

       END DECLARATIVES.

       MAIN-PROCEDURE SECTION.
       000-MAIN.
           PERFORM 100-INITIALIZE
           PERFORM 200-AUTHENTICATE-USER
           IF WS-SECURITY-PASS
               PERFORM 300-PROCESS-COMMAND
               PERFORM 400-CALCULATE-INTEREST
               PERFORM 500-APPLY-FEES-AND-TAXES
               PERFORM 600-GENERATE-REPORTS
               PERFORM 700-UPDATE-AUDIT-TRAIL
           ELSE
               PERFORM 800-HANDLE-SECURITY-FAILURE
           END-IF
           PERFORM 900-TERMINATE
           GOBACK.

       100-INITIALIZE.
           MOVE FUNCTION CURRENT-DATE TO WS-CURRENT-DATE
           MOVE FUNCTION TIME-OF-DAY TO WS-CURRENT-TIME
           MOVE FUNCTION WHEN-COMPILED TO WS-JOB-ID
           
           OPEN I-O CUSTOMER-MASTER-FILE
           OPEN INPUT TRANSACTION-FILE
           OPEN OUTPUT AUDIT-TRAIL-FILE
           OPEN OUTPUT TEMPORARY-WORK-FILE
           
           IF NOT CM-SUCCESS
               MOVE 'FILE OPEN FAILED' TO LS-ERROR-MSG
               MOVE 9001 TO LS-RETURN-CODE
               PERFORM 900-TERMINATE
               GOBACK
           END-IF
           
           PERFORM 110-LOAD-CONFIGURATION
           PERFORM 120-INITIALIZE-SECURITY
           PERFORM 130-SETUP-REPORTING.

       110-LOAD-CONFIGURATION.
           PERFORM VARYING RATE-INDEX FROM 1 BY 1
                     UNTIL RATE-INDEX > 5
               DISPLAY 'LOADED RATE: ' WS-ACCOUNT-CODE(RATE-INDEX)
                         ' ' WS-BASE-RATE(RATE-INDEX)
           END-PERFORM.

       120-INITIALIZE-SECURITY.
           MOVE 'SECRET_KEY_1234567890ABCDEF' TO WS-ENCRYPTION-KEY
           CALL 'GENSESSION' USING WS-SESSION-TOKEN
           MOVE 2 TO WS-AUTH-LEVEL.

       130-SETUP-REPORTING.
           MOVE 'ULTIMATE BANKING SYSTEM REPORT' TO WS-REPORT-TITLE
           MOVE WS-CURRENT-DATE TO WS-REPORT-DATE
           MOVE WS-CURRENT-TIME TO WS-REPORT-TIME.

       200-AUTHENTICATE-USER.
           IF LS-USER-ID = SPACES
               MOVE 'F' TO WS-SECURITY-FLAG
               MOVE 'NO USER ID' TO WS-ERROR-MESSAGE
               PERFORM LOG-SECURITY-EVENT
               EXIT PARAGRAPH
           END-IF
           
           CALL 'AUTHMODULE' USING LS-USER-ID
                                 WS-SESSION-TOKEN
                                 WS-AUTH-LEVEL
                                 WS-SECURITY-FLAG
           
           IF WS-SECURITY-PASS
               PERFORM 210-CHECK-IP-AUTHORIZATION
               PERFORM 220-VALIDATE-SESSION
           END-IF.

       210-CHECK-IP-AUTHORIZATION.
           MOVE 'F' TO WS-SECURITY-FLAG
           PERFORM VARYING WS-INDEX FROM 1 BY 1
                     UNTIL WS-INDEX > WS-IP-WHITELIST-COUNT
               IF WS-IP-WHITELIST(WS-INDEX) = TRANS-IP-ADDRESS
                   MOVE 'P' TO WS-SECURITY-FLAG
                   EXIT PERFORM
               END-IF
           END-PERFORM.

       220-VALIDATE-SESSION.
           CALL 'VALIDATESESSION' USING LS-SESSION-ID
                                      WS-CURRENT-DATE
                                      WS-CURRENT-TIME
                                      WS-SECURITY-FLAG.

       300-PROCESS-COMMAND.
           EVALUATE LS-COMMAND
               WHEN 'DEPOSIT  '
                   PERFORM 310-PROCESS-DEPOSIT
               WHEN 'WITHDRAW '
                   PERFORM 320-PROCESS-WITHDRAWAL
               WHEN 'TRANSFER '
                   PERFORM 330-PROCESS-TRANSFER
               WHEN 'BALANCE  '
                   PERFORM 340-CHECK-BALANCE
               WHEN 'HISTORY  '
                   PERFORM 350-GET-HISTORY
               WHEN 'CALCINT  '
                   PERFORM 360-CALCULATE-INTEREST-ADV
               WHEN OTHER
                   MOVE 'INVALID COMMAND' TO LS-ERROR-MSG
                   MOVE 9002 TO LS-RETURN-CODE
           END-EVALUATE.

       310-PROCESS-DEPOSIT.
           MOVE LS-FROM-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
               KEY IS CM-CUSTOMER-ID
               INVALID KEY
                   MOVE 'ACCOUNT NOT FOUND' TO LS-ERROR-MSG
                   MOVE 9003 TO LS-RETURN-CODE
               NOT INVALID KEY
                   IF ACCOUNT-LOCKED
                       MOVE 'ACCOUNT LOCKED' TO LS-ERROR-MSG
                       MOVE 9004 TO LS-RETURN-CODE
                   ELSE
                       PERFORM 311-VALIDATE-DEPOSIT
                       IF WS-DATA-VALID
                           PERFORM 312-EXECUTE-DEPOSIT
                           PERFORM 313-UPDATE-RECORD
                       END-IF
                   END-IF
           END-READ.

       311-VALIDATE-DEPOSIT.
           MOVE 'Y' TO WS-VALIDATION-FLAG
           
           IF LS-AMOUNT <= 0
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'INVALID AMOUNT' TO WS-ERROR-MESSAGE
           END-IF
           
           IF LS-AMOUNT > WS-MAX-SINGLE-TRANSACTION
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'EXCEEDS LIMIT' TO WS-ERROR-MESSAGE
           END-IF
           
           PERFORM CALCULATE-FRAUD-SCORE
           IF WS-FRAUD-SCORE > WS-FRAUD-THRESHOLD
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'FRAUD SUSPECTED' TO WS-ERROR-MESSAGE
           END-IF.

       312-EXECUTE-DEPOSIT.
           COMPUTE WS-NEW-BALANCE = 
               CM-ACCOUNT-BALANCE + LS-AMOUNT
           
           COMPUTE WS-TAX-AMOUNT = 
               LS-AMOUNT * WS-TAX-RATE
           
           COMPUTE WS-NET-AMOUNT = 
               LS-AMOUNT - WS-TAX-AMOUNT
           
           ADD WS-NET-AMOUNT TO CM-ACCOUNT-BALANCE
           ADD WS-NET-AMOUNT TO CM-AVAILABLE-BALANCE
           
           ADD 1 TO WS-TRANSACTION-COUNT
           ADD LS-AMOUNT TO WS-TOTAL-AMOUNT
           
           MOVE 'DEP' TO TRANS-TYPE
           MOVE LS-AMOUNT TO TRANS-AMOUNT
           MOVE 'C' TO TRANS-STATUS.

       313-UPDATE-RECORD.
           MOVE WS-CURRENT-DATE TO CM-LAST-INTEREST-DATE
           REWRITE CUSTOMER-RECORD
           
           IF NOT CM-SUCCESS
               MOVE 'UPDATE FAILED' TO LS-ERROR-MSG
               MOVE 9005 TO LS-RETURN-CODE
           ELSE
               MOVE 'SUCCESS' TO LS-ERROR-MSG
               MOVE 0 TO LS-RETURN-CODE
               PERFORM LOG-TRANSACTION
           END-IF.

       320-PROCESS-WITHDRAWAL.
           MOVE LS-FROM-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
               KEY IS CM-CUSTOMER-ID
               INVALID KEY
                   MOVE 'ACCOUNT NOT FOUND' TO LS-ERROR-MSG
                   MOVE 9006 TO LS-RETURN-CODE
               NOT INVALID KEY
                   PERFORM 321-VALIDATE-WITHDRAWAL
                   IF WS-DATA-VALID
                       PERFORM 322-EXECUTE-WITHDRAWAL
                       PERFORM 323-UPDATE-RECORD
                   END-IF
           END-READ.

       321-VALIDATE-WITHDRAWAL.
           MOVE 'Y' TO WS-VALIDATION-FLAG
           
           IF LS-AMOUNT <= 0
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'INVALID AMOUNT' TO WS-ERROR-MESSAGE
           END-IF
           
           IF LS-AMOUNT > CM-AVAILABLE-BALANCE
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'INSUFFICIENT FUNDS' TO WS-ERROR-MESSAGE
           END-IF
           
           IF LS-AMOUNT > WS-DAILY-LIMIT
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'DAILY LIMIT EXCEEDED' TO WS-ERROR-MESSAGE
           END-IF
           
           COMPUTE WS-DAILY-TOTAL = WS-DAILY-TOTAL + LS-AMOUNT
           IF WS-DAILY-TOTAL > WS-DAILY-LIMIT
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'DAILY TOTAL EXCEEDED' TO WS-ERROR-MESSAGE
           END-IF
           
           PERFORM CALCULATE-FRAUD-SCORE
           IF WS-FRAUD-SCORE > WS-FRAUD-THRESHOLD
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'FRAUD SUSPECTED' TO WS-ERROR-MESSAGE
           END-IF.

       322-EXECUTE-WITHDRAWAL.
           COMPUTE WS-NEW-BALANCE = 
               CM-ACCOUNT-BALANCE - LS-AMOUNT
           
           COMPUTE WS-FEE-AMOUNT = 
               LS-AMOUNT * 0.015
           IF WS-FEE-AMOUNT < 5.00
               MOVE 5.00 TO WS-FEE-AMOUNT
           END-IF
           
           COMPUTE WS-NET-AMOUNT = 
               LS-AMOUNT + WS-FEE-AMOUNT
           
           SUBTRACT WS-NET-AMOUNT FROM CM-ACCOUNT-BALANCE
           SUBTRACT WS-NET-AMOUNT FROM CM-AVAILABLE-BALANCE
           
           ADD 1 TO WS-TRANSACTION-COUNT
           SUBTRACT LS-AMOUNT FROM WS-TOTAL-AMOUNT
           
           MOVE 'WDR' TO TRANS-TYPE
           MOVE LS-AMOUNT TO TRANS-AMOUNT
           MOVE 'C' TO TRANS-STATUS.

       330-PROCESS-TRANSFER.
           MOVE LS-FROM-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
               KEY IS CM-CUSTOMER-ID
               INVALID KEY
                   MOVE 'SOURCE ACCOUNT NOT FOUND' TO LS-ERROR-MSG
                   MOVE 9007 TO LS-RETURN-CODE
               NOT INVALID KEY
                   PERFORM 331-VALIDATE-TRANSFER
                   IF WS-DATA-VALID
                       PERFORM 332-DEBIT-SOURCE
                       PERFORM 333-CREDIT-TARGET
                       PERFORM 334-UPDATE-BOTH
                   END-IF
           END-READ.

       331-VALIDATE-TRANSFER.
           MOVE 'Y' TO WS-VALIDATION-FLAG
           
           IF LS-AMOUNT <= 0
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'INVALID AMOUNT' TO WS-ERROR-MESSAGE
           END-IF
           
           IF LS-AMOUNT > CM-AVAILABLE-BALANCE
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'INSUFFICIENT FUNDS' TO WS-ERROR-MESSAGE
           END-IF
           
           IF LS-FROM-ACCOUNT = LS-TO-ACCOUNT
               MOVE 'N' TO WS-VALIDATION-FLAG
               MOVE 'SAME ACCOUNT' TO WS-ERROR-MESSAGE
           END-IF.

       332-DEBIT-SOURCE.
           COMPUTE WS-NEW-BALANCE = 
               CM-ACCOUNT-BALANCE - LS-AMOUNT
           
           COMPUTE WS-FEE-AMOUNT = 
               LS-AMOUNT * 0.010
           IF WS-FEE-AMOUNT < 10.00
               MOVE 10.00 TO WS-FEE-AMOUNT
           END-IF
           
           COMPUTE WS-TOTAL-DEBIT = 
               LS-AMOUNT + WS-FEE-AMOUNT
           
           SUBTRACT WS-TOTAL-DEBIT FROM CM-ACCOUNT-BALANCE
           SUBTRACT WS-TOTAL-DEBIT FROM CM-AVAILABLE-BALANCE.

       333-CREDIT-TARGET.
           MOVE LS-TO-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
               KEY IS CM-CUSTOMER-ID
               INVALID KEY
                   MOVE 'TARGET ACCOUNT NOT FOUND' TO LS-ERROR-MSG
                   MOVE 9008 TO LS-RETURN-CODE
                   PERFORM 335-REVERSE-TRANSACTION
               NOT INVALID KEY
                   ADD LS-AMOUNT TO CM-ACCOUNT-BALANCE
                   ADD LS-AMOUNT TO CM-AVAILABLE-BALANCE
                   REWRITE CUSTOMER-RECORD
           END-READ.

       334-UPDATE-BOTH.
           MOVE 'TRF' TO TRANS-TYPE
           MOVE LS-AMOUNT TO TRANS-AMOUNT
           MOVE 'C' TO TRANS-STATUS
           
           ADD 1 TO WS-TRANSACTION-COUNT
           PERFORM LOG-TRANSACTION.

       335-REVERSE-TRANSACTION.
           MOVE LS-FROM-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
           ADD LS-AMOUNT TO CM-ACCOUNT-BALANCE
           ADD LS-AMOUNT TO CM-AVAILABLE-BALANCE
           REWRITE CUSTOMER-RECORD
           
           MOVE 'REV' TO TRANS-TYPE
           MOVE 'REVERSED - TARGET NOT FOUND' TO TRANS-DESCRIPTION.

       340-CHECK-BALANCE.
           MOVE LS-FROM-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
               KEY IS CM-CUSTOMER-ID
               INVALID KEY
                   MOVE 'ACCOUNT NOT FOUND' TO LS-ERROR-MSG
                   MOVE 9009 TO LS-RETURN-CODE
               NOT INVALID KEY
                   MOVE CM-ACCOUNT-BALANCE TO LS-AMOUNT
                   MOVE 'SUCCESS' TO LS-ERROR-MSG
                   MOVE 0 TO LS-RETURN-CODE
           END-READ.

       350-GET-HISTORY.
           PERFORM 351-INITIALIZE-SEARCH
           PERFORM 352-SEARCH-TRANSACTIONS
           PERFORM 353-FORMAT-RESULTS.

       351-INITIALIZE-SEARCH.
           MOVE 1 TO TRANS-KEY
           START TRANSACTION-FILE
               KEY >= TRANS-KEY
           IF TRANS-STATUS NOT = '00'
               MOVE 'NO TRANSACTIONS' TO LS-ERROR-MSG
               MOVE 9010 TO LS-RETURN-CODE
           END-IF.

       352-SEARCH-TRANSACTIONS.
           MOVE 1 TO WS-INDEX
           PERFORM UNTIL WS-INDEX > 100 OR WS-END-OF-FILE
               READ TRANSACTION-FILE NEXT
                   AT END
                       MOVE 'Y' TO WS-EOF-FLAG
                   NOT AT END
                       IF TRANS-SOURCE-ACCOUNT = LS-FROM-ACCOUNT
                           PERFORM 354-ADD-TO-RESULTS
                       END-IF
               END-READ
               ADD 1 TO WS-INDEX
           END-PERFORM.

       353-FORMAT-RESULTS.
           MOVE 'HISTORY RETRIEVED' TO LS-ERROR-MSG
           MOVE 0 TO LS-RETURN-CODE.

       354-ADD-TO-RESULTS.
           ADD 1 TO WS-TRANSACTION-COUNT.

       360-CALCULATE-INTEREST-ADV.
           PERFORM 361-GET-ACCOUNT-DATA
           PERFORM 362-DETERMINE-RATE
           PERFORM 363-CALCULATE-COMPOUND
           PERFORM 364-APPLY-INTEREST.

       361-GET-ACCOUNT-DATA.
           MOVE LS-FROM-ACCOUNT TO CM-CUSTOMER-ID
           READ CUSTOMER-MASTER-FILE
               KEY IS CM-CUSTOMER-ID
               INVALID KEY
                   MOVE 'ACCOUNT NOT FOUND' TO LS-ERROR-MSG
                   MOVE 9011 TO LS-RETURN-CODE
               NOT INVALID KEY
                   MOVE CM-ACCOUNT-BALANCE TO WS-PRINCIPAL
                   MOVE CM-INTEREST-RATE TO WS-ANNUAL-RATE
           END-READ.

       362-DETERMINE-RATE.
           SET RATE-INDEX TO 1
           SEARCH WS-RATE-ENTRY
               AT END
                   MOVE 0.005 TO WS-ANNUAL-RATE
               WHEN WS-ACCOUNT-CODE(RATE-INDEX) = CM-ACCOUNT-TYPE
                   MOVE WS-BASE-RATE(RATE-INDEX) TO WS-ANNUAL-RATE
           END-SEARCH
           
           IF CM-ACCOUNT-BALANCE > 1000000
               COMPUTE WS-ANNUAL-RATE = 
                   WS-ANNUAL-RATE + WS-PREMIUM-RATE(RATE-INDEX)
           END-IF.

       363-CALCULATE-COMPOUND.
           COMPUTE WS-COMPOUND-FACTOR = 
               (1 + (WS-ANNUAL-RATE / 365)) ** 30
           
           COMPUTE WS-FUTURE-VALUE = 
               WS-PRINCIPAL * WS-COMPOUND-FACTOR
           
           COMPUTE WS-INTEREST-AMOUNT = 
               WS-FUTURE-VALUE - WS-PRINCIPAL.

       364-APPLY-INTEREST.
           COMPUTE WS-TAX-AMOUNT = 
               WS-INTEREST-AMOUNT * WS-TAX-RATE
           
           COMPUTE WS-NET-INTEREST = 
               WS-INTEREST-AMOUNT - WS-TAX-AMOUNT
           
           ADD WS-NET-INTEREST TO CM-ACCOUNT-BALANCE
           ADD WS-NET-INTEREST TO CM-AVAILABLE-BALANCE
           
           MOVE WS-CURRENT-DATE TO CM-LAST-INTEREST-DATE
           REWRITE CUSTOMER-RECORD
           
           MOVE 'INT' TO TRANS-TYPE
           MOVE WS-NET-INTEREST TO TRANS-AMOUNT
           PERFORM LOG-TRANSACTION.

       400-CALCULATE-INTEREST.
           MOVE 1 TO CM-CUSTOMER-ID
           START CUSTOMER-MASTER-FILE
               KEY >= CM-CUSTOMER-ID
           
           PERFORM UNTIL WS-END-OF-FILE
               READ CUSTOMER-MASTER-FILE NEXT
                   AT END
                       MOVE 'Y' TO WS-EOF-FLAG
                   NOT AT END
                       IF CM-ACCOUNT-BALANCE > 0
                           PERFORM 410-CALCULATE-DAILY-INTEREST
                           PERFORM 411-UPDATE-ACCOUNT
                       END-IF
               END-READ
           END-PERFORM.

       410-CALCULATE-DAILY-INTEREST.
           MOVE CM-ACCOUNT-BALANCE TO WS-PRINCIPAL
           MOVE CM-INTEREST-RATE TO WS-ANNUAL-RATE
           
           COMPUTE WS-DAILY-RATE = 
               WS-ANNUAL-RATE / 365
           
           COMPUTE WS-INTEREST-AMOUNT = 
               WS-PRINCIPAL * WS-DAILY-RATE
           
           COMPUTE WS-TAX-AMOUNT = 
               WS-INTEREST-AMOUNT * WS-TAX-RATE
           
           COMPUTE WS-NET-INTEREST = 
               WS-INTEREST-AMOUNT - WS-TAX-AMOUNT.

       411-UPDATE-ACCOUNT.
           ADD WS-NET-INTEREST TO CM-ACCOUNT-BALANCE
           ADD WS-NET-INTEREST TO CM-AVAILABLE-BALANCE
           
           MOVE WS-CURRENT-DATE TO CM-LAST-INTEREST-DATE
           REWRITE CUSTOMER-RECORD
           
           ADD WS-NET-INTEREST TO WS-TOTAL-AMOUNT
           ADD 1 TO WS-TRANSACTION-COUNT.

       500-APPLY-FEES-AND-TAXES.
           PERFORM 510-CALCULATE-MONTHLY-FEES
           PERFORM 520-APPLY-TAX-WITHHOLDING
           PERFORM 530-UPDATE-COMPLIANCE.

       510-CALCULATE-MONTHLY-FEES.
           IF CM-ACCOUNT-BALANCE < WS-MIN-BALANCE
               COMPUTE WS-FEE-AMOUNT = 25.00
               SUBTRACT WS-FEE-AMOUNT FROM CM-ACCOUNT-BALANCE
               PERFORM LOG-FEE
           END-IF.

       520-APPLY-TAX-WITHHOLDING.
           IF WS-TOTAL-AMOUNT > 10000
               COMPUTE WS-TAX-AMOUNT = 
                   WS-TOTAL-AMOUNT * CM-WITHHOLDING-RATE
               SUBTRACT WS-TAX-AMOUNT FROM CM-ACCOUNT-BALANCE
               PERFORM LOG-TAX
           END-IF.

       530-UPDATE-COMPLIANCE.
           MOVE WS-CURRENT-DATE TO CM-LAST-TAX-REPORT
           REWRITE CUSTOMER-RECORD.

       600-GENERATE-REPORTS.
           PERFORM 610-GENERATE-TRANSACTION-REPORT
           PERFORM 620-GENERATE-BALANCE-REPORT
           PERFORM 630-GENERATE-RISK-REPORT
           PERFORM 640-GENERATE-AUDIT-REPORT.

       610-GENERATE-TRANSACTION-REPORT.
           WRITE REPORT-LINE FROM WS-REPORT-HEADER
           MOVE WS-TRANSACTION-COUNT TO WS-DISPLAY-COUNT
           MOVE WS-TOTAL-AMOUNT TO WS-DISPLAY-AMOUNT
           WRITE REPORT-LINE FROM WS-REPORT-DETAIL.

       620-GENERATE-BALANCE-REPORT.
           MOVE 'BALANCE REPORT' TO WS-REPORT-TITLE
           PERFORM 621-PROCESS-BALANCES.

       621-PROCESS-BALANCES.
           MOVE 1 TO CM-CUSTOMER-ID
           START CUSTOMER-MASTER-FILE
               KEY >= CM-CUSTOMER-ID
           
           PERFORM UNTIL WS-END-OF-FILE
               READ CUSTOMER-MASTER-FILE NEXT
                   AT END
                       MOVE 'Y' TO WS-EOF-FLAG
                   NOT AT END
                       MOVE CM-ACCOUNT-BALANCE TO WS-DISPLAY-BALANCE
                       WRITE REPORT-LINE FROM WS-REPORT-DETAIL
               END-READ
           END-PERFORM.

       630-GENERATE-RISK-REPORT.
           MOVE 'RISK ASSESSMENT REPORT' TO WS-REPORT-TITLE
           PERFORM 631-ANALYZE-RISKS.

       631-ANALYZE-RISKS.
           MOVE 1 TO WS-INDEX
           PERFORM UNTIL WS-INDEX > 100 OR WS-END-OF-FILE
               READ CUSTOMER-MASTER-FILE NEXT
                   AT END
                       MOVE 'Y' TO WS-EOF-FLAG
                   NOT AT END
                       PERFORM 632-CALCULATE-RISK-SCORE
                       IF CM-RISK-SCORE > WS-RISK-THRESHOLD
                           PERFORM 633-FLAG-HIGH-RISK
                       END-IF
               END-READ
               ADD 1 TO WS-INDEX
           END-PERFORM.

       632-CALCULATE-RISK-SCORE.
           MOVE 50 TO CM-RISK-SCORE
           IF CM-ACCOUNT-BALANCE > 1000000
               ADD 20 TO CM-RISK-SCORE
           END-IF
           IF SUSPICIOUS-ACTIVITY
               ADD 30 TO CM-RISK-SCORE
           END-IF
           
           IF CM-RISK-SCORE > 75
               MOVE 'H' TO CM-RISK-LEVEL
           ELSE 
               IF CM-RISK-SCORE > 50
                   MOVE 'M' TO CM-RISK-LEVEL
               ELSE
                   MOVE 'L' TO CM-RISK-LEVEL
               END-IF
           END-IF.

       633-FLAG-HIGH-RISK.
           MOVE 'C' TO CM-RISK-LEVEL
           REWRITE CUSTOMER-RECORD
           PERFORM LOG-RISK-EVENT.

       640-GENERATE-AUDIT-REPORT.
           MOVE 'AUDIT TRAIL REPORT' TO WS-REPORT-TITLE
           PERFORM 641-GATHER-AUDIT-DATA.

       641-GATHER-AUDIT-DATA.
           WRITE AUDIT-RECORD FROM WS-REPORT-HEADER
           PERFORM 642-WRITE-AUDIT-ENTRIES.

       642-WRITE-AUDIT-ENTRIES.
           MOVE 'AUDIT COMPLETED' TO AUDIT-RECORD
           WRITE AUDIT-RECORD.

       700-UPDATE-AUDIT-TRAIL.
           IF WS-AUDIT-ENABLED
               PERFORM 710-LOG-ALL-ACTIONS
               PERFORM 720-ARCHIVE-AUDIT-DATA
               PERFORM 730-VERIFY-AUDIT-INTEGRITY
           END-IF.

       710-LOG-ALL-ACTIONS.
           MOVE WS-CURRENT-DATE TO AUDIT-RECORD(1:8)
           MOVE WS-CURRENT-TIME TO AUDIT-RECORD(9:6)
           MOVE LS-USER-ID TO AUDIT-RECORD(15:20)
           MOVE LS-COMMAND TO AUDIT-RECORD(35:8)
           WRITE AUDIT-RECORD.

       720-ARCHIVE-AUDIT-DATA.
           CLOSE AUDIT-TRAIL-FILE
           OPEN EXTEND AUDIT-TRAIL-FILE.

       730-VERIFY-AUDIT-INTEGRITY.
           CALL 'VERIFYAUDIT' USING AUDIT-STATUS
           IF AUDIT-STATUS NOT = '00'
               MOVE 'AUDIT INTEGRITY FAILED' TO WS-ERROR-MESSAGE
               PERFORM LOG-ERROR
           END-IF.

       800-HANDLE-SECURITY-FAILURE.
           MOVE 'SECURITY VIOLATION' TO LS-ERROR-MSG
           MOVE 9998 TO LS-RETURN-CODE
           PERFORM LOG-SECURITY-EVENT
           PERFORM LOCK-ACCOUNT
           PERFORM NOTIFY-SECURITY-TEAM.

       LOCK-ACCOUNT.
           MOVE 'Y' TO CM-ACCOUNT-LOCKED
           REWRITE CUSTOMER-RECORD.

       NOTIFY-SECURITY-TEAM.
           DISPLAY 'SECURITY ALERT: ' LS-USER-ID
           CALL 'SECURITYALERT' USING LS-USER-ID
                                    WS-CURRENT-DATE
                                    WS-CURRENT-TIME.

       900-TERMINATE.
           CLOSE CUSTOMER-MASTER-FILE
           CLOSE TRANSACTION-FILE
           CLOSE AUDIT-TRAIL-FILE
           CLOSE TEMPORARY-WORK-FILE
           CLOSE REPORT-FILE
           
           PERFORM 910-GENERATE-PERFORMANCE-REPORT
           PERFORM 920-CLEANUP-TEMP-FILES
           PERFORM 930-UPDATE-METRICS.

       910-GENERATE-PERFORMANCE-REPORT.
           COMPUTE WS-ELAPSED-TIME = 
               WS-END-TIME - WS-START-TIME
           
           IF WS-ELAPSED-TIME > 0
               COMPUTE WS-TRANSACTIONS-PER-SECOND = 
                   WS-TRANSACTION-COUNT / WS-ELAPSED-TIME
           END-IF
           
           DISPLAY 'PERFORMANCE METRICS:'
           DISPLAY 'TRANSACTIONS: ' WS-TRANSACTION-COUNT
           DISPLAY 'ELAPSED TIME: ' WS-ELAPSED-TIME
           DISPLAY 'TPS: ' WS-TRANSACTIONS-PER-SECOND.

       920-CLEANUP-TEMP-FILES.
           CALL 'DELETETEMP' USING WS-JOB-ID.

       930-UPDATE-METRICS.
           CALL 'UPDATEMETRICS' USING WS-TRANSACTION-COUNT
                                    WS-TOTAL-AMOUNT
                                    WS-ELAPSED-TIME.

       LOG-ERROR.
           MOVE WS-ERROR-CODE TO AUDIT-RECORD(1:4)
           MOVE WS-ERROR-MESSAGE TO AUDIT-RECORD(5:100)
           MOVE WS-ERROR-SEVERITY TO AUDIT-RECORD(105:1)
           WRITE AUDIT-RECORD.

       LOG-TRANSACTION.
           MOVE TRANS-REFERENCE TO AUDIT-RECORD(1:20)
           MOVE TRANS-TYPE TO AUDIT-RECORD(21:3)
           MOVE TRANS-AMOUNT TO AUDIT-RECORD(24:12)
           MOVE TRANS-STATUS TO AUDIT-RECORD(36:1)
           WRITE AUDIT-RECORD.

       LOG-SECURITY-EVENT.
           MOVE 'SECURITY' TO AUDIT-RECORD(1:8)
           MOVE LS-USER-ID TO AUDIT-RECORD(9:20)
           MOVE WS-SECURITY-FLAG TO AUDIT-RECORD(29:1)
           WRITE AUDIT-RECORD.

       LOG-FEE.
           MOVE 'FEE' TO AUDIT-RECORD(1:3)
           MOVE WS-FEE-AMOUNT TO AUDIT-RECORD(4:12)
           WRITE AUDIT-RECORD.

       LOG-TAX.
           MOVE 'TAX' TO AUDIT-RECORD(1:3)
           MOVE WS-TAX-AMOUNT TO AUDIT-RECORD(4:12)
           WRITE AUDIT-RECORD.

       LOG-RISK-EVENT.
           MOVE 'RISK' TO AUDIT-RECORD(1:4)
           MOVE CM-RISK-LEVEL TO AUDIT-RECORD(5:1)
           MOVE CM-RISK-SCORE TO AUDIT-RECORD(6:3)
           WRITE AUDIT-RECORD.

       CALCULATE-FRAUD-SCORE.
           MOVE 0 TO WS-FRAUD-SCORE
           IF TRANS-AMOUNT > 100000
               ADD 30 TO WS-FRAUD-SCORE
           END-IF
           IF TRANS-CHANNEL = 'O' AND TRANS-AMOUNT > 50000
               ADD 25 TO WS-FRAUD-SCORE
           END-IF
           IF WS-INDEX > 10
               ADD 20 TO WS-FRAUD-SCORE
           END-IF.

       END PROGRAM ULTIMATE-BANKING-SYSTEM.
