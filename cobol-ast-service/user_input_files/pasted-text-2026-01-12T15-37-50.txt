import re
from typing import Dict, List

class BusinessRuleExtractor:
    """Extrait les règles métier du code COBOL"""
    
    def __init__(self, cobol_code: str):
        self.code = cobol_code
        self.rules = []
    
    def extract_calculation_rules(self):
        """Extrait les règles de calcul"""
        # Détection des formules de calcul
        calc_patterns = [
            r'COMPUTE\s+(\w+-?\w+)\s*=\s*(.+)',
            r'MULTIPLY\s+(.+)\s+BY\s+(.+)',
            r'DIVIDE\s+(.+)\s+INTO\s+(.+)'
        ]
        
        for pattern in calc_patterns:
            matches = re.finditer(pattern, self.code, re.IGNORECASE)
            for match in matches:
                rule = {
                    'type': 'CALCULATION',
                    'target': match.group(1),
                    'formula': match.group(2),
                    'line': match.start()
                }
                self.rules.append(rule)
    
    def extract_business_constants(self):
        """Extrait les constantes métier"""
        # Recherche des valeurs initiales
        const_pattern = r'(\d+\.?\d*)\s+PIC'
        matches = re.finditer(const_pattern, self.code)
        
        constants = {}
        for match in matches:
            var_name = self._find_variable_name(match.start())
            if var_name:
                constants[var_name] = {
                    'value': match.group(1),
                    'type': self._infer_type(match.group(0))
                }
        return constants

from sympy import symbols, sympify
import math

class FormulaTransformer:
    """Transforme les formules COBOL en Python"""
    
    COBOL_TO_PYTHON_OPS = {
        '**': '**',          # Exponentiation
        '*': '*',            # Multiplication
        '/': '/',            # Division
        '+': '+',            # Addition
        '-': '-',            # Soustraction
        '>': '>',            # Supérieur
        '<': '<',            # Inférieur
        '=': '==',           # Égalité
        'NOT =': '!=',       # Différent
    }
    
    def transform_cobol_formula(self, cobol_formula: str) -> str:
        """Transforme une formule COBOL en Python"""
        
        # 1. Nettoyage
        formula = cobol_formula.upper()
        formula = formula.replace('COMPUTE', '').strip()
        
        # 2. Remplacement des opérateurs
        for cobol_op, python_op in self.COBOL_TO_PYTHON_OPS.items():
            formula = formula.replace(cobol_op, python_op)
        
        # 3. Gestion des arrondis COBOL
        if 'ROUNDED' in formula:
            formula = formula.replace('ROUNDED', '')
            formula = f"round({formula}, 2)"
        
        # 4. Conversion des noms de variables
        formula = self._convert_variable_names(formula)
        
        return formula
    
    def _convert_variable_names(self, formula: str) -> str:
        """Convertit WS-VARIABLE en variable_python"""
        # Détection des variables COBOL (commençant par WS-)
        variables = re.findall(r'WS-\w+', formula)
        
        for var in variables:
            python_var = var.replace('WS-', '').lower().replace('-', '_')
            formula = formula.replace(var, python_var)
        
        return formula

    def get(self, id: str) -> Optional[T]:
        pass

    def save(self, entity: T) -> None:

        if v <= 0 or v > 1:  # 100% max

        return v
    
    @validator('period_days')
    def validate_period(cls, v):
        if v < 0 or v > 366:
            raise ValueError('La période doit être entre 0 et 366 jours')
        return v

class InterestCalculationService:
    """Service avec validation métier complète"""
    
    def __init__(self, repository: InterestCalculationRepository):
        self.repository = repository
        self.logger = logging.getLogger(__name__)
    
    def calculate_interest(self, request: InterestCalculationRequest) -> Dict:
        """Calcule avec toutes les règles métier"""

        self._validate_business_rules(request)

        # 3. Application de règles spécifiques
        calculation = self._apply_business_rules(calculation, request)

class BusinessLogicMonitor:
    """Surveille les écarts entre COBOL et Python"""
    
    def __init__(self):
        self.discrepancies = []
    
    def log_discrepancy(self, cobol_value, python_value, context):
        """Enregistre un écart détecté"""
        discrepancy = {
            'cobol': cobol_value,
            'python': python_value,
            'difference': abs(cobol_value - python_value),
            'context': context,
            'timestamp': datetime.now()
        }
        self.discrepancies.append(discrepancy)
        
        if discrepancy['difference'] > Decimal('0.10'):
            self.alert_team(discrepancy)
    
    def generate_report(self):
        """Génère un rapport d'écarts"""
        report = {
            'total_calculations': len(self.discrepancies),
            'within_tolerance': sum(1 for d in self.discrepancies 
                                   if d['difference'] <= Decimal('0.01')),
            'major_discrepancies': [d for d in self.discrepancies 
                                   if d['difference'] > Decimal('1.00')]
        }
        return report

class EnhancedBusinessLogicTransformer:
    """Transformer amélioré pour CodeSwitch"""
    
    def __init__(self):
        self.extractor = BusinessRuleExtractor()
        self.transformer = FormulaTransformer()
        self.validator = BusinessLogicValidator()
        self.monitor = BusinessLogicMonitor()
    
    def transform_with_validation(self, cobol_code: str) -> Dict:
        """Transforme avec validation approfondie"""
        
        # 1. Extraction des règles
        rules = self.extractor.extract_all_rules(cobol_code)
        
        # 2. Transformation
        python_code = self._transform_to_python(cobol_code, rules)
        
        # 3. Génération de tests
        test_cases = self.validator.generate_test_cases(cobol_code, python_code)
        
        # 4. Validation croisée
        results = self._cross_validate(cobol_code, python_code, test_cases)
        
        # 5. Génération de documentation
        documentation = self._generate_documentation(rules, results)
        
        return {
            'python_code': python_code,
            'test_cases': test_cases,
            'validation_results': results,
            'documentation': documentation,
            'confidence_score': self._calculate_confidence(results)
        }
    
    def _calculate_confidence(self, validation_results):
        """Calcule un score de confiance basé sur les tests"""
        passed = sum(1 for r in validation_results if r['passed'])
        total = len(validation_results)
        
        if total == 0:
            return 0.0
        
        confidence = passed / total
        
        # Pénalités pour certains types d'écarts
        for result in validation_results:
            if not result['passed'] and result['difference'] > 1.0:
                confidence *= 0.8  # Pénalité pour gros écart
        
        return confidence
